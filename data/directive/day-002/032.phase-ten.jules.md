import { TraumaModel } from '../trauma.model.js'
import type TraumaBit from '../fce/trauma.bit.js'

async function emitConsole(ste: State | undefined, src: string): Promise<void> {
    if (ste && typeof ste.hunt === 'function') {
        await ste
            .hunt('[Console action] Update Console', { idx: 'cns00', src })
            .catch(() => {})
    }
    const lib =
        (globalThis as any).LIBRARY ||
        (typeof global !== 'undefined' ? (global as any).LIBRARY : undefined)
    if (lib && typeof lib.hunt === 'function') {
        await lib
            .hunt('[Console action] Update Console', { idx: 'cns00', src })

export function assertBasisPoint(val: number, field: string): void {
    if (!Number.isInteger(val)) {
        throw new TypeError(
            `[FLOAT_PROHIBITED] ${field} must be a discrete integer. Received: ${val}`,
        )
    }
    if (val < 0 || val > 10000) {
        throw new RangeError(
            `[OUT_OF_BOUNDS] ${field} out of bounds [0, 10000]: ${val}`,
        )
    }
}

export function deriveLocomotionTier(
    legsAvailable: number,
    painBps: number,
    fatigueBps: number,
): LocomotionTier {
    if (legsAvailable === 0) {
        return 'IMMOBILIZED'
    }
    if (legsAvailable === 1 || painBps >= 8000 || fatigueBps >= 9000) {
        return 'LIMPING'
    }
    return 'UNRESTRICTED'
}

export const applyTraumaAction = async (
    cpy: TraumaModel,
    bal: TraumaBit,
    ste?: State,
): Promise<TraumaModel> => {
    const delta: TraumaDeltaPayload | undefined = bal.dat?.delta
    if (!delta || !delta.entityId) {
        const errorMsg = '[INVALID_DELTA] Missing delta or entityId'
        await emitConsole(ste, `:: [FAIL] ${errorMsg}`)
        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'apply-trauma-fail',
                    val: 0,
                    dat: { error: errorMsg },
                },
            })
        }
        return cpy
    }

    try {
        const current = cpy.entities[delta.entityId] || {
            entityId: delta.entityId,
            armsAvailable: 2,
            legsAvailable: 2,
            concussionLevel: 0,
            fatigueBps: 0,
            painBps: 0,
            refusalStrainBps: 0,
            locomotionTier: 'UNRESTRICTED',
            updatedAtMs: delta.timestampMs || 1700000000000,
        }

        const newArms = Math.max(0, Math.min(2, current.armsAvailable + (delta.deltaArms || 0)))
        const newLegs = Math.max(0, Math.min(2, current.legsAvailable + (delta.deltaLegs || 0)))
        const newConcussion = Math.max(0, Math.min(3, current.concussionLevel + (delta.deltaConcussion || 0)))

        const rawFatigue = current.fatigueBps + (delta.deltaFatigueBps || 0)
        const rawPain = current.painBps + (delta.deltaPainBps || 0)
        const rawRefusal = current.refusalStrainBps + (delta.deltaRefusalStrainBps || 0)

        const clampedFatigue = Math.max(0, Math.min(10000, rawFatigue))
        const clampedPain = Math.max(0, Math.min(10000, rawPain))
        const clampedRefusal = Math.max(0, Math.min(10000, rawRefusal))

        assertBasisPoint(clampedFatigue, 'fatigueBps')
        assertBasisPoint(clampedPain, 'painBps')
        assertBasisPoint(clampedRefusal, 'refusalStrainBps')

        const newTier = deriveLocomotionTier(newLegs, clampedPain, clampedFatigue)

        const updatedState: SomaticTraumaState = {
            entityId: delta.entityId,
            armsAvailable: newArms,
            legsAvailable: newLegs,
            concussionLevel: newConcussion,
            fatigueBps: clampedFatigue,
            painBps: clampedPain,
            refusalStrainBps: clampedRefusal,
            locomotionTier: newTier,
            updatedAtMs: delta.timestampMs || current.updatedAtMs,
        }

        cpy.entities[delta.entityId] = updatedState

        await emitConsole(
            ste,
            `>> [TRAUMA:APPLY] [${updatedState.entityId}] Arms: ${updatedState.armsAvailable}/2 | Legs: ${updatedState.legsAvailable}/2 | Pain: ${updatedState.painBps} bps | Fatigue: ${updatedState.fatigueBps} bps | Locomotion: [${updatedState.locomotionTier}]`,
        )

        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'apply-trauma-pass',
                    val: 1,
                    dat: { state: updatedState },
                },
            })
        }
    } catch (err: any) {
        const errorMsg = err.message || String(err)
        await emitConsole(ste, `:: [FAIL] Trauma application aborted: ${errorMsg}`)
        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'apply-trauma-fail',
                    val: 0,
                    dat: { error: errorMsg },
                },
            })
        }
    }

    return cpy
}

export const decayFatigueTraumaAction = async (
    cpy: TraumaModel,
    bal: TraumaBit,
    ste?: State,
): Promise<TraumaModel> => {
    const entityId = bal.dat?.entityId || bal.src
    const decayBps = bal.dat?.decayBps ?? 500
    const timestampMs = bal.dat?.timestampMs || 1700000000000

    if (!entityId || !cpy.entities[entityId]) {
        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'decay-fatigue-fail',
                    val: 0,
                    dat: { error: `Entity '${entityId}' not found in trauma registry` },
                },
            })
        }
        return cpy
    }

    try {
        assertBasisPoint(decayBps, 'decayBps')
        const current = cpy.entities[entityId]
        const newFatigue = Math.max(0, current.fatigueBps - decayBps)
        const newTier = deriveLocomotionTier(current.legsAvailable, current.painBps, newFatigue)

        const updated: SomaticTraumaState = {
            ...current,
            fatigueBps: newFatigue,
            locomotionTier: newTier,
            updatedAtMs: timestampMs,
        }
        cpy.entities[entityId] = updated

        await emitConsole(
            ste,
            `>> [TRAUMA:DECAY] [${entityId}] Fatigue decayed: ${current.fatigueBps} -> ${newFatigue} bps`,
        )

        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'decay-fatigue-pass',
                    val: newFatigue,
                    dat: { state: updated },
                },
            })
        }
    } catch (err: any) {
        if (bal.slv) {
            bal.slv({
                trmBit: {
                    idx: 'decay-fatigue-fail',
                    val: 0,
                    dat: { error: err.message || String(err) },
                },
            })
        }
    }

    return cpy
}

export const clampBasisPointsTraumaAction = (
    cpy: TraumaModel,
    bal: TraumaBit,
    _ste?: State,
): TraumaModel => {
    const rawVal = bal.val ?? 0
    let clamped = 0
    try {
        assertBasisPoint(rawVal, 'rawVal')
        clamped = rawVal
    } catch {
        clamped = Math.max(0, Math.min(10000, Math.round(rawVal)))
    }

    if (bal.slv) {
        bal.slv({
            trmBit: {
                idx: 'clamp-basis-points-pass',
                val: clamped,
            },
        })
    }
    return cpy
}

export const getSomaticStateAction = (
    cpy: TraumaModel,
    bal: TraumaBit,
    _ste?: State,
): TraumaModel => {
    const entityId = bal.dat?.entityId || bal.src
    const state = entityId ? cpy.entities[entityId] : undefined

    if (bal.slv) {
        bal.slv({
            trmBit: {
                idx: state ? 'get-somatic-pass' : 'get-somatic-not-found',
                val: state ? 1 : 0,
                dat: { state },
            },
        })
    }
    return cpy
}

import clone from 'clone-deep'

        case Act.GET_SOMATIC_STATE:
            return Buzz.getSomaticStateAction(clone(model), act.bale as any, state)

        default:
            return model
    }

export function compileSomaticMask(
    entityId: string,
    wounds: readonly WoundRecord[],
    armsAvailable: number,
    legsAvailable: number,
    locomotionTier: string,
    timestampMs: number = 1700000000000,
): SomaticMask {
    const banned = new Set<string>()
    const openWoundsSet = new Set<string>()

    // 1. Roll-up from open wounds & condition tags
    for (const w of wounds) {
        openWoundsSet.add(w.conditionTag)
        const families = CONDITION_VERB_ROLLUP[w.conditionTag] || []
        for (const fam of families) {
            for (const verb of VERB_FAMILIES[fam]) {
                banned.add(verb)
            }
        }
    }

    // 2. Anatomical limb constraints
    if (armsAvailable < 2) {
        for (const verb of VERB_FAMILIES.BIMANUAL_GRIP) {
            banned.add(verb)
        }
        banned.add('HEAVY_REACH')
    }

    if (armsAvailable === 0) {
        for (const verb of VERB_FAMILIES.MANIPULATION_FINE) {
            banned.add(verb)
        }
        banned.add('GRASP')
    }

    // 3. Locomotion tier constraints
    if (locomotionTier === 'LIMPING') {
        for (const verb of VERB_FAMILIES.LOCOMOTION_RAPID) {
            banned.add(verb)
        }
    } else if (locomotionTier === 'IMMOBILIZED') {
        for (const verb of VERB_FAMILIES.LOCOMOTION_RAPID) {
            banned.add(verb)
        }
        for (const verb of VERB_FAMILIES.LOCOMOTION_VERTICAL) {
            banned.add(verb)
        }
        banned.add('STAND')
        banned.add('WALK')
    }

    const sortedOpenWounds = Object.freeze(Array.from(openWoundsSet).sort())
    const sortedBannedVerbs = Object.freeze(Array.from(banned).sort())

    return {
        entityId,
        armsAvailable,
        legsAvailable,
        locomotionTier: locomotionTier as any,
        openWounds: sortedOpenWounds,
        bannedKineticVerbs: sortedBannedVerbs,
        compiledAtMs: timestampMs,
    }
}

export const addWoundAction = async (
    cpy: WoundsModel,
    bal: WoundsBit,
    ste?: State,
): Promise<WoundsModel> => {
    const wound: WoundRecord | undefined = bal.dat?.wound
    if (!wound || !wound.entityId || !wound.conditionTag) {
        const errorMsg = '[INVALID_WOUND] Missing entityId or conditionTag'
        await emitConsole(ste, `:: [FAIL] ${errorMsg}`)
        if (bal.slv) {
            bal.slv({
                wndBit: {
                    idx: 'add-wound-fail',
                    val: 0,
                    dat: { error: errorMsg },
                },
            })
        }
        return cpy
    }

    if (!cpy.woundsByEntity[wound.entityId]) {
        cpy.woundsByEntity[wound.entityId] = []
    }
    cpy.woundsByEntity[wound.entityId].push(wound)

    await emitConsole(
        ste,
        `>> [WOUNDS:ADD] [${wound.entityId}] Incurred wound '${wound.woundId}': [${wound.conditionTag}] at ${wound.locus} (${wound.severityBps} bps)`,
    )

    if (bal.slv) {
        bal.slv({
            wndBit: {
                idx: 'add-wound-pass',
                val: cpy.woundsByEntity[wound.entityId].length,
                dat: { wound },
            },
        })
    }
    return cpy
}

export const compileSomaticMaskWoundsAction = async (
    cpy: WoundsModel,
    bal: WoundsBit,
    ste?: State,
): Promise<WoundsModel> => {
    const entityId = bal.dat?.entityId || bal.src
    if (!entityId) {
        if (bal.slv) {
            bal.slv({
                wndBit: {
                    idx: 'compile-mask-fail',
                    val: 0,
                    dat: { error: '[MISSING_ENTITY_ID] entityId required to compile mask' },
                },
            })
        }
        return cpy
    }

    const wounds = cpy.woundsByEntity[entityId] || []
    const arms = bal.dat?.armsAvailable ?? 2
    const legs = bal.dat?.legsAvailable ?? 2
    const tier = bal.dat?.locomotionTier || 'UNRESTRICTED'
    const timestampMs = bal.dat?.timestampMs || 1700000000000

    const mask = compileSomaticMask(entityId, wounds, arms, legs, tier, timestampMs)
    cpy.masksByEntity[entityId] = mask

    await emitConsole(
        ste,
        `>> [PANOPTICON:MASK] [${entityId}] Compiled negative motor mask: ${mask.bannedKineticVerbs.length} banned verbs across ${mask.openWounds.length} open wounds.`,
    )

    if (bal.slv) {
        bal.slv({
            mask,
            wndBit: {
                idx: 'compile-mask-pass',
                val: mask.bannedKineticVerbs.length,
                dat: { mask },
            },
        })
    }
    return cpy
}

export const assertKineticVerbsWoundsAction = async (
    cpy: WoundsModel,
    bal: WoundsBit,
    ste?: State,
): Promise<WoundsModel> => {
    const query: KineticAuditQuery | undefined = bal.dat?.auditQuery
    if (!query || !query.entityId || !query.proposedVerbs) {
        const errorMsg = '[INVALID_AUDIT_QUERY] Missing entityId or proposedVerbs'
        await emitConsole(ste, `:: [FAIL] ${errorMsg}`)
        if (bal.slv) {
            bal.slv({
                wndBit: {
                    idx: 'assert-kinetic-verbs-fail',
                    val: 0,
                    dat: { error: errorMsg },
                },
            })
        }
        return cpy
    }

    const mask = cpy.masksByEntity[query.entityId] || {
        entityId: query.entityId,
        armsAvailable: 2,
        legsAvailable: 2,
        locomotionTier: 'UNRESTRICTED',
        openWounds: [],
        bannedKineticVerbs: [],
        compiledAtMs: 1700000000000,
    }

    // O(1) Set-lookup for fast collision detection
    const bannedLookup = new Set(mask.bannedKineticVerbs)
    const breached: string[] = []

    for (const v of query.proposedVerbs) {
        const upper = v.trim().toUpperCase()
        if (bannedLookup.has(upper)) {
            breached.push(upper)
        }
    }

    const permitted = breached.length === 0
    const receipt: KineticAuditReceipt = {
        entityId: query.entityId,
        permitted,
        breachedVerbs: Object.freeze(breached),
        totalBannedVerbsCount: mask.bannedKineticVerbs.length,
        timestampMs: 1700000000000,
    }

    if (!permitted) {
        await emitConsole(
            ste,
            `:: [SOMATIC_CAPABILITY_BREACH] [${query.entityId}] Blocked motor verbs at T_0 = $0.00: [${breached.join(', ')}]`,
        )
    } else {
        await emitConsole(
            ste,
            `>> [PANOPTICON:PASS] [${query.entityId}] Proposed motor verbs cleared in O(1) set-lookup.`,
        )
    }

    if (bal.slv) {
        bal.slv({
            auditReceipt: receipt,
            wndBit: {
                idx: permitted ? 'assert-kinetic-verbs-pass' : 'somatic-capability-breach',
                val: permitted ? 1 : -1,
                src: permitted ? '[OK]' : '[SOMATIC_CAPABILITY_BREACH]',
                dat: { auditReceipt: receipt },
            },
        })
    }

    return cpy
}

import clone from 'clone-deep'
import * as Act from './wounds.action.js'
import { WoundsModel } from './wounds.model.js'
import * as Buzz from './wounds.buzzer.js'
import type State from '../99.core/state.js'

export function reducer(
    model: WoundsModel = new WoundsModel(),
    act: Act.Actions,
    state?: State,
): WoundsModel | Promise<WoundsModel> {
    switch (act.type) {
        case Act.INIT_WOUNDS:
            return Buzz.initWounds(clone(model), act.bale, state)

        case Act.UPDATE_WOUNDS:
            return Buzz.updateWounds(clone(model), act.bale, state)

        case Act.ADD_WOUND:
            return Buzz.addWoundAction(clone(model), act.bale as any, state)

        case Act.COMPILE_SOMATIC_MASK_WOUNDS:
            return Buzz.compileSomaticMaskWoundsAction(clone(model), act.bale as any, state)

        case Act.ASSERT_KINETIC_VERBS_WOUNDS:
            return Buzz.assertKineticVerbsWoundsAction(clone(model), act.bale as any, state)

        default:
            return model
    }
}
