  public async appendEvent(event: EventLogRecord): Promise<void> {

    }

    const stmt = this.db.prepare(`

        event_id, stream_id, tick, type, version, actor_id, target_id,
        seed, payload_json, prev_hash, state_hash, timestamp_ms
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)

      event.actor_id,
      event.target_id,
