import { Action } from '../99.core/interface/action.interface.js'
import type ArithmeticBit from './fce/arithmetic.bit.js'
import {
    type BasisPoint,
    assertBasisPoint,
    toBasisPoint,
    type ScalingDivisionResult,
} from './fce/arithmetic.interface.js'

export const MAX_INTERMEDIATE_PRODUCT = 100000000

export const INIT_ARITHMETIC = '[Arithmetic action] Init Arithmetic'
export class InitArithmetic implements Action {
    readonly type = INIT_ARITHMETIC
    constructor(public bale: ArithmeticBit) {}
}

export const UPDATE_ARITHMETIC = '[Arithmetic action] Update Arithmetic'
export class UpdateArithmetic implements Action {
    readonly type = UPDATE_ARITHMETIC
    constructor(public bale: ArithmeticBit) {}
}

export type Actions = InitArithmetic | UpdateArithmetic

export function divRoundHalfEven(
    numerator: number,
    denominator: number,
): number {
    if (denominator === 0) {
        throw new RangeError('[DIV_ZERO] Division by zero is prohibited')
    }
    if (!Number.isInteger(numerator) || !Number.isInteger(denominator)) {
        throw new TypeError('[FLOAT_PROHIBITED] Inputs must be discrete integers')
    }

    const sign =
        (numerator < 0 && denominator > 0) ||
        (numerator > 0 && denominator < 0)
            ? -1
            : 1
    const absN = Math.abs(numerator)
    const absD = Math.abs(denominator)

    const quotient = Math.floor(absN / absD)
    const remainder = absN % absD
    const remainderDoubled = remainder * 2

    let roundedMagnitude: number
    if (remainderDoubled < absD) {
        roundedMagnitude = quotient
    } else if (remainderDoubled > absD) {
        roundedMagnitude = quotient + 1
    } else {
        roundedMagnitude = quotient % 2 === 0 ? quotient : quotient + 1
    }

    return sign === -1 && roundedMagnitude !== 0
        ? -roundedMagnitude
        : roundedMagnitude
}

export function evaluateScalingDivision(
    a: number,
    b: number,
): ScalingDivisionResult {
    assertBasisPoint(a)
    assertBasisPoint(b)

    const intermediate = a * b
    if (intermediate > MAX_INTERMEDIATE_PRODUCT) {
        throw new RangeError(
            `[PRODUCT_OVERFLOW] Intermediate product ${intermediate} exceeded ceiling ${MAX_INTERMEDIATE_PRODUCT}`,
        )
    }

    const quotient = Math.floor(intermediate / 10000)
    const remainder = intermediate % 10000
    const tieDetected = remainder === 5000

    let roundedOutput: number
    if (remainder < 5000) {
        roundedOutput = quotient
    } else if (remainder > 5000) {
        roundedOutput = quotient + 1
    } else {
        roundedOutput = quotient % 2 === 0 ? quotient : quotient + 1
    }

    return {
        quotient,
        remainder,
        tieDetected,
        roundedOutput,
    }
}

export function scaleBasisPoints(a: number, b: number): BasisPoint {
    const result = evaluateScalingDivision(a, b)
    return toBasisPoint(result.roundedOutput)
}

export function scaleBasisPointsChained(
    value: number,
    ...multipliers: number[]
): BasisPoint {
    if (multipliers.length === 0) {
        throw new RangeError(
            '[EMPTY_MULTIPLIERS] Chained scaling requires at least one multiplier',
        )
    }

    let current = toBasisPoint(value)
    for (const multiplier of multipliers) {
        current = scaleBasisPoints(current, multiplier)
    }
    return current
}

function canonicalize(obj: any): string {
    if (obj === null || typeof obj !== 'object') {
        return JSON.stringify(obj)
    }
    if (Array.isArray(obj)) {
        return `[${obj.map(canonicalize).join(',')}]`
    }
    const sortedKeys = Object.keys(obj).sort((a, b) => (a < b ? -1 : a > b ? 1 : 0))
    const parts = sortedKeys.map((k) => `${JSON.stringify(k)}:${canonicalize(obj[k])}`)
    return `{${parts.join(',')}}`
}

var sim: {
    hunt: any
    state: any

    hunt: null,
    state: null,

sim.hunt = (typ: any, obj: any) => {
    return host(obj, typ)

var host = (obj: any, typ: any) => {
    init()

    var slv: any
    const promo = new Promise((rslv) => (slv = rslv))

    if (obj == null) obj = {}
    if (obj.slv == null) obj.slv = (val0: any) => slv(val0)

    sim.state.dispatch({ type: typ, bale: obj })
    return promo
}

var init = () => {
    if (sim.state != null) return
    sim.state = new State()
    sim.state.pivot = sim
    sim.state.hunt = sim.hunt
    for (var k in Import.list) new (Import.list as any)[k](sim.state)
}
