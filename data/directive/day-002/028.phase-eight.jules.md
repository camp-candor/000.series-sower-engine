    cpy.hRoot = hRoot

export const verifyGenesisPrimordialAction = async (
    cpy: PrimordialModel,
    bal: PrimordialBit,
    ste?: State,
): Promise<PrimordialModel> => {
    let isSeeded = false
    let entitiesCount = 0
    let recordedHash = cpy.hRoot

    if (ste && typeof ste.hunt === 'function') {
        const eventRes: any = await ste.hunt(ActDbs.QUERY_DATABASE, {
            dat: { table: 'event_log', key: 'evt_genesis_000' },
        })
        const entityRes: any = await ste.hunt(ActDbs.QUERY_DATABASE, {
            dat: { table: 'entities_live' },
        })

        const hasGenesisEvent = eventRes?.databaseBit?.val === 1 && eventRes?.databaseBit?.dat?.record
        entitiesCount = entityRes?.databaseBit?.val || 0
        isSeeded = Boolean(hasGenesisEvent && entitiesCount > 0)
        if (hasGenesisEvent) {
            recordedHash = eventRes.databaseBit.dat.record.state_hash || recordedHash
        }
    } else {
        isSeeded = cpy.genesisSeeded
    }

    cpy.genesisSeeded = isSeeded

    await emitConsole(
        ste,
        `>> [PRIMORDIAL:VERIFY] [OK] Genesis state verified: ${isSeeded ? 'SEEDED' : 'EMPTY'} (${entitiesCount} entities online). Zero rows overwritten.`,
    )

    if (bal.slv) {
        bal.slv({
            prmBit: {
                idx: isSeeded ? 'verify-genesis-pass' : 'verify-genesis-empty',
                val: isSeeded ? 1 : 0,
                src: recordedHash,
                dat: { isSeeded, entitiesCount, hRoot: recordedHash },
            },
            hRoot: recordedHash,
        })
    }

    return cpy
}

/**
 * 13.primordial.unit - Primordial Reducer Router

import clone from 'clone-deep'
import * as Act from './primordial.action.js'
import { PrimordialModel } from './primordial.model.js'
import * as Buzz from './primordial.buzzer.js'
import type State from '../99.core/state.js'

export function reducer(
    model: PrimordialModel = new PrimordialModel(),
    act: Act.Actions,
    state?: State,
): PrimordialModel | Promise<PrimordialModel> {
    switch (act.type) {
        case Act.INIT_PRIMORDIAL:
            return Buzz.initPrimordial(clone(model), act.bale, state)

        case Act.UPDATE_PRIMORDIAL:
            return Buzz.updatePrimordial(clone(model), act.bale, state)

        case Act.COMPUTE_HROOT_PRIMORDIAL:
            return Buzz.computeHrootPrimordialAction(clone(model), act.bale as any, state)

        case Act.SEED_PRIMORDIAL:
            return Buzz.seedPrimordialAction(clone(model), act.bale as any, state)

        case Act.VERIFY_GENESIS_PRIMORDIAL:
            return Buzz.verifyGenesisPrimordialAction(clone(model), act.bale as any, state)

        default:
            return model
    }
}
