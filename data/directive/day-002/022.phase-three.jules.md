    return await stmt.bind(sceneId).first<SceneRecord>();
  }

  /**
   * Queries the passive graph centrality telemetry view to evaluate narrative in-degree.
   */
  public async getFactCentralityTelemetry(): Promise<FactCentralityTelemetry[]> {
    const stmt = this.db.prepare(`
      SELECT * FROM v_fact_centrality_telemetry
      ORDER BY active_dependent_scenes_count DESC
    `);
    const res = await stmt.all<FactCentralityTelemetry>();
    return res.results ?? [];
  }
}
