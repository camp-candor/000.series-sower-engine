async function emitConsole(ste: State | undefined, src: string): Promise<void> {
    if (ste && typeof ste.hunt === 'function') {
        await ste
            .hunt(UPDATE_CONSOLE, { idx: 'cns00', src })
            .catch(() => {})
    }
    const lib =
        (globalThis as any).LIBRARY ||
        (typeof global !== 'undefined' ? (global as any).LIBRARY : undefined)
    if (lib && typeof lib.hunt === 'function') {
        await lib
            .hunt(UPDATE_CONSOLE, { idx: 'cns00', src })

    }
}

export const initMenu = async (
    cpy: MenuModel,
    bal?: MenuBit,
    ste?: State,
): Promise<MenuModel> => {
    cpy.activeSelection = ''
    cpy.lastActionTimestamp = bal?.dat?.timestampMs || 1700000000000

    await emitConsole(
        ste,
        '>> [FLIGHT_DECK:READY] 300.ENGINE SIMULATION COCKPIT ONLINE [OK]',
    )

    const lib =
        (globalThis as any).LIBRARY ||
        (typeof global !== 'undefined' ? (global as any).LIBRARY : undefined)

    if (!lib || typeof lib.hunt !== 'function') {
        cpy.isHeadless = true
        if (bal?.slv) {
            bal.slv({
                mnuBit: {
                    idx: 'init-menu-headless-pass',
                    val: 1,
                    src: 'HEADLESS',
                },
            })
        }
        return cpy
    }

    cpy.isHeadless = false
    await updateMenu(cpy, bal, ste)

    if (bal?.slv) {
        bal.slv({
            mnuBit: {
                idx: 'init-menu-pass',
                val: 1,
                src: 'READY',
            },
        })
    }
    return cpy
}

export const updateMenu = async (
    cpy: MenuModel,
    bal?: MenuBit,
    ste?: State,
): Promise<MenuModel> => {
    const lib =
        (globalThis as any).LIBRARY ||
        (typeof global !== 'undefined' ? (global as any).LIBRARY : undefined)

    if (!lib || typeof lib.hunt !== 'function') {
        cpy.isHeadless = true
        if (bal?.slv) {
            bal.slv({
                mnuBit: {
                    idx: 'update-menu-headless-pass',
                    val: 1,
                    src: 'HEADLESS',
                },
            })
        }
        return cpy
    }

    try {
        // Allocate left 4 columns in 12x12 grid
        const gridBit: any = await lib.hunt(UPDATE_GRID, {
            x: 0,
            y: 4,
            xSpan: 4,
            ySpan: 8,
        })

        const choiceBit: any = await lib.hunt(OPEN_CHOICE, {
            dat: { clr0: 'black', clr1: 'yellow' },
            src: 'vertical',
            lst: [...MENU_CHOICES],
            net: gridBit?.grdBit?.dat,
        })

        const selection: string = choiceBit?.chcBit?.src || ''
        cpy.activeSelection = selection
        cpy.lastActionTimestamp = bal?.dat?.timestampMs || 1700000000000

        await handleMenuSelection(selection, ste)

        if (selection === '[ROOT MENU]') {
            if (bal?.slv) {
                bal.slv({
                    mnuBit: {
                        idx: 'root-menu-exit',
                        val: 1,
                        src: '[ROOT MENU]',
                    },
                })
            }
            return cpy
        }
    } catch (err: any) {
        await emitConsole(ste, `:: [FAIL] Menu flight deck encounter: ${err.message}`)
    }

    // Non-blocking timer tick loop (prevent stack recursion)
    if (ste && typeof ste.hunt === 'function') {
        setTimeout(() => {
            ste.hunt(ActMnu.UPDATE_MENU, {}).catch(() => {})
        }, 333)
    }

    if (bal?.slv) {
        bal.slv({
            mnuBit: {
                idx: 'update-menu-pass',
                val: 1,
                src: cpy.activeSelection,
            },
        })
    }

    return cpy
}

/**

import clone from 'clone-deep'
import * as Act from './menu.action.js'
import { MenuModel } from './menu.model.js'
import * as Buzz from './menu.buzzer.js'
import type State from '../99.core/state.js'

export function reducer(
    model: MenuModel = new MenuModel(),
    act: Act.Actions,
    state?: State,
): MenuModel | Promise<MenuModel> {
    switch (act.type) {
        case Act.INIT_MENU:
            return Buzz.initMenu(clone(model), act.bale, state)

        case Act.UPDATE_MENU:
            return Buzz.updateMenu(clone(model), act.bale, state)

        case Act.SELECT_MENU_OPTION:
            return Buzz.selectMenuOptionAction(clone(model), act.bale as any, state)

        default:
            return model
    }
}
