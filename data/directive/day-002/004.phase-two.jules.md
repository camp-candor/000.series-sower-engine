import { BehaviorSubject, Subject } from 'rx-lite'
import type { Action } from './interface/action.interface.js'
import UnitModel, * as Effect from '../BEE.js'

export default class State extends BehaviorSubject<any> {
  public hunt: any
  public value: any
  public pivot: any
  public bus: any

  private dispatcher = new Subject<Action>()

  constructor(init: UnitModel = new UnitModel()) {
    super(init)

    this.dispatcher
      .scan((state: any, action: Action) => this.reducedApp(state, action), init)
      .subscribe((state: any) => {
        super.onNext(state)
      })
  }

  reducedApp(nextState: any, action: Action) {
    for (const key in Effect.reducer) {
      if (typeof Effect.reducer[key] === 'function') {
        nextState[key] = Effect.reducer[key](nextState[key], action, this)
      }
    }
    return nextState
  }

  dispatch(value: Action) {
    return this.dispatcher.onNext(value)
  }

  pat(value: Action) {
    this.dispatch(value)
  }

  next(value: any) {
    this.dispatcher.onNext(value)
  }
}

export const updateMenu = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  const library = (globalThis as any).LIBRARY

  // Headless or Non-TUI Fallback
  if (!library || typeof library.hunt !== 'function') {
    if (bal && bal.slv != null) {
      bal.slv({ mnuBit: { idx: 'update-menu-headless', src: 'HEADLESS_BYPASS' } })
    }
    return cpy
  }

  const lst = [
    'INSPECT FACTS LEDGER',
    'RUN SOMATIC PANOPTICON LINTER',
    'AUDIT SIMULATION KERNEL',
    'ROOT MENU',
  ]

  try {
    const bit = await library.hunt('[Grid action] Update Grid', {
      x: 0,
      y: 4,
      xSpan: 4,
      ySpan: 8,
    })

    const choiceBit = await library.hunt('[Open action] Open Choice', {
      dat: { clr0: 'black', clr1: 'yellow' },
      src: 'vertical',
      lst,
      net: bit.grdBit?.dat,
    })

    const selection = choiceBit.chcBit?.src

    switch (selection) {
      case 'INSPECT FACTS LEDGER':
        await ste.hunt(ActEng.UPDATE_ENGINE, {
          src: 'Inspecting Canon Facts Ledger D1 State',
        })
        break

      case 'RUN SOMATIC PANOPTICON LINTER':
        await ste.hunt(ActEng.UPDATE_ENGINE, {
          src: 'Executing O(1) Somatic SAT Panopticon',
        })
        break

      case 'AUDIT SIMULATION KERNEL':
        await ste.hunt(ActEng.AUDIT_ENGINE, {
          src: 'Verifying dependency purity and monotonic triggers',
        })
        break

      case 'ROOT MENU':
      default:
        if (bal && bal.slv != null) {
          bal.slv({ mnuBit: { idx: 'root-menu' } })
          return cpy
        }
        break
    }
  } catch (err: any) {
    await ste.hunt(ActEng.UPDATE_ENGINE, {
      src: `:: [FAIL] Menu error: ${err.message}`,
    })
  }

  if (bal && bal.slv != null) {
    bal.slv({ mnuBit: { idx: 'update-menu-complete' } })
  }
  return cpy
