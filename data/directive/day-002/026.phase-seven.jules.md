export type UpcasterFn<TOld = any, TNew = any> = (oldPayload: TOld) => TNew

export class EventUpcasterRegistry {
    private readonly upcasters = new Map<string, UpcasterFn>()

    register<TOld = any, TNew = any>(
        type: CanonicalSimulationEventType | string,
        fromVersion: number,
        fn: UpcasterFn<TOld, TNew>,
    ): void {
        this.upcasters.set(`${type}:v${fromVersion}->v${fromVersion + 1}`, fn)
    }

    has(type: string, fromVersion: number): boolean {
        return this.upcasters.has(`${type}:v${fromVersion}->v${fromVersion + 1}`)

        let currentVersion = envelope.schemaVersion
        let currentPayload = envelope.payload
        const targetVersion =
            CURRENT_EVENT_VERSIONS[envelope.eventType as CanonicalSimulationEventType] ||
            envelope.schemaVersion

        while (currentVersion < targetVersion) {
            const key = `${envelope.eventType}:v${currentVersion}->v${currentVersion + 1}`
            const upcaster = this.upcasters.get(key)
            if (!upcaster) {
                throw new Error(
                    `[MISSING_UPCASTER] Cannot migrate event '${envelope.eventType}' from v${currentVersion} to v${targetVersion}`,
                )
            }
            currentPayload = upcaster(currentPayload)
            currentVersion++
        }

        return {
            ...envelope,
            schemaVersion: currentVersion,
            payload: currentPayload,
        }
    }
}

            rawRecords = [...queryRes.databaseBit.dat.records]
        }
    }

    rawRecords.sort((a, b) => a.sequence_id - b.sequence_id)

    const fromSeq = bal.dat?.streamParams?.fromSequenceId || 0
    const limit = bal.dat?.streamParams?.limit || 1000

    const filtered = rawRecords
        .filter((r) => r.sequence_id >= fromSeq)
        .slice(0, limit)

    // Reconstruct and upcast in volatile memory
    const upgradedEnvelopes: EventEnvelope[] = []

    for (const record of filtered) {
        let payload: any
        try {
            payload = JSON.parse(record.payload_json)
        } catch {
            payload = record.payload_json
        }

        const rawEnvelope: EventEnvelope = {
            eventId: record.event_id,
            streamId: record.stream_id,
            sequenceId: record.sequence_id,
            tick: record.tick,
            eventType: record.type,

            seed: record.seed,
            payload,
            prevEventHash: record.prev_hash,
            stateHash: record.prev_hash,
            eventHash: record.state_hash,
            timestampMs: record.timestamp_ms,
        }

        const upcasted = cpy.registry.upcast(rawEnvelope)
        upgradedEnvelopes.push(upcasted)
    }

    await emitConsole(
        ste,
        `>> [EVENT:STREAM] Streamed and upcasted ${upgradedEnvelopes.length} events from sequence #${fromSeq}`,

        bal.slv({

                idx: 'stream-events-pass',
                val: upgradedEnvelopes.length,
                dat: { envelopes: upgradedEnvelopes, totalCount: upgradedEnvelopes.length },
                lst: upgradedEnvelopes,
            },
        })
    }

    return cpy
