1. **`GrievanceStatusRecord` Contract:** Model `grievance_id`, `instigator_id`, `victim_id`, `escalation_depth`, `cooldown_cycles`, `intensity_bps` ($0\text{--}10{,}000$), `status` (`'DORMANT' | 'ACTIVE' | 'SIMMERING' | 'RESOLVED'`), and `updated_at_ms`.


2. **`escalateGrievance`:** Increments `escalation_depth`, updates intensity by delta, auto-transitions status to `'SIMMERING'` when $\ge 8000\text{ bps}$, and enforces integer bounds.


3. **`queryEligibleGrievances`:** Filters out grievances in cooldown (`cooldown_cycles > 0`) or marked `'RESOLVED'`, matching specific actor dyads.


4. **`ParanoiaClockRecord` Contract:** Model `clock_id`, `entity_id`, `clock_name`, `value_bps`, `threshold_bps`, `decay_rate_bps`, `is_triggered`, and `updated_at_ms`.


5. **`tickClockParanoia` & `decayClockParanoia`:** Manages tension increments, evaluates boil-over tripwires (`is_triggered = 1`), and applies thermodynamic passive decay with zero-clamping.


6. **`sweepBoilOverParanoia`:** Sweeps all currently breached clocks across entities to feed emergency scenes.


7. **Colocated Vitest Batteries:** Authors and passes `grievance.test.ts` and `paranoia.test.ts`.

import type { GrievanceModel } from '../grievance.model';
import type GrievanceBit from '../fce/grievance.bit';
import type State from '../../99.core/state';
import type {
  GrievanceStatusRecord,
  UpsertGrievancePayload,
  EscalateGrievancePayload,
  QueryEligiblePayload,

export const assertBasisPoints = (val: number, label: string): number => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be an integer, got: ${val}`);
  }
  if (val < 0 || val > 10000) {
    throw new Error(`[OUT_OF_BOUNDS] ${label} must be between 0 and 10000 bps, got: ${val}`);
  }
  return val;
};

export const initGrievance = (cpy: GrievanceModel, bal: GrievanceBit, _ste: State): GrievanceModel => {
  cpy.records = {};
  if (bal?.slv) {
    bal.slv({ grvBit: { idx: 'init-grievance', val: 0 } });
  }
  return cpy;
};

export const updateGrievance = (cpy: GrievanceModel, bal: GrievanceBit, _ste: State): GrievanceModel => {
  if (bal?.slv) {
    bal.slv({ grvBit: { idx: 'update-grievance' } });
  }
  return cpy;
};

export const upsertGrievance = async (
  cpy: GrievanceModel,
  bal: GrievanceBit<UpsertGrievancePayload>,
  ste: State
): Promise<GrievanceModel> => {
  try {
    const p = bal?.dat;
    if (!p?.grievance_id || !p?.instigator_id || !p?.victim_id) {
      throw new Error('[MALFORMED_INPUT] grievance_id, instigator_id, and victim_id are required');
    }

    if (p.instigator_id.trim() === p.victim_id.trim()) {
      throw new Error(
        `[SELF_GRIEVANCE_BREACH] Instigator ${p.instigator_id} cannot target self in grievance ${p.grievance_id}`
      );
    }

    const intensity =
      p.intensity_bps !== undefined ? assertBasisPoints(p.intensity_bps, 'intensity_bps') : 2000;

    const record: GrievanceStatusRecord = {
      grievance_id: p.grievance_id.trim(),
      instigator_id: p.instigator_id.trim(),
      victim_id: p.victim_id.trim(),
      escalation_depth: Math.max(0, p.escalation_depth ?? 0),
      cooldown_cycles: Math.max(0, p.cooldown_cycles ?? 0),
      intensity_bps: intensity,
      status: p.status || 'DORMANT',
      updated_at_ms: p.updated_at_ms ?? Date.now(),
    };

    cpy.records[record.grievance_id] = record;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [GRIEVANCE:UPSERT] ${record.grievance_id} [${record.status}] Depth: ${record.escalation_depth} | CD: ${record.cooldown_cycles}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ grvBit: { idx: 'upsert-grievance', dat: record, val: record.escalation_depth } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [GRIEVANCE:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ grvBit: { idx: 'upsert-grievance-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const escalateGrievance = async (
  cpy: GrievanceModel,
  bal: GrievanceBit<EscalateGrievancePayload>,
  ste: State
): Promise<GrievanceModel> => {
  try {
    const p = bal?.dat;
    if (!p?.grievance_id || !cpy.records[p.grievance_id]) {
      throw new Error(`[GRIEVANCE_NOT_FOUND] Grievance ${p?.grievance_id} does not exist`);
    }

    const current = cpy.records[p.grievance_id];
    const deltaIntensity = p.intensity_delta_bps ?? 1500;
    if (!Number.isInteger(deltaIntensity)) {
      throw new Error(`[FLOAT_PROHIBITED] intensity_delta_bps must be an integer, got: ${deltaIntensity}`);
    }

    const newIntensity = Math.min(10000, Math.max(0, current.intensity_bps + deltaIntensity));
    const nextDepth = current.escalation_depth + 1;
    const nextStatus = p.next_status || (newIntensity >= 8000 ? 'SIMMERING' : 'ACTIVE');
    const cooldown = Math.max(0, (p.add_cooldown_cycles ?? 0) + current.cooldown_cycles);

    const updated: GrievanceStatusRecord = {
      ...current,
      escalation_depth: nextDepth,
      intensity_bps: newIntensity,
      status: nextStatus,
      cooldown_cycles: cooldown,
      updated_at_ms: p.timestamp_ms ?? Date.now(),
    };

    cpy.records[updated.grievance_id] = updated;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [GRIEVANCE:ESCALATE] ${updated.grievance_id} -> Depth ${nextDepth} (${updated.intensity_bps} bps) [${updated.status}]`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ grvBit: { idx: 'escalate-grievance', dat: updated, val: nextDepth } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [GRIEVANCE:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ grvBit: { idx: 'escalate-grievance-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const tickCooldownGrievance = async (
  cpy: GrievanceModel,
  bal: GrievanceBit,
  _ste: State
): Promise<GrievanceModel> => {
  let decrementedCount = 0;
  for (const id in cpy.records) {
    const grv = cpy.records[id];
    if (grv.cooldown_cycles > 0) {
      grv.cooldown_cycles -= 1;
      decrementedCount++;
    }
  }

  if (bal?.slv) {
    bal.slv({ grvBit: { idx: 'tick-cooldown-grievance', val: decrementedCount } });
  }
  return cpy;
};

export const resolveGrievance = async (
  cpy: GrievanceModel,
  bal: GrievanceBit<{ grievance_id?: string }>,
  _ste: State
): Promise<GrievanceModel> => {
  const grvId = bal?.src || bal?.dat?.grievance_id;
  if (!grvId || !cpy.records[grvId]) {
    if (bal?.slv) {
      bal.slv({ grvBit: { idx: 'resolve-grievance-error', src: 'Grievance not found', val: -1 } });
    }
    return cpy;
  }

  const grv = cpy.records[grvId];
  grv.status = 'RESOLVED';
  grv.intensity_bps = 0;

  if (bal?.slv) {
    bal.slv({ grvBit: { idx: 'resolve-grievance', dat: grv, val: 0 } });
  }
  return cpy;
};

export const queryEligibleGrievances = (
  cpy: GrievanceModel,
  bal: GrievanceBit<QueryEligiblePayload>,
  _ste: State
): GrievanceModel => {
  const p = bal?.dat;
  const eligible = Object.values(cpy.records).filter((g) => {
    if (g.cooldown_cycles !== 0 || g.status === 'RESOLVED') return false;
    if (p?.actorA && p?.actorB) {
      const matchPair =
        (g.instigator_id === p.actorA && g.victim_id === p.actorB) ||
        (g.instigator_id === p.actorB && g.victim_id === p.actorA);
      return matchPair;
    }
    return true;
  });

  if (bal?.slv) {
    bal.slv({ grvBit: { idx: 'query-eligible-grievances', lst: eligible, val: eligible.length } });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './grievance.action';
import { GrievanceModel } from './grievance.model';
import * as Buzz from './grievance.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: GrievanceModel = new GrievanceModel(),
  act: Act.Actions,
  state?: State
): GrievanceModel {
  switch (act.type) {
    case Act.INIT_GRIEVANCE:
      return Buzz.initGrievance(clone(model, true), act.bale, state!);

    case Act.UPDATE_GRIEVANCE:
      return Buzz.updateGrievance(clone(model, true), act.bale, state!);

    case Act.UPSERT_GRIEVANCE:
      Buzz.upsertGrievance(clone(model, true), act.bale, state!);
      return model;

    case Act.ESCALATE_GRIEVANCE:
      Buzz.escalateGrievance(clone(model, true), act.bale, state!);
      return model;

    case Act.TICK_COOLDOWN_GRIEVANCE:
      Buzz.tickCooldownGrievance(clone(model, true), act.bale, state!);
      return model;

    case Act.RESOLVE_GRIEVANCE:
      Buzz.resolveGrievance(clone(model, true), act.bale, state!);
      return model;

    case Act.QUERY_ELIGIBLE_GRIEVANCES:
      return Buzz.queryEligibleGrievances(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}

export const upsertClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit<UpsertClockPayload>,
  ste: State
): Promise<ParanoiaModel> => {
  try {
    const p = bal?.dat;
    if (!p?.clock_id || !p?.entity_id || !p?.clock_name) {
      throw new Error('[MALFORMED_INPUT] clock_id, entity_id, and clock_name are required');
    }

    const value = p.value_bps !== undefined ? assertBasisPoints(p.value_bps, 'value_bps') : 0;
    const threshold = p.threshold_bps !== undefined ? assertBasisPoints(p.threshold_bps, 'threshold_bps') : 8000;
    const decay = p.decay_rate_bps !== undefined ? assertBasisPoints(p.decay_rate_bps, 'decay_rate_bps') : 500;
    const triggered = value >= threshold ? 1 : 0;

    const record: ParanoiaClockRecord = {
      clock_id: p.clock_id.trim(),
      entity_id: p.entity_id.trim(),
      clock_name: p.clock_name.trim(),
      value_bps: value,
      threshold_bps: threshold,
      decay_rate_bps: decay,
      is_triggered: triggered,
      updated_at_ms: p.timestamp_ms ?? Date.now(),
    };

    cpy.clocks[record.clock_id] = record;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [PARANOIA:UPSERT] ${record.clock_id} (${record.clock_name}) | ${value}/${threshold} bps${triggered ? ' [!] BOIL-OVER' : ''}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'upsert-clock-paranoia', dat: record, val: record.is_triggered } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [PARANOIA:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'upsert-clock-paranoia-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const tickClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit<TickClockPayload>,
  ste: State
): Promise<ParanoiaModel> => {
  try {
    const p = bal?.dat;
    if (!p?.clock_id || !cpy.clocks[p.clock_id]) {
      throw new Error(`[CLOCK_NOT_FOUND] Paranoia clock ${p?.clock_id} does not exist`);
    }

    const current = cpy.clocks[p.clock_id];
    const delta = p.delta_bps ?? 1000;
    if (!Number.isInteger(delta)) {
      throw new Error(`[FLOAT_PROHIBITED] delta_bps must be an integer, got: ${delta}`);
    }

    const nextVal = Math.min(10000, Math.max(0, current.value_bps + delta));
    const triggered = nextVal >= current.threshold_bps ? 1 : 0;

    const updated: ParanoiaClockRecord = {
      ...current,
      value_bps: nextVal,
      is_triggered: triggered,
      updated_at_ms: p.timestamp_ms ?? Date.now(),
    };

    cpy.clocks[updated.clock_id] = updated;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [PARANOIA:TICK] ${updated.clock_id}: ${nextVal}/${current.threshold_bps} bps${triggered ? ' [!] BOIL-OVER TRIPPED' : ''}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'tick-clock-paranoia', dat: updated, val: updated.value_bps } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [PARANOIA:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'tick-clock-paranoia-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const decayClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State
): Promise<ParanoiaModel> => {
  let decayedCount = 0;
  for (const id in cpy.clocks) {
    const clk = cpy.clocks[id];
    if (clk.value_bps > 0) {
      const nextVal = Math.max(0, clk.value_bps - clk.decay_rate_bps);
      clk.value_bps = nextVal;
      clk.is_triggered = nextVal >= clk.threshold_bps ? 1 : 0;
      clk.updated_at_ms = Date.now();
      decayedCount++;
    }
  }

  if (ste?.hunt) {
    await ste.hunt('[Console action] Update Console', {
      idx: 'cns00',
      src: `>> [PARANOIA:DECAY] Decayed ${decayedCount} active pressure meter(s)`,
    }).catch(() => {});
  }

  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'decay-clock-paranoia', val: decayedCount } });
  }
  return cpy;
};

export const sweepBoilOverParanoia = (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  _ste: State
): ParanoiaModel => {
  const triggeredClocks = Object.values(cpy.clocks).filter((c) => c.is_triggered === 1);

  if (bal?.slv) {
    bal.slv({
      parBit: {
        idx: 'sweep-boil-over-paranoia',
        lst: triggeredClocks,
        val: triggeredClocks.length,
      },
    });
  }
  return cpy;
};

export const resetClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit<{ clock_id?: string }>,
  _ste: State
): Promise<ParanoiaModel> => {
  const clockId = bal?.src || bal?.dat?.clock_id;
  if (!clockId || !cpy.clocks[clockId]) {
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'reset-clock-paranoia-error', src: 'Clock not found', val: -1 } });
    }
    return cpy;
  }

  const clk = cpy.clocks[clockId];
  clk.value_bps = 0;
  clk.is_triggered = 0;
  clk.updated_at_ms = Date.now();

  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'reset-clock-paranoia', dat: clk, val: 0 } });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './paranoia.action';
import { ParanoiaModel } from './paranoia.model';
import * as Buzz from './paranoia.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: ParanoiaModel = new ParanoiaModel(),
  act: Act.Actions,
  state?: State
): ParanoiaModel {
  switch (act.type) {
    case Act.INIT_PARANOIA:
      return Buzz.initParanoia(clone(model, true), act.bale, state!);

    case Act.UPDATE_PARANOIA:
      return Buzz.updateParanoia(clone(model, true), act.bale, state!);

    case Act.UPSERT_CLOCK_PARANOIA:
      Buzz.upsertClockParanoia(clone(model, true), act.bale, state!);
      return model;

    case Act.TICK_CLOCK_PARANOIA:
      Buzz.tickClockParanoia(clone(model, true), act.bale, state!);
      return model;

    case Act.DECAY_CLOCK_PARANOIA:
      Buzz.decayClockParanoia(clone(model, true), act.bale, state!);
      return model;

    case Act.SWEEP_BOIL_OVER_PARANOIA:
      return Buzz.sweepBoilOverParanoia(clone(model, true), act.bale, state!);

    case Act.RESET_CLOCK_PARANOIA:
      Buzz.resetClockParanoia(clone(model, true), act.bale, state!);
      return model;

    default:
      return model;
  }
}
