 * Deterministically canonicalizes an object using RFC 8785 UTF-16 code-point order.
 */
export function canonicalizeStateJson(obj: unknown): string {
  if (obj === null || typeof obj !== 'object') {
    return JSON.stringify(obj);
  }
  if (Array.isArray(obj)) {
    return `[${obj.map((item) => canonicalizeStateJson(item)).join(',')}]`;
  }
  const sortedKeys = Object.keys(obj as Record<string, unknown>).sort();
  const pairs = sortedKeys
    .map((key) => {
      const val = (obj as Record<string, unknown>)[key];
      if (val === undefined || typeof val === 'symbol') {
        return '';
      }
      return `${JSON.stringify(key)}:${canonicalizeStateJson(val)}`;
    })
    .filter(Boolean);
  return `{${pairs.join(',')}}`;
}

/**
 * Computes a 64-character lowercase hexadecimal SHA-256 state hash.
 */
export function computeWorkingStateHash(maps: {
  entities: Record<string, CompositeEntityLiveRecord>;
  strings: Record<string, StringsLedgerRecord>;
  grievances: Record<string, GrievanceStatusRecord>;
  clocks: Record<string, ParanoiaClockRecord>;
  beliefs: Record<string, ActorBeliefRecord>;
  unratifiedDetails: Record<string, UnratifiedDetailRecord>;
}): string {
  const canonical = canonicalizeStateJson({
    beliefs: maps.beliefs,
    clocks: maps.clocks,
    details: maps.unratifiedDetails,
    entities: maps.entities,
    grievances: maps.grievances,
    strings: maps.strings,
  });
  return createHash('sha256').update(canonical, 'utf8').digest('hex');
}

   * Stages an event into volatile RAM. Evaluates the pure state transition immediately.

  public stageEvent<T>(

      eventId: string;
      streamId: string;
      tick: number;
      actorId: string | null;
      targetId: string | null;

      overlayHash?: string;
      queryHash?: string;
      receiptHash?: string;

    const unsignedEnvelope = {
      sequenceNumber: nextSeq,

      streamId: meta.streamId,
      tick: meta.tick,

      sealedEnvelope as EventEnvelope,
      meta.seed
    );

    this.lastHash = envelopeHash;
    this.pendingEvents.push(sealedEnvelope as EventEnvelope);
    return sealedEnvelope;

      if (ent.pain_bps < 0 || ent.pain_bps > 10000) {

export function clampBps(val: number): number {
  if (val <= 0) return 0;

/**
 * Pure state reducer function: S_{n+1} = R(S_n, E_n, sigma_n)
 * STRICT INVARIANT: Pure function. Zero I/O, zero async, zero Date.now(), zero Math.random().
 */
export function reduceLedgerState(
  prevState: LedgerWorkingState,
  event: EventEnvelope,
  _seed: string
): LedgerWorkingState {
  // 1. Assert Monotonic Sequence Continuity
  if (event.sequenceNumber !== prevState.lastSequenceNumber + 1) {
    throw new Error(
      `[OCC_SEQUENCE_CONFLICT]: Stale sequence token. Expected #${prevState.lastSequenceNumber + 1}, received #${event.sequenceNumber}`
    );
  }

  // 2. Clone maps cleanly
  const nextEntities: Record<string, CompositeEntityLiveRecord> = clone(prevState.entities, true);
  const nextStrings: Record<string, StringsLedgerRecord> = clone(prevState.strings, true);
  const nextGrievances: Record<string, GrievanceStatusRecord> = clone(prevState.grievances, true);
  const nextClocks: Record<string, ParanoiaClockRecord> = clone(prevState.clocks, true);
  const nextBeliefs: Record<string, ActorBeliefRecord> = clone(prevState.beliefs, true);
  const nextDetails: Record<string, UnratifiedDetailRecord> = clone(prevState.unratifiedDetails, true);

  const payload: any = event.payload || {};

  // 3. Sub-reduction by discriminated event type
  switch (event.type) {
    case 'SOMATIC_TRAUMA_APPLIED': {
      const ent = nextEntities[payload.entity_id];
      if (ent) {
        const addedWounds: string[] = payload.added_wounds || [];
        const addedVerbs: string[] = payload.banned_verbs || [];
        const combinedWounds = Array.from(new Set([...ent.open_wounds, ...addedWounds])).sort();
        const combinedVerbs = Array.from(new Set([...ent.banned_kinetic_verbs, ...addedVerbs])).sort();

        nextEntities[payload.entity_id] = {
          ...ent,
          arms_available: payload.arms_available ?? ent.arms_available,
          legs_available: payload.legs_available ?? ent.legs_available,
          locomotion_tier: payload.locomotion_tier ?? ent.locomotion_tier,
          concussion_level: payload.concussion_level ?? ent.concussion_level,
          fatigue_bps: clampBps(ent.fatigue_bps + (payload.fatigue_bps_delta ?? 0)),
          pain_bps: clampBps(ent.pain_bps + (payload.pain_bps_delta ?? 0)),
          refusal_strain_bps: clampBps(ent.refusal_strain_bps + (payload.refusal_strain_bps_delta ?? 0)),
          open_wounds: combinedWounds,
          banned_kinetic_verbs: combinedVerbs,
          updated_at_ms: event.timestamp_ms,
        };
      }
      break;
    }

    case 'INVENTORY_TRANSFERRED': {
      const ent = nextEntities[payload.entity_id];
      if (ent) {
        const nextHeld = payload.held_items ? [...payload.held_items] : ent.held_items;
        if (nextHeld.length > ent.arms_available) {
          throw new Error(
            `[HAND_CAPACITY_BREACH]: Entity ${payload.entity_id} attempted to hold ${nextHeld.length} items with only ${ent.arms_available} arm(s)`
          );
        }
        nextEntities[payload.entity_id] = {
          ...ent,
          held_items: nextHeld,
          worn_gear: payload.worn_gear ? { ...payload.worn_gear } : ent.worn_gear,
          slung_items: payload.slung_items ? [...payload.slung_items] : ent.slung_items,
          cached_items: payload.cached_items ? [...payload.cached_items] : ent.cached_items,
          updated_at_ms: event.timestamp_ms,
        };
      }
      break;
    }

    case 'SPATIAL_DISPLACEMENT_COMMITTED': {
      const ent = nextEntities[payload.entity_id];
      if (ent) {
        nextEntities[payload.entity_id] = {
          ...ent,
          location_id: payload.to_location_id ?? ent.location_id,
          q: payload.to_q ?? ent.q,
          r: payload.to_r ?? ent.r,
          updated_at_ms: event.timestamp_ms,
        };
      }
      break;
    }

    case 'SOCIAL_STRING_EXCHANGED': {
      if (payload.holder_id === payload.debtor_id) {
        throw new Error(
          `[ANTI_REFLEXIVITY_BREACH]: Social string holder and debtor cannot be identical (${payload.holder_id})`
        );
      }
      const pairKey = `${payload.holder_id}:${payload.debtor_id}`;
      const existing = nextStrings[pairKey] || {
        relation_id: payload.relation_id || `rel_${pairKey}`,
        holder_id: payload.holder_id,
        debtor_id: payload.debtor_id,
        weight_bps: 0,
        strings_count: 0,
        updated_at_ms: event.timestamp_ms,
      };

      const countDelta = payload.strings_count_delta ?? 0;
      const nextCount = existing.strings_count + countDelta;
      if (nextCount < 0) {
        throw new Error(
          `[SOLVENCY_BREACH]: Insufficient strings balance (${existing.strings_count}) for debit of ${Math.abs(countDelta)}`
        );
      }

      nextStrings[pairKey] = {
        ...existing,
        weight_bps: clampBps(existing.weight_bps + (payload.weight_bps_delta ?? 0)),
        strings_count: nextCount,
        updated_at_ms: event.timestamp_ms,
      };
      break;
    }

    case 'GRIEVANCE_LIFECYCLE_MUTATED': {
      if (payload.instigator_id && payload.victim_id && payload.instigator_id === payload.victim_id) {
        throw new Error(
          `[ANTI_REFLEXIVITY_BREACH]: Grievance instigator and victim cannot be identical (${payload.instigator_id})`
        );
      }
      const existing = nextGrievances[payload.grievance_id] || {
        grievance_id: payload.grievance_id,
        instigator_id: payload.instigator_id,
        victim_id: payload.victim_id,
        escalation_depth: 0,
        cooldown_cycles: 0,
        intensity_bps: 0,
        status: 'DORMANT',
        updated_at_ms: event.timestamp_ms,
      };

      nextGrievances[payload.grievance_id] = {
        ...existing,
        escalation_depth: Math.max(0, existing.escalation_depth + (payload.escalation_depth_delta ?? 0)),
        cooldown_cycles: Math.max(0, payload.cooldown_cycles ?? existing.cooldown_cycles),
        intensity_bps: clampBps(existing.intensity_bps + (payload.intensity_bps_delta ?? 0)),
        status: payload.status ?? existing.status,
        updated_at_ms: event.timestamp_ms,
      };
      break;
    }

    case 'PARANOIA_CLOCK_TICKED': {
      const clockKey = payload.clock_id;
      const existing = nextClocks[clockKey];
      if (existing) {
        const nextVal = clampBps(existing.value_bps + (payload.value_bps_delta ?? 0));
        nextClocks[clockKey] = {
          ...existing,
          value_bps: nextVal,
          is_triggered: nextVal >= existing.threshold_bps ? 1 : 0,
          updated_at_ms: event.timestamp_ms,
        };
      }
      break;
    }

    case 'BELIEF_RECORDED': {
      const bKey = payload.belief_id;
      nextBeliefs[bKey] = {
        belief_id: payload.belief_id,
        observer_id: payload.observer_id,
        fact_id: payload.fact_id,
        subjective_probability_bps: clampBps(payload.subjective_probability_bps ?? 10000),
        source_type: payload.source_type ?? 'DIRECT_OBSERVATION',
        is_false_belief: payload.is_false_belief ?? 0,
        learned_at_ms: payload.learned_at_ms ?? event.timestamp_ms,
        updated_at_ms: event.timestamp_ms,
      };
      break;
    }

    case 'UNRATIFIED_DETAIL_STAGED': {
      const dKey = payload.detail_id;
      nextDetails[dKey] = {
        detail_id: payload.detail_id,
        scene_id: payload.scene_id,
        entity_id: payload.entity_id,
        salience_bps: clampBps(payload.salience_bps ?? 7000),
        claim_proposal: payload.claim_proposal,
        status: payload.status ?? 'STAGED',
        created_at_ms: payload.created_at_ms ?? event.timestamp_ms,
        reviewed_at_ms: payload.reviewed_at_ms,
      };
      break;
    }

    case 'SIMULATION_TICK_COMMITTED': {
      // Ticks advance through the wrapper envelope
      break;
    }

    default:
      break;
  }

  // 4. Compute next canonical state hash & deep-freeze
  const nextStateHash = computeWorkingStateHash({
    entities: nextEntities,
    strings: nextStrings,
    grievances: nextGrievances,
    clocks: nextClocks,
    beliefs: nextBeliefs,
    unratifiedDetails: nextDetails,
  });

  const nextState: LedgerWorkingState = {
    reducerVersion: prevState.reducerVersion,
    seriesSlug: prevState.seriesSlug,
    tick: Math.max(prevState.tick, event.tick),
    lastSequenceNumber: event.sequenceNumber,
    lastEventHash: event.stateHash,
    entities: nextEntities,
    strings: nextStrings,
    grievances: nextGrievances,
    clocks: nextClocks,
    beliefs: nextBeliefs,
    unratifiedDetails: nextDetails,
    stateHash: nextStateHash,
  };

  return recursiveDeepFreeze(nextState);
}

/**
 * Pure historical fold operation: State_N = fold(reduceLedgerState, State_0, [E_1 ... E_N], seed)

export function foldLedgerEvents(

  events: readonly EventEnvelope[],
  seed: string
): LedgerWorkingState {
  return events.reduce((currentState, ev) => {
    return reduceLedgerState(currentState, ev, seed);
  }, initialState);
}

    ste,
