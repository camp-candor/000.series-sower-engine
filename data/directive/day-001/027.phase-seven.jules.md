      'Execute Levels 1 through 4 sequentially and certify arithmetic kernel at T_0 = $0.00.',
    'ROOT MENU': 'Return to the root engine switchboard.'
  };

  const gridBit = await lib.hunt(UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: 8
  });

    dat: {
      clr0: 0, // Color.BLACK
      clr1: 2, // Color.GREEN
      cb: (choice: string) => {

      const p4 = await runBigIntPrecisionAudit(lib);

      if (p1 && p2 && p3 && p4 && p5) {
        await streamLog(lib, '>> ============================================================');
        await streamLog(lib, '>> [SUCCESS] ALL ARITHMETIC INVARIANTS CERTIFIED AT T_0 = $0.00');
        await streamLog(lib, '>> ============================================================');
      } else {
        await streamLog(lib, '>> [HALT] GAUNTLET TRIPPED: One or more invariants failed.');
      }
      break;
    }

    case 'ROOT MENU':
      if (bale?.slv) bale.slv({ mnuBit: { idx: 'root-menu' } });
      return model;

    default:
      await lib.hunt(CLOSE_TERMINAL, {});
      break;
  }

  setTimeout(async () => {
    if (ste?.hunt) {
      await ste.hunt(OPEN_ARITHMETIC_MENU, {});
    }
  }, 333);

  return model;
}
