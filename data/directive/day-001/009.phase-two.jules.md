import type Event from './fce/event.interface';
import type { UpcasterFunction } from './fce/event.interface';

export const DEFAULT_TARGET_SCHEMA_VERSIONS: Record<string, number> = {
  SOMATIC_TRAUMA_APPLIED: 3,
  INVENTORY_TRANSFERRED: 1,
  LEVERAGE_STRING_MUTATED: 1,
  GRIEVANCE_ESCALATED: 1,
  PARANOIA_CLOCK_TICKED: 1,
  BELIEF_RECORDED: 1,
  UNRATIFIED_DETAIL_STAGED: 1,
  SIMULATION_TICK_COMMITTED: 1,
};

export class EventUpcasterRegistry {
  private upcasters = new Map<string, UpcasterFunction>();

  public register(type: string, fromVersion: number, upcaster: UpcasterFunction): void {
    const key = `${type}:v${fromVersion}->v${fromVersion + 1}`;
    this.upcasters.set(key, upcaster);
  }

  public upcast<T = unknown>(envelope: any, targetVersion: number): any {
    let currentVersion = envelope.schemaVersion || envelope.version || 1;
    let currentPayload = envelope.payload;

    while (currentVersion < targetVersion) {
      const key = `${envelope.eventType || envelope.type}:v${currentVersion}->v${currentVersion + 1}`;
      const fn = this.upcasters.get(key);

      if (!fn) {
        throw new Error(
          `[MISSING_UPCASTER] Cannot migrate ${envelope.eventType || envelope.type} from v${currentVersion} to v${targetVersion}`
        );
      }

      currentPayload = fn(currentPayload);
      currentVersion += 1;
    }

    return {
      ...envelope,
      schemaVersion: currentVersion,
      payload: currentPayload as T,
    };
  }

export const appendEventAction = async (
  cpy: EventModel,
  bal: EventBit,
  ste: State,
): Promise<EventModel> => {
  const dbPort = (bal?.dat as any)?.dbPort || ste?.value?.database?.port;
  const envelope: EventEnvelope = bal?.dat?.envelope;

  if (!envelope || !envelope.eventId) {
    const err = '[APPEND_EVENT_ERROR] Missing valid event envelope payload';
    await emitConsole(ste, `:: ${err}`);
    if (bal?.slv) bal.slv({ err });
    return cpy;
  }

  if (!dbPort) {
    const err = '[DATABASE_ERROR] No local SQLite port available for event persistence';
    await emitConsole(ste, `:: ${err}`);
    if (bal?.slv) bal.slv({ err });
    return cpy;
  }

  try {
    // 1. Verify and seal cryptographic hash if missing
    let finalHash = envelope.eventHash;
    if (!finalHash) {
      finalHash = computeChainedEventHash(envelope.prevEventHash || GENESIS_HASH_64, envelope);
      envelope.eventHash = finalHash;
    }

    // 2. Insert into event_log with ON CONFLICT(event_id) DO NOTHING for idempotency
    const insertSql = `
      INSERT INTO event_log (
        event_id, stream_id, tick, event_type, schema_version,
        actor_id, target_id, seed, payload_json,
        topology_hash, overlay_hash, query_hash, receipt_hash,
        prev_event_hash, state_hash, timestamp_ms
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(event_id) DO NOTHING;
    `;

    const proof = envelope.spatialProof || {
      topologyHash: '0'.repeat(64),
      overlayHash: '0'.repeat(64),
      queryHash: '0'.repeat(64),
      receiptHash: '0'.repeat(64),
    };

    const payloadJson = canonicalizeJson(envelope.payload);

    const stmt = dbPort.prepare(insertSql);
    const runRes = stmt.run(
      envelope.eventId,
      envelope.streamId || 'main',
      envelope.tick || 0,
      envelope.eventType,
      envelope.schemaVersion || 1,
      envelope.actorId ?? null,
      envelope.targetId ?? null,
      envelope.seed || '0',
      payloadJson,
      proof.topologyHash,
      proof.overlayHash,
      proof.queryHash,
      proof.receiptHash,
      envelope.prevEventHash || GENESIS_HASH_64,
      envelope.stateHash,
      envelope.timestampMs || Date.now()
    );

    const duplicate = runRes.changes === 0;

    // Retrieve the persistent sequence_id assigned by SQLite
    const row = dbPort
      .prepare('SELECT sequence_id FROM event_log WHERE event_id = ?')
      .get(envelope.eventId);

    const assignedSeq = row ? row.sequence_id : envelope.sequenceId;

    if (!duplicate) {
      cpy.totalAppended += 1;
      await emitConsole(
        ste,
        `>> [EVENT:APPEND] [OK] Seq #${assignedSeq} | ${envelope.eventType} | ${envelope.eventId} | Hash: ${finalHash.slice(0, 8)}...`
      );
    } else {
      await emitConsole(
        ste,
        `>> [EVENT:IDEMPOTENT] Duplicate event '${envelope.eventId}' skipped at Seq #${assignedSeq}.`
      );
    }

    const appendResult: AppendEventResult = {
      appended: !duplicate,
      duplicate,
      sequenceId: assignedSeq,
      eventId: envelope.eventId,
      eventHash: finalHash,
    };

    if (bal?.slv) {
      bal.slv({
        evtBit: { idx: 'append-event', val: assignedSeq, src: envelope.eventId },
        result: appendResult,
        envelope,
      });
    }
  } catch (err: any) {
    const errorMsg = err instanceof Error ? err.message : String(err);
    await emitConsole(ste, `:: [EVENT:APPEND_FAIL] ${errorMsg}`);
    if (bal?.slv) bal.slv({ err: errorMsg });
  }

  return cpy;
};

export const streamEventsAction = async (
  cpy: EventModel,
  bal: EventBit,

): Promise<EventModel> => {

  const streamId = bal?.src || 'main';
  const fromSeq = typeof bal?.val === 'number' ? bal.val : 0;
  const applyUpcasters = bal?.dat?.applyUpcasters !== false;

    if (bal?.slv) bal.slv({ err: '[DATABASE_ERROR] dbPort required to stream events' });
