import type { ActorModel } from '../actor.model';
import type ActorBit from '../fce/actor.bit';
import type State from '../../99.core/state';
import type {
  ActorBeliefRecord,

  BeliefSourceType,
  RecordBeliefPayload,

  StageDetailPayload,
  ReviewDetailPayload,

export const assertBasisPoints = (val: number, label: string): number => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be an integer, got: ${val}`);
  }
  if (val < 0 || val > 10000) {
    throw new Error(`[OUT_OF_BOUNDS] ${label} must be between 0 and 10000 bps, got: ${val}`);
  }
  return val;
};

export const makeBeliefId = (observerId: string, factId: string): string => {
  return `${observerId.trim()}->${factId.trim()}`;
};

export const initActor = (cpy: ActorModel, bal: ActorBit, _ste: State): ActorModel => {
  cpy.beliefs = {};
  cpy.details = {};
  if (bal?.slv) {
    bal.slv({ actBit: { idx: 'init-actor', val: 0 } });
  }
  return cpy;
};

export const updateActor = (cpy: ActorModel, bal: ActorBit, _ste: State): ActorModel => {
  if (bal?.slv) {
    bal.slv({ actBit: { idx: 'update-actor' } });
  }
  return cpy;
};

export const recordBeliefActor = async (
  cpy: ActorModel,
  bal: ActorBit<RecordBeliefPayload>,
  ste: State
): Promise<ActorModel> => {
  try {
    const p = bal?.dat;
    if (!p?.observer_id || !p?.fact_id) {
      throw new Error('[MALFORMED_INPUT] observer_id and fact_id are required');
    }

    const prob =
      p.subjective_probability_bps !== undefined
        ? assertBasisPoints(p.subjective_probability_bps, 'subjective_probability_bps')
        : 10000;

    const validSources: BeliefSourceType[] = ['INFERENCE', 'TESTIMONY', 'DIRECT_OBSERVATION'];
    const source: BeliefSourceType = p.source_type || 'DIRECT_OBSERVATION';
    if (!validSources.includes(source)) {
      throw new Error(`[INVALID_SOURCE_TYPE] source_type must be one of: ${validSources.join(', ')}`);
    }

    const beliefId = makeBeliefId(p.observer_id, p.fact_id);
    const existing = cpy.beliefs[beliefId];
    const timestamp = p.timestamp_ms ?? (existing?.updated_at_ms || Date.now());

    const isFalse = p.is_false_belief === true || p.is_false_belief === 1 ? 1 : 0;

    const record: ActorBeliefRecord = {
      belief_id: beliefId,
      observer_id: p.observer_id.trim(),
      fact_id: p.fact_id.trim(),
      subjective_probability_bps: prob,
      source_type: source,
      is_false_belief: isFalse,
      learned_at_ms: existing?.learned_at_ms ?? timestamp,
      updated_at_ms: timestamp,
    };

    cpy.beliefs[beliefId] = record;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [EPISTEMIC:BELIEF] ${beliefId} | Prob: ${prob} bps | Source: ${source} | False: ${isFalse}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ actBit: { idx: 'record-belief-actor', dat: record as any, val: prob } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [ACTOR:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ actBit: { idx: 'record-belief-actor-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const queryHorizonActor = (
  cpy: ActorModel,
  bal: ActorBit<{ observer_id?: string }>,
  _ste: State
): ActorModel => {
  const observerId = bal?.src || bal?.dat?.observer_id;
  if (!observerId) {
    if (bal?.slv) {
      bal.slv({ actBit: { idx: 'query-horizon-actor-error', src: 'Missing observer_id', val: -1 } });
    }
    return cpy;
  }

  const horizon = Object.values(cpy.beliefs).filter(
    (b) => b.observer_id === observerId && b.subjective_probability_bps > 0
  );

  if (bal?.slv) {
    bal.slv({
      actBit: {
        idx: 'query-horizon-actor',
        lst: horizon as any[],
        val: horizon.length,
      },
    });
  }
  return cpy;
};

export const scrubEpistemicContextActor = (
  cpy: ActorModel,
  bal: ActorBit<ScrubContextPayload>,
  _ste: State
): ActorModel => {
  const p = bal?.dat;
  if (!p?.observer_id || !p?.world_facts) {
    if (bal?.slv) {
      bal.slv({ actBit: { idx: 'scrub-context-error', src: 'Missing observer_id or world_facts', val: -1 } });
    }
    return cpy;
  }

  const scrubbed: Record<string, string> = {};
  let omittedCount = 0;

  for (const [factId, factText] of Object.entries(p.world_facts)) {
    const beliefId = makeBeliefId(p.observer_id, factId);
    const belief = cpy.beliefs[beliefId];

    // Ignorance by Omission: Keep only facts known with probability > 0
    if (belief && belief.subjective_probability_bps > 0) {
      scrubbed[factId] = factText;
    } else {
      omittedCount++;
    }
  }

  if (bal?.slv) {
    bal.slv({
      actBit: {
        idx: 'scrub-epistemic-context-actor',
        dat: scrubbed as any,
        val: omittedCount,
      },
    });
  }
  return cpy;
};

export const stageDetailActor = async (

      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ actBit: { idx: 'stage-detail-actor', dat: record as any, val: salience } });

  } catch (err: any) {

export const ratifyDetailActor = async (

  if (bal?.slv) {
    bal.slv({ actBit: { idx: 'ratify-detail-actor', dat: detail as any, val: 1 } });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './actor.action';
import { ActorModel } from './actor.model';
import * as Buzz from './actor.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: ActorModel = new ActorModel(),
  act: Act.Actions,
  state?: State
): ActorModel {
  switch (act.type) {
    case Act.INIT_ACTOR:
      return Buzz.initActor(clone(model, true), act.bale, state!);

    case Act.UPDATE_ACTOR:
      return Buzz.updateActor(clone(model, true), act.bale, state!);

    case Act.RECORD_BELIEF_ACTOR:
      Buzz.recordBeliefActor(clone(model, true), act.bale, state!);
      return model;

    case Act.QUERY_HORIZON_ACTOR:
      return Buzz.queryHorizonActor(clone(model, true), act.bale, state!);

    case Act.SCRUB_EPISTEMIC_CONTEXT_ACTOR:
      return Buzz.scrubEpistemicContextActor(clone(model, true), act.bale, state!);

    case Act.STAGE_DETAIL_ACTOR:
      Buzz.stageDetailActor(clone(model, true), act.bale, state!);
      return model;

    case Act.RATIFY_DETAIL_ACTOR:
      Buzz.ratifyDetailActor(clone(model, true), act.bale, state!);
      return model;

    case Act.DISCARD_DETAIL_ACTOR:
      Buzz.discardDetailActor(clone(model, true), act.bale, state!);
      return model;

    case Act.LIST_UNRATIFIED_DETAILS_ACTOR:
      return Buzz.listUnratifiedDetailsActor(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}
