import type LeverageBit from '../fce/leverage.bit';
import type State from '../../99.core/state';

const assertBasisPoints = (val: number, label: string): number => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be an integer, got: ${val}`);
  }
  if (val < 0 || val > 10000) {
    throw new Error(`[OUT_OF_BOUNDS] ${label} must be between 0 and 10000 bps, got: ${val}`);
  }
  return val;
};

const makeRelationId = (holderId: string, debtorId: string): string => {
  return `${holderId.trim()}->${debtorId.trim()}`;
};

export const initLeverage = (cpy: LeverageModel, bal: LeverageBit, ste: State) => {
  cpy.records = {};
  if (bal?.slv) {
    bal.slv({ levBit: { idx: 'init-leverage', val: 0 } });
  }
  return cpy;
};

export const updateLeverage = (cpy: LeverageModel, bal: LeverageBit, ste: State) => {
  if (bal?.slv) {
    bal.slv({ levBit: { idx: 'update-leverage' } });
  }
  return cpy;
};

export const mutateStringLeverage = async (
  cpy: LeverageModel,
  bal: LeverageBit,
  ste: State,
) => {
  try {
    const payload = bal?.dat as {
      holder_id: string;
      debtor_id: string;
      weight_bps?: number;
      strings_count?: number;
      timestamp_ms?: number;
    };

    if (!payload?.holder_id || !payload?.debtor_id) {
      throw new Error('[MALFORMED_INPUT] holder_id and debtor_id are required');
    }

    if (payload.holder_id.trim() === payload.debtor_id.trim()) {
      throw new Error(
        `[ANTI_REFLEXIVITY_BREACH] Actor ${payload.holder_id} cannot hold leverage strings over self`,
      );
    }

    const relId = makeRelationId(payload.holder_id, payload.debtor_id);
    const existing = cpy.records[relId];

    const weight =
      payload.weight_bps !== undefined
        ? assertBasisPoints(payload.weight_bps, 'weight_bps')
        : existing?.weight_bps ?? 1000;

    const countDelta = payload.strings_count !== undefined ? payload.strings_count : 1;
    if (!Number.isInteger(countDelta)) {
      throw new Error(`[FLOAT_PROHIBITED] strings_count must be an integer, got: ${countDelta}`);
    }

    const newCount = Math.max(0, (existing?.strings_count ?? 0) + countDelta);
    const timestamp = payload.timestamp_ms ?? (existing?.updated_at_ms || 0);

    const record: StringsLedgerRecord = {
      relation_id: relId,
      holder_id: payload.holder_id.trim(),
      debtor_id: payload.debtor_id.trim(),
      weight_bps: weight,
      strings_count: newCount,
      updated_at_ms: timestamp,
    };

    cpy.records[relId] = record;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [LEVERAGE:MUTATE] ${relId} | Count: ${newCount} | Weight: ${weight} bps`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ levBit: { idx: 'mutate-string-leverage', dat: record, val: newCount } });
    }
  } catch (err: any) {
    if (ste?.hunt) {

        src: `:: [LEVERAGE:ERR] ${err.message}`,

      bal.slv({ levBit: { idx: 'mutate-string-leverage-error', src: err.message, val: -1 } });
    }
  }

    if (!payload?.holder_id || !payload?.debtor_id) {
      throw new Error('[MALFORMED_INPUT] holder_id and debtor_id are required');
    }

    if (payload.holder_id.trim() === payload.debtor_id.trim()) {
      throw new Error(
        `[ANTI_REFLEXIVITY_BREACH] Actor ${payload.holder_id} cannot hold or spend leverage strings over self`,
      );
    }

    const relId = makeRelationId(payload.holder_id, payload.debtor_id);
    const existing = cpy.records[relId];
    const currentCount = existing?.strings_count ?? 0;
    const spend = payload.spend_count ?? 1;

    if (!Number.isInteger(spend) || spend <= 0) {
      throw new Error(`[INVALID_SPEND_COUNT] spend_count must be a positive integer, got: ${spend}`);
    }

    if (currentCount < spend) {
      throw new Error(
        `[INSUFFICIENT_STRINGS_BREACH] Holder ${payload.holder_id} holds only ${currentCount} strings over ${payload.debtor_id}, but attempted to spend ${spend}`,
      );
    }

    const updatedRecord: StringsLedgerRecord = {
      ...existing,
      strings_count: currentCount - spend,
      updated_at_ms: payload.timestamp_ms ?? existing.updated_at_ms,
    };

    cpy.records[relId] = updatedRecord;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [LEVERAGE:SPEND] ${relId} | Spent: ${spend} | Remaining: ${updatedRecord.strings_count}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({
        levBit: {
          idx: 'spend-string-leverage',
          dat: updatedRecord,
          val: updatedRecord.strings_count,
        },
      });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [LEVERAGE:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ levBit: { idx: 'spend-string-leverage-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const getPairLeverage = (cpy: LeverageModel, bal: LeverageBit, ste: State) => {
  const payload = bal?.dat as { holder_id: string; debtor_id: string };
  if (!payload?.holder_id || !payload?.debtor_id) {
    if (bal?.slv) {
      bal.slv({ levBit: { idx: 'get-pair-leverage-error', src: 'Missing IDs', val: -1 } });

      },
    });
  }
  return cpy;
};

    bal.slv({ grvBit: { idx: 'query-eligible-grievances', lst: eligible, val: eligible.length } });

const assertBasisPoints = (val: number, label: string): number => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] ${label} must be an integer, got: ${val}`);
  }
  if (val < 0 || val > 10000) {
    throw new Error(`[OUT_OF_BOUNDS] ${label} must be between 0 and 10000 bps, got: ${val}`);
  }
  return val;
};

export const initParanoia = (cpy: ParanoiaModel, bal: ParanoiaBit, ste: State) => {
  cpy.clocks = {};
  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'init-paranoia', val: 0 } });
  }
  return cpy;
};

export const updateParanoia = (cpy: ParanoiaModel, bal: ParanoiaBit, ste: State) => {
  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'update-paranoia' } });
  }

export const upsertClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State,
) => {
  try {
    const p = bal?.dat as Partial<ParanoiaClockRecord>;
    if (!p?.clock_id || !p?.entity_id || !p?.clock_name) {
      throw new Error('[MALFORMED_INPUT] clock_id, entity_id, and clock_name are required');
    }

    const valBps = assertBasisPoints(p.value_bps ?? 0, 'value_bps');
    const thresholdBps = assertBasisPoints(p.threshold_bps ?? 8000, 'threshold_bps');
    const decayRateBps = assertBasisPoints(p.decay_rate_bps ?? 500, 'decay_rate_bps');
    const isTriggered = valBps >= thresholdBps ? 1 : 0;

    const record: ParanoiaClockRecord = {
      clock_id: p.clock_id.trim(),
      entity_id: p.entity_id.trim(),
      clock_name: p.clock_name.trim(),
      value_bps: valBps,
      threshold_bps: thresholdBps,
      decay_rate_bps: decayRateBps,
      is_triggered: isTriggered,
      updated_at_ms: p.updated_at_ms ?? 0,
    };

    cpy.clocks[record.clock_id] = record;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [PARANOIA:UPSERT] ${record.clock_id} [${record.clock_name}] | ${record.value_bps}/${record.threshold_bps} bps | Triggered: ${record.is_triggered}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'upsert-clock-paranoia', dat: record, val: record.value_bps } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [PARANOIA:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'upsert-clock-paranoia-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const tickClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State,
) => {
  try {
    const p = bal?.dat as {
      clock_id: string;
      delta_bps: number;
      timestamp_ms?: number;
    };

    if (!p?.clock_id || !cpy.clocks[p.clock_id]) {
      throw new Error(`[CLOCK_NOT_FOUND] Paranoia clock ${p?.clock_id} does not exist`);
    }

    if (!Number.isInteger(p.delta_bps)) {
      throw new Error(`[FLOAT_PROHIBITED] delta_bps must be integer, got: ${p.delta_bps}`);
    }

    const clock = cpy.clocks[p.clock_id];
    const nextVal = Math.min(10000, Math.max(0, clock.value_bps + p.delta_bps));
    const wasTriggered = clock.is_triggered;
    const isTriggered = nextVal >= clock.threshold_bps ? 1 : 0;

    clock.value_bps = nextVal;
    clock.is_triggered = isTriggered;
    clock.updated_at_ms = p.timestamp_ms ?? clock.updated_at_ms;

    if (ste?.hunt) {
      const boilNote = isTriggered && !wasTriggered ? ' [!] BOIL-OVER TRIPPED' : '';
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [PARANOIA:TICK] ${clock.clock_id}: ${nextVal}/${clock.threshold_bps} bps${boilNote}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'tick-clock-paranoia', dat: clock, val: nextVal } });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [PARANOIA:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'tick-clock-paranoia-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const decayClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State,
) => {
  let decayedCount = 0;
  for (const id in cpy.clocks) {
    const clk = cpy.clocks[id];
    if (clk.value_bps > 0) {
      const nextVal = Math.max(0, clk.value_bps - clk.decay_rate_bps);
      clk.value_bps = nextVal;
      clk.is_triggered = nextVal >= clk.threshold_bps ? 1 : 0;
      decayedCount++;
    }
  }

  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'decay-clock-paranoia', val: decayedCount } });
  }
  return cpy;
};

export const sweepBoilOverParanoia = (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State,
) => {
  const breached = Object.values(cpy.clocks).filter((c) => c.is_triggered === 1);
  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'sweep-boil-over-paranoia', lst: breached, val: breached.length } });
  }
  return cpy;
};

export const resetClockParanoia = async (
  cpy: ParanoiaModel,
  bal: ParanoiaBit,
  ste: State,
) => {
  const clockId = bal?.src || (bal?.dat as any)?.clock_id;
  if (!clockId || !cpy.clocks[clockId]) {
    if (bal?.slv) {
      bal.slv({ parBit: { idx: 'reset-clock-paranoia-error', src: 'Clock not found', val: -1 } });
    }
    return cpy;
  }

  const clock = cpy.clocks[clockId];
  clock.value_bps = 0;
  clock.is_triggered = 0;

  if (bal?.slv) {
    bal.slv({ parBit: { idx: 'reset-clock-paranoia', dat: clock, val: 0 } });
  }
  return cpy;
};
