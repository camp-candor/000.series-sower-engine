class InMemorySqliteAdapter implements ILocalDatabasePort {
  private db: Database.Database;

  constructor() {
    this.db = new Database(':memory:');
    this.db.pragma('foreign_keys = ON');
    this.db.pragma('journal_mode = WAL');
  }

  public exec(sql: string): void {
    this.db.exec(sql);
  }

  public query<T = any>(sql: string, params: any[] = []): T[] {
    return this.db.prepare(sql).all(...params) as T[];
  }

  public run(sql: string, params: any[] = []): { changes: number; lastInsertRowid: number | bigint } {
    const info = this.db.prepare(sql).run(...params);
    return { changes: info.changes, lastInsertRowid: info.lastInsertRowid };
  }

  public batch(statements: PreparedBatchStatement[]): void {
    const txn = this.db.transaction((stmts: PreparedBatchStatement[]) => {
      for (const s of stmts) {
        this.db.prepare(s.sql).run(...(s.params as any[]));
      }
    });
    txn(statements);
  }

  public close(): void {
    this.db.close();
  }
}
