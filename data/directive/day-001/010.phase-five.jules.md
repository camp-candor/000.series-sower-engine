export interface SomaticProfile {
  armsAvailable: number;
  locomotionTier: 'UNRESTRICTED' | 'LIMPING' | 'IMMOBILIZED';
  openWounds: string[];
  status?: string;
}

export interface SomaticProjection {
  movementBudgetUnits: number;
  bannedKineticVerbs: Set<string>;
}

export function getCharacterSomaticMask(profile: SomaticProfile): SomaticProjection {
  const banned = new Set<string>();

  if (profile.status === 'departed') {
    banned.add('*');
    return { movementBudgetUnits: 0, bannedKineticVerbs: banned };
  }

  let budget = 1000;
  if (profile.locomotionTier === 'LIMPING') {
    budget = 400;
    banned.add('SPRINT');
    banned.add('VAULT');
    banned.add('RUN');
  } else if (profile.locomotionTier === 'IMMOBILIZED') {
    budget = 0;
    banned.add('SPRINT');
    banned.add('VAULT');
    banned.add('RUN');
    banned.add('WALK');
    banned.add('LIMP');
  }

  if (profile.armsAvailable < 2) {
    banned.add('BIMANUAL_CLIMB');
    banned.add('TWO_HANDED_LEVER');
  }
  if (profile.armsAvailable === 0) {
    banned.add('GRASP');
    banned.add('CARRY_HAND');
  }

  return { movementBudgetUnits: budget, bannedKineticVerbs: banned };
}

export type EpistemicHorizon =
  | 'TOTAL_IGNORANCE'
  | 'RUMOR_ONLY'
  | 'FALSE_BELIEF'
  | 'PARTIAL_FACT'
  | 'WITHHELD'
  | 'UNRESOLVED_BY_CANON';

export interface MysteryDirective {
  topicId: string;
  horizon: EpistemicHorizon;
  injectedDirectives: string[];
  omittedTopicIds: string[];
}

export interface ClueLemmaContract {
  permittedClues: string[];
  triggerLemmas: string[];
}

export function compileEpistemicDirectives(
  actorId: string,
  topicId: string,
  horizon: EpistemicHorizon,
  distortedBelief?: string
): MysteryDirective {
  const omitted: string[] = [];
  const injected: string[] = [];

  switch (horizon) {
    case 'TOTAL_IGNORANCE':
      omitted.push(topicId);
      break;
    case 'RUMOR_ONLY':
      injected.push(`[UNVERIFIED HEARSAY: ${distortedBelief || 'vague chatter'}]`);
      break;
    case 'FALSE_BELIEF':
      injected.push(`[SUBJECTIVE CONVICTION: ${distortedBelief || 'absolute certainty'}]`);
      break;
    case 'WITHHELD':
      injected.push('[RESTRAINED KNOWLEDGE: Do not disclose root truth]');
      break;
    case 'UNRESOLVED_BY_CANON':
      omitted.push(topicId);
      injected.push('[CANONICAL TABOO: Resolution forbidden]');
      break;
    case 'PARTIAL_FACT':
      injected.push('[PERCEPTUAL EVIDENCE: Clues observed only]');
      break;
  }

  return {
    topicId,
    horizon,
    injectedDirectives: injected,
    omittedTopicIds: omitted,
  };
}

export function verifyClueLemmaDisjointness(contract: ClueLemmaContract): boolean {
  const triggerSet = new Set(contract.triggerLemmas.map((l) => l.toLowerCase()));
  for (const clue of contract.permittedClues) {
    if (triggerSet.has(clue.toLowerCase())) {
      return false;
    }
  }
  return true;
}

import { DialogueAst } from '../../types/dialogueAst.js';
import { SomaticProjection } from '../canon/somaticMask.js';

export interface LinterViolation {
  beatId: string;
  speakerId: string;
  code: 'SOMATIC_MOTOR_BREACH' | 'HAND_CAPACITY_BREACH';
  detail: string;
}

export function lintDialogueAstSomaticInvariants(
  ast: DialogueAst,
  actorProjections: Record<string, { projection: SomaticProjection; armsAvailable: number }>
): { valid: boolean; violations: LinterViolation[] } {
  const violations: LinterViolation[] = [];

  for (const beat of ast.beats) {
    const actor = actorProjections[beat.speakerId];
    if (!actor) continue;

    // 1. Hand capacity SAT check
    if (beat.heldItems.length > actor.armsAvailable) {
      violations.push({
        beatId: beat.beatId,
        speakerId: beat.speakerId,
        code: 'HAND_CAPACITY_BREACH',
        detail: `Actor requires ${beat.heldItems.length} hands, but has ${actor.armsAvailable} available.`,
      });
    }

    // 2. O(1) set-intersection check for banned motor actions
    const banned = actor.projection.bannedKineticVerbs;
    if (banned.has('*')) {
      violations.push({
        beatId: beat.beatId,
        speakerId: beat.speakerId,
        code: 'SOMATIC_MOTOR_BREACH',
        detail: `Departed actor cannot execute kinetic actions.`,
      });
      continue;
    }

    for (const verb of beat.kineticVerbs) {
      if (banned.has(verb)) {
        violations.push({
          beatId: beat.beatId,
          speakerId: beat.speakerId,
          code: 'SOMATIC_MOTOR_BREACH',
          detail: `Actor attempted banned kinetic verb: ${verb}`,
        });
      }
    }
  }

  return {
    valid: violations.length === 0,
    violations,
  };
}
