import { createHash } from 'node:crypto';
import type { StateModel } from '../state.model';
import type StateBit from '../fce/state.bit';
import type CoreState from '../../99.core/state';
import {
  REDUCER_CONTRACT_VERSION,
  type DeepReadonly,
  type LedgerWorkingState,
  type StateWorkingMaps,
  type HydrateStatePayload,
} from '../fce/state.interface';
import { deserializeDbRowToEntity } from '../../03.entity.unit/buz/entity.buzz';

export function recursiveDeepFreeze<T>(obj: T): DeepReadonly<T> {
  if (obj === null || typeof obj !== 'object') {
    return obj as DeepReadonly<T>;
  }

  if (Object.isFrozen(obj)) {
    return obj as DeepReadonly<T>;
  }

  Object.freeze(obj);

  const propNames = Object.getOwnPropertyNames(obj);
  for (const name of propNames) {
    const value = (obj as any)[name];
    if (value !== null && typeof value === 'object' && !Object.isFrozen(value)) {
      recursiveDeepFreeze(value);
    }
  }

  return obj as DeepReadonly<T>;
}

export function canonicalizeStateJson(obj: unknown): string {
  if (obj === null || typeof obj !== 'object') {
    if (typeof obj === 'number') {
      if (!Number.isFinite(obj)) {
        throw new TypeError('[CANONICAL_JSON_ERROR] Non-finite numbers are prohibited in RFC 8785');
      }
      return JSON.stringify(obj);
    }
    if (typeof obj === 'string') {
      return JSON.stringify(obj.normalize('NFC'));
    }
    return JSON.stringify(obj);
  }

  if (Array.isArray(obj)) {
    const elements = obj.map((element) =>
      element === undefined ? 'null' : canonicalizeStateJson(element)
    );
    return `[${elements.join(',')}]`;
  }

  const sortedKeys = Object.keys(obj as Record<string, unknown>).sort((a, b) => {
    return a < b ? -1 : a > b ? 1 : 0;
  });

  const parts: string[] = [];
  for (const key of sortedKeys) {
    const value = (obj as Record<string, unknown>)[key];
    if (value !== undefined) {
      const normalizedKey = key.normalize('NFC');
      parts.push(`${JSON.stringify(normalizedKey)}:${canonicalizeStateJson(value)}`);
    }
  }

  return `{${parts.join(',')}}`;
}

export function computeWorkingStateHash(stateMaps: {
  entities: Record<string, unknown>;
  strings: Record<string, unknown>;
  grievances: Record<string, unknown>;
  clocks: Record<string, unknown>;
  beliefs: Record<string, unknown>;
  unratifiedDetails?: Record<string, unknown>;
  details?: Record<string, unknown>;
}): string {
  const normalizedMaps = {
    beliefs: stateMaps.beliefs || {},
    clocks: stateMaps.clocks || {},
    details: stateMaps.details || stateMaps.unratifiedDetails || {},
    entities: stateMaps.entities || {},
    grievances: stateMaps.grievances || {},
    strings: stateMaps.strings || {},
  };

  const canonicalJson = canonicalizeStateJson(normalizedMaps);
  return createHash('sha256').update(canonicalJson, 'utf8').digest('hex');
}

    clocks: JSON.parse(JSON.stringify(state.clocks)),
    beliefs: JSON.parse(JSON.stringify(state.beliefs)),
    unratifiedDetails: JSON.parse(JSON.stringify(state.unratifiedDetails)),
  };
}

export const hydrateState = async (
  cpy: StateModel,
  bal: StateBit<HydrateStatePayload>,
  ste: CoreState
): Promise<StateModel> => {
  try {
    const payload = bal?.dat || {};
    const seriesSlug = payload.seriesSlug || 'sower-main';

    let entityRows: any[] = [];
    let stringRows: any[] = [];
    let grievanceRows: any[] = [];
    let clockRows: any[] = [];
    let beliefRows: any[] = [];
    let detailRows: any[] = [];
    let latestEvent: any = null;

    if (ste?.hunt) {
      const q = async (sql: string) => {
        try {
          const res: any = await ste.hunt('[Database action] Query Database', { src: sql, dat: [] });
          return res?.datBit?.lst || [];
        } catch {
          return [];
        }
      };

      entityRows = await q('SELECT * FROM entities_live ORDER BY entity_id ASC');
      stringRows = await q('SELECT * FROM strings_ledger ORDER BY relation_id ASC');
      grievanceRows = await q('SELECT * FROM grievance_status ORDER BY grievance_id ASC');
      clockRows = await q('SELECT * FROM paranoia_clocks ORDER BY clock_id ASC');
      beliefRows = await q('SELECT * FROM actor_beliefs ORDER BY belief_id ASC');
      detailRows = await q('SELECT * FROM unratified_details ORDER BY detail_id ASC');

      const events = await q('SELECT sequence_id, state_hash, tick FROM event_log ORDER BY sequence_id DESC LIMIT 1');
      latestEvent = events[0] || null;
    }

    const entities: Record<string, any> = {};
    for (const r of entityRows) {
      const parsed = deserializeDbRowToEntity(r);
      entities[parsed.entityId] = parsed;
    }

    const strings: Record<string, any> = {};
    for (const r of stringRows) {
      strings[r.relation_id] = r;
    }

    const grievances: Record<string, any> = {};
    for (const r of grievanceRows) {
      grievances[r.grievance_id] = r;
    }

    const clocks: Record<string, any> = {};
    for (const r of clockRows) {
      clocks[r.clock_id] = r;
    }

    const beliefs: Record<string, any> = {};
    for (const r of beliefRows) {
      beliefs[r.belief_id] = r;
    }

    const unratifiedDetails: Record<string, any> = {};
    for (const r of detailRows) {
      unratifiedDetails[r.detail_id] = r;
    }

    const stateMaps = {
      entities,
      strings,
      grievances,
      clocks,
      beliefs,
      unratifiedDetails,
    };

    const computedHash = computeWorkingStateHash(stateMaps);

    const mutableState: LedgerWorkingState = {
      reducerVersion: REDUCER_CONTRACT_VERSION,
      seriesSlug,
      tick: latestEvent?.tick ?? payload.fallbackTick ?? 0,
      lastSequenceNumber: latestEvent?.sequence_id ?? 0,
      lastEventHash: latestEvent?.state_hash ?? '0'.repeat(64),
      entities,
      strings,
      grievances,
      clocks,
      beliefs,
      unratifiedDetails,
      stateHash: computedHash,
    };

    const frozenState = recursiveDeepFreeze(mutableState);
    cpy.workingState = frozenState as LedgerWorkingState;

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [STATE:HYDRATE] Sealed LedgerWorkingState | Hash: ${computedHash.slice(0, 16)}... | Entities: ${Object.keys(entities).length}`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({
        staBit: {
          idx: 'hydrate-state-ok',
          src: computedHash,
          val: latestEvent?.sequence_id ?? 0,
          dat: frozenState as any,
        },
      });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [STATE:ERR] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ staBit: { idx: 'hydrate-state-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const freezeState = (
  cpy: StateModel,
  bal: StateBit<{ target: any }>,
  _ste: CoreState
): StateModel => {
  const target = bal?.dat?.target;
  if (!target) {
    if (bal?.slv) bal.slv({ staBit: { idx: 'freeze-state-error', src: 'Missing target', val: -1 } });
    return cpy;
  }

  const frozen = recursiveDeepFreeze(target);
  if (bal?.slv) {
    bal.slv({ staBit: { idx: 'freeze-state-ok', dat: frozen as any, val: 1 } });
  }
  return cpy;
};

export const getState = (cpy: StateModel, bal: StateBit, _ste: CoreState): StateModel => {
  if (bal?.slv) {
    bal.slv({
      staBit: {
        idx: cpy.workingState ? 'get-state-ok' : 'get-state-uninitialized',
        src: cpy.workingState?.stateHash ?? '',
        val: cpy.workingState ? cpy.workingState.lastSequenceNumber : -1,
        dat: cpy.workingState as any,
      },
    });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './state.action';
import { StateModel } from './state.model';
import * as Buzz from './state.buzzer';
import type CoreState from '../99.core/state';

export function reducer(
  model: StateModel = new StateModel(),
  act: Act.Actions,
  state?: CoreState
): StateModel {
  switch (act.type) {
    case Act.INIT_STATE:
      return Buzz.initState(clone(model, true), act.bale, state!);

    case Act.UPDATE_STATE:
      return Buzz.updateState(clone(model, true), act.bale, state!);

    case Act.HYDRATE_STATE:
      Buzz.hydrateState(clone(model, true), act.bale, state!);
      return model;

    case Act.FREEZE_STATE:
      return Buzz.freezeState(clone(model, true), act.bale, state!);

    case Act.GET_STATE:
      return Buzz.getState(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}
