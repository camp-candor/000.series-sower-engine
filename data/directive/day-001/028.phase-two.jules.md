    const payload = bal?.dat as {
      observer_id: string;
      world_facts: Record<string, string>;
    };

    if (!payload?.observer_id || !payload?.world_facts) {
      throw new Error('[MALFORMED_INPUT] observer_id and world_facts map required');
    }

    const observerId = payload.observer_id.trim();
    const scrubbedFacts: Record<string, string> = {};
    let omittedCount = 0;

    for (const [factId, factContent] of Object.entries(payload.world_facts)) {
      const beliefId = makeBeliefId(observerId, factId);
      const belief = cpy.beliefs[beliefId];

      // Ignorance by Omission Invariant:
      // If observer has no belief record OR subjective probability is 0 bps, omit from context.
      if (belief && belief.subjective_probability_bps > 0) {
        scrubbedFacts[factId] = factContent;
      } else {
        omittedCount++;
      }
    }

    if (bal?.slv) {
      bal.slv({
        actBit: {
          idx: 'scrub-epistemic-context-actor',
          dat: scrubbedFacts as any,
          val: omittedCount,
        },
      });
    }
  } catch (err: any) {
    if (bal?.slv) {
      bal.slv({
        actBit: {
          idx: 'scrub-epistemic-context-actor-error',
          src: err.message,
          val: -1,
        },
      });
    }
  }

  return cpy;
};

export const ratifyDetailActor = async (

import clone from 'clone-deep';
import * as Act from './actor.action';
import { ActorModel } from './actor.model';
import * as Buzz from './actor.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: ActorModel = new ActorModel(),
  act: Act.Actions,
  state?: State,
) {
  switch (act.type) {
    case Act.INIT_ACTOR:
      return Buzz.initActor(clone(model, true), act.bale, state);

    case Act.UPDATE_ACTOR:
      return Buzz.updateActor(clone(model, true), act.bale, state);

    case Act.RECORD_BELIEF_ACTOR:
      return Buzz.recordBeliefActor(clone(model, true), act.bale, state);

    case Act.QUERY_HORIZON_ACTOR:
      return Buzz.queryHorizonActor(clone(model, true), act.bale, state);

    case Act.SCRUB_EPISTEMIC_CONTEXT_ACTOR:
      return Buzz.scrubEpistemicContextActor(clone(model, true), act.bale, state);

    case Act.STAGE_DETAIL_ACTOR:
      return Buzz.stageDetailActor(clone(model, true), act.bale, state);

    case Act.RATIFY_DETAIL_ACTOR:
      return Buzz.ratifyDetailActor(clone(model, true), act.bale, state);

    case Act.DISCARD_DETAIL_ACTOR:
      return Buzz.discardDetailActor(clone(model, true), act.bale, state);

    case Act.LIST_UNRATIFIED_DETAILS_ACTOR:
      return Buzz.listUnratifiedDetailsActor(clone(model, true), act.bale, state);

    default:
      return model;
  }
}
