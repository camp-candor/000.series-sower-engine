export const validateCapacityLocation = async (
  cpy: LocationModel,
  bal: LocationBit<ValidateCapacityPayload>,
  ste: State

  const payload = bal?.dat;
  if (!payload?.locationId || !payload?.incomingEntityId) {
    if (bal?.slv) bal.slv({ locBit: { idx: 'validate-capacity-err', src: 'MISSING_PAYLOAD', val: -1 } });
    return cpy;
  }

  const location = cpy.locations[payload.locationId] || createDefaultLocationRecord(payload.locationId);

  // If entity is already inside, capacity check succeeds
  if (location.occupants.includes(payload.incomingEntityId)) {
    const receipt: CapacityValidationReceipt = {
      allowed: true,
      locationId: payload.locationId,
      currentOccupancy: location.occupants.length,
      maxOccupancy: location.maxOccupancy,
    };
    if (bal?.slv) bal.slv({ locBit: { idx: 'validate-capacity-ok', dat: receipt } });
    return cpy;
  }

  if (location.occupants.length >= location.maxOccupancy) {
    const errorMsg = `>> [ROOM_CAPACITY_EXCEEDED] Location ${payload.locationId} saturated: ${location.occupants.length}/${location.maxOccupancy} occupants!`;

    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: errorMsg,
      }).catch(() => {});
    }

    const receipt: CapacityValidationReceipt = {
      allowed: false,
      locationId: payload.locationId,
      currentOccupancy: location.occupants.length,
      maxOccupancy: location.maxOccupancy,
      error: errorMsg,
    };

    if (bal?.slv) {
      bal.slv({
        locBit: {
          idx: 'room-capacity-exceeded',
          src: payload.locationId,
          val: -1,
          dat: receipt,
        },
      });
    }
    return cpy;
  }

  const receipt: CapacityValidationReceipt = {
    allowed: true,
    locationId: payload.locationId,
    currentOccupancy: location.occupants.length,
    maxOccupancy: location.maxOccupancy,
  };

  if (bal?.slv) {
    bal.slv({
      locBit: {
        idx: 'validate-capacity-ok',
        src: payload.locationId,
        val: location.occupants.length,
        dat: receipt,
      },
    });
  }

  return cpy;
};

export const assertAxialInvariant = (input: { q: number; r: number; s?: number }): AxialCoord => {
  if (!Number.isInteger(input.q) || !Number.isInteger(input.r)) {
    throw new TypeError('[FLOAT_PROHIBITED] Axial coordinates must be strict integers');
  }

  const derivedS = -input.q - input.r;
  if (input.s !== undefined) {
    if (!Number.isInteger(input.s)) {
      throw new TypeError('[FLOAT_PROHIBITED] Axial coordinates must be strict integers');
    }
    if (input.q + input.r + input.s !== 0) {
      throw new Error(`[AXIAL_INVARIANT_BREACH] Coordinates (${input.q}, ${input.r}, ${input.s}) violate q + r + s === 0`);
    }
  }

  return {
    q: input.q,
    r: input.r,
    s: derivedS,
  };
};

export const computeAxialDistance = (a: AxialCoord, b: AxialCoord): number => {
  return Math.max(
    Math.abs(a.q - b.q),
    Math.abs(a.r - b.r),
    Math.abs(a.s - b.s)
  );
};

export const evaluateMoveMovement = async (
  cpy: MovementModel,
  bal: MovementBit<MoveEvaluationQuery>,
  ste: State
): Promise<MovementModel> => {
  const query = bal?.dat;
  if (!query?.entityId || !query?.origin || !query?.destination) {
    if (bal?.slv) bal.slv({ movBit: { idx: 'evaluate-move-err', src: 'MISSING_PAYLOAD', val: -1 } });
    return cpy;
  }

  const originCoord = assertAxialInvariant(query.origin);
  const destCoord = assertAxialInvariant(query.destination);
  const dist = computeAxialDistance(originCoord, destCoord);

  // 1. Locomotion Tier Gate: IMMOBILIZED actors cannot execute movement
  if (query.locomotionTier === 'IMMOBILIZED') {
    const errorMsg = `>> [LOCOMOTION_IMMOBILIZED_BREACH] Entity ${query.entityId} has locomotionTier IMMOBILIZED, cannot execute movement!`;
    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: errorMsg,
      }).catch(() => {});
    }

    const receipt: MoveEvaluationReceipt = {
      allowed: false,
      entityId: query.entityId,
      axialDistance: dist,
      traversalCostUnits: 0,
      error: errorMsg,
    };

    if (bal?.slv) bal.slv({ movBit: { idx: 'move-evaluation-fail', dat: receipt } });
    return cpy;
  }

  // 2. Cross-Room Portal Occlusion Gate
  if (query.destinationLocationId && query.destinationLocationId !== query.sourceLocationId) {
    if (query.portalState === 'CLOSED' || query.portalState === 'LOCKED') {
      const errorMsg = `>> [PORTAL_OCCLUDED_BREACH] Portal ${query.portalId || 'unknown'} between ${query.sourceLocationId} and ${query.destinationLocationId} is ${query.portalState}!`;
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt('[Console action] Update Console', {
          idx: 'cns00',
          src: errorMsg,
        }).catch(() => {});
      }

      const receipt: MoveEvaluationReceipt = {
        allowed: false,
        entityId: query.entityId,
        axialDistance: dist,
        traversalCostUnits: 0,
        error: errorMsg,
      };

      if (bal?.slv) bal.slv({ movBit: { idx: 'move-evaluation-fail', dat: receipt } });
      return cpy;
    }
  }

  // 3. Traversal Cost Calculation (LIMPING doubles cost)
  const speedMultiplier = query.locomotionTier === 'LIMPING' ? 2 : 1;
  const costUnits = dist * speedMultiplier * 100;

  const receipt: MoveEvaluationReceipt = {
    allowed: true,
    entityId: query.entityId,
    axialDistance: dist,
    traversalCostUnits: costUnits,

      movBit: {
        idx: 'move-evaluation-ok',
        src: query.entityId,
        val: dist,
        dat: receipt,
      },
    });
  }

  return cpy;
};
