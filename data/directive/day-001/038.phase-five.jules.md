function logToConsolePane(lines: string[]): void {
  if (typeof global !== 'undefined' && global.LIBRARY && typeof global.LIBRARY.hunt === 'function') {

      idx: 'cns00',
      lst: lines

export function executeStage(stage: StageConfig): GauntletStageResult {
  const start = process.hrtime.bigint();
  try {
    const stdout = execSync(stage.command, { stdio: 'pipe' }).toString().trim();
    if (stage.id.includes('FENCE') && stdout.length > 0) {
      throw new Error(`Uncommitted files detected in protected boundary:\n${stdout}`);
    }
    const end = process.hrtime.bigint();
    const durationMs = Number(end - start) / 1_000_000;
    return { stageId: stage.id, name: stage.name, passed: true, durationMs };
  } catch (err: any) {
    const end = process.hrtime.bigint();
    const durationMs = Number(end - start) / 1_000_000;
    const errorMsg = err.stderr ? err.stderr.toString().trim() : err.message;
    return { stageId: stage.id, name: stage.name, passed: false, durationMs, error: errorMsg };
  }
}

export async function runGauntlet(cpy: any, bal: GauntletBit, ste?: any): Promise<any> {

    '>> =================================================================='

  const results: GauntletStageResult[] = [];
  let allPassed = true;

  for (const stage of GAUNTLET_STAGES) {
    logToConsolePane([`>> [STAGE:RUN] ${stage.name}...`]);
    const res = executeStage(stage);
    results.push(res);

    if (res.passed) {
      const line = `>> [STAGE:PASS] ${res.name.padEnd(38)} [${res.durationMs.toFixed(2)}ms]`;
      receipts.push(line);
      logToConsolePane([line]);
    } else {
      allPassed = false;
      const line = `>> [STAGE:FAIL] ${res.name.padEnd(38)} [BREACH]`;
      receipts.push(line);
      if (res.error) {
        receipts.push(`>>   Error: ${res.error.split('\n')[0]}`);
      }
      logToConsolePane([line, `>>   Error: ${res.error}`]);
      break; // Fail closed on first violation
    }
  }

  receipts.push('>> ------------------------------------------------------------------');
  if (allPassed) {
    receipts.push('>> [VERDICT:PASS] 100% OF VERIFICATION GATES CLEARED AT T_0 = $0.00');
    receipts.push('>> [MERKLE:SEAL] ENGINE REPOSITORY INTEGRITY CERTIFIED GREEN');
  } else {
    receipts.push('>> [VERDICT:FAIL] GAUNTLET TRIPPED. EXECUTION HALTED AT CHEAPEST STAGE');
  }
  receipts.push('>> ==================================================================');
  logToConsolePane(receipts.slice(-4));

  bal.val = allPassed ? 0 : -1;
  bal.lst = receipts;
  bal.dat = {
    stages: results,
    passed: allPassed,
    totalDurationMs: results.reduce((acc, r) => acc + r.durationMs, 0)
  };

  if (bal.slv) bal.slv({ gntBit: bal });
  return cpy;
}
