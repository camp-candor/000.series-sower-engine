import {
  WorldSnapshot,
  EventEnvelope,
  ActorRuntimeState,
  ActorMovedPayload,
  SomaticAppliedPayload,
} from './events';

/**
 * Clamps basis-point values strictly to integer bounds [0, 10000].
 */
function clampBps(val: number): number {
  if (val <= 0) return 0;
  if (val >= 10000) return 10000;
  return Math.floor(val);
}

/**
 * Pure state reducer function: S_{n+1} = R(S_n, E_n, seed)
 * STRICT INVARIANT: Zero I/O, zero async, zero Date.now(), zero Math.random().
 */
export function reduceWorldState(
  prevState: WorldSnapshot,
  event: EventEnvelope
): WorldSnapshot {
  const nextActors = new Map<string, ActorRuntimeState>(prevState.actors);

  switch (event.eventType) {
    case 'ACTOR_MOVED': {
      const payload = event.payload as ActorMovedPayload;
      const current = nextActors.get(payload.entityId) ?? {
        entityId: payload.entityId,
        currentCellId: payload.fromCellId,
        fatigueBps: 0,
        painBps: 0,
        activeTrauma: [],
        lastActiveTick: event.tick,
      };

      const updated: ActorRuntimeState = {
        ...current,
        currentCellId: payload.toCellId,
        lastActiveTick: event.tick,
      };

      nextActors.set(payload.entityId, updated);
      break;
    }

    case 'SOMATIC_APPLIED': {
      const payload = event.payload as SomaticAppliedPayload;
      const current = nextActors.get(payload.entityId) ?? {
        entityId: payload.entityId,
        currentCellId: 0,
        fatigueBps: 0,
        painBps: 0,
        activeTrauma: [],
        lastActiveTick: event.tick,
      };

      const combinedTrauma = Array.from(
        new Set([...current.activeTrauma, ...payload.addedTrauma])
      ).sort();

      const updated: ActorRuntimeState = {
        ...current,
        fatigueBps: clampBps(current.fatigueBps + payload.fatigueDeltaBps),
        painBps: clampBps(current.painBps + payload.painDeltaBps),
        activeTrauma: combinedTrauma,
        lastActiveTick: event.tick,
      };

      nextActors.set(payload.entityId, updated);
      break;
    }

    case 'ACTION_REJECTED':
      // Physical coordinates and somatic gauges do not mutate upon rejection
      break;

    case 'TICK_COMPLETED':
      break;

    default:
      break;
  }

  return {
    tick: Math.max(prevState.tick, event.tick),
    actors: nextActors,
    lastEventHash: event.eventHash,
  };
}

/**

export function foldEvents(
  initialState: WorldSnapshot,
  events: readonly EventEnvelope[]
): WorldSnapshot {
  return events.reduce((state, evt) => reduceWorldState(state, evt), initialState);
}

  prevHash: string,
  eventId: string,
  streamId: string,
  tick: number,
  eventType: string,
  payloadJson: string

  const rawString = `${prevHash}:${eventId}:${streamId}:${tick}:${eventType}:${payloadJson}`;
  const encoder = new TextEncoder();
  const data = encoder.encode(rawString);
  const digestBuffer = await crypto.subtle.digest('SHA-256', data);

  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

          seed, payload_json, prev_event_hash, event_hash, created_at_ms
        ) VALUES (?, ?, ?, 'ACTION_REJECTED', ?, NULL, ?, ?, ?, ?, ?)
      `).bind(
        eventId,
        streamId,
        tick,
        actorId,
        seed,
        payloadJson,
        prevEventHash,
        hash,
        createdAtMs
      );

      await stmt.run();

      return {
        success: true,
        committedEventIds: [eventId],
      };
    } catch (err: unknown) {
      return {
        success: false,
        committedEventIds: [],
        errorCode: 'ERR_REJECTION_PERSISTENCE_FAILED',
        errorMessage: err instanceof Error ? err.message : String(err),
      };
    }
  }
}
