import type { LocationModel } from '../location.model';
import type LocationBit from '../fce/location.bit';
import type State from '../../99.core/state';
import type { LocationRecord, PortalState, CapacityAuditReceipt } from '../fce/location.interface';

export const createDefaultLocationRecord = (locationId: string, name = locationId): LocationRecord => ({
  locationId,
  name,
  maxOccupancy: 2,
  acousticDampingCdB: 1500,
  ambientLuxBps: 2000,
  portals: {},
  occupants: [],
  updatedAtMs: 0,
});

export const initLocation = (cpy: LocationModel, bal: LocationBit, _ste: State): LocationModel => {
  cpy.locations = {};
  if (bal?.slv) bal.slv({ locBit: { idx: 'init-location', val: 0 } });
  return cpy;
};

export const updateLocation = (cpy: LocationModel, bal: LocationBit, _ste: State): LocationModel => {
  if (bal?.slv) bal.slv({ locBit: { idx: 'update-location' } });
  return cpy;
};

export const registerLocation = (
  cpy: LocationModel,
  bal: LocationBit<Partial<LocationRecord> & { locationId: string }>,
  _ste: State
): LocationModel => {
  const payload = bal?.dat;
  if (!payload?.locationId) {
    if (bal?.slv) bal.slv({ locBit: { idx: 'register-location-err', src: 'MISSING_LOCATION_ID', val: -1 } });
    return cpy;
  }

  const existing = cpy.locations[payload.locationId] || createDefaultLocationRecord(payload.locationId);
  const updated: LocationRecord = {
    ...existing,
    ...payload,
    portals: { ...existing.portals, ...(payload.portals || {}) },
    occupants: payload.occupants ? Array.from(new Set(payload.occupants)) : existing.occupants,
    updatedAtMs: bal?.val && bal.val > 0 ? bal.val : existing.updatedAtMs + 1,
  };

  cpy.locations[payload.locationId] = updated;

  if (bal?.slv) {
    bal.slv({
      locBit: {
        idx: 'register-location-ok',
        src: payload.locationId,
        val: updated.occupants.length,
        dat: updated,
      },
    });
  }
  return cpy;
};

export const setPortalStateLocation = (
  cpy: LocationModel,
  bal: LocationBit<{ locationId: string; targetPortalId: string; state: PortalState }>,
  _ste: State
): LocationModel => {
  const payload = bal?.dat;
  if (!payload?.locationId || !payload?.targetPortalId || !payload?.state) {
    if (bal?.slv) bal.slv({ locBit: { idx: 'set-portal-state-err', src: 'INVALID_PAYLOAD', val: -1 } });
    return cpy;
  }

  const loc = cpy.locations[payload.locationId] || createDefaultLocationRecord(payload.locationId);
  loc.portals[payload.targetPortalId] = payload.state;
  loc.updatedAtMs += 1;
  cpy.locations[payload.locationId] = loc;

  if (bal?.slv) {
    bal.slv({
      locBit: {
        idx: 'set-portal-state-ok',
        src: payload.locationId,
        dat: { locationId: payload.locationId, portalId: payload.targetPortalId, state: payload.state },
      },
    });
  }
  return cpy;
};

export const validateCapacityLocation = async (
  cpy: LocationModel,
  bal: LocationBit<{ locationId: string; incomingEntityId?: string }>,
  ste: State
): Promise<LocationModel> => {
  const payload = bal?.dat;
  if (!payload?.locationId) {
    if (bal?.slv) bal.slv({ locBit: { idx: 'validate-capacity-err', src: 'MISSING_LOCATION_ID', val: -1 } });
    return cpy;
  }

  const loc = cpy.locations[payload.locationId] || createDefaultLocationRecord(payload.locationId);
  const isAlreadyOccupant = payload.incomingEntityId ? loc.occupants.includes(payload.incomingEntityId) : false;
  const projectedCount = isAlreadyOccupant ? loc.occupants.length : loc.occupants.length + 1;
  const allowed = projectedCount <= loc.maxOccupancy;

  const receipt: CapacityAuditReceipt = {
    locationId: payload.locationId,
    maxOccupancy: loc.maxOccupancy,
    currentCount: projectedCount,
    allowed,
    error: allowed
      ? undefined
      : `>> [ROOM_CAPACITY_EXCEEDED] Location '${payload.locationId}' capacity saturated (${projectedCount}/${loc.maxOccupancy})`,
  };

  if (!allowed && ste && typeof ste.hunt === 'function') {
    await ste.hunt('[Console action] Update Console', {
      idx: 'cns00',
      src: receipt.error,
    }).catch(() => {});
  }

  if (bal?.slv) {
    bal.slv({
      locBit: {
        idx: allowed ? 'validate-capacity-ok' : 'validate-capacity-exceeded',
        src: payload.locationId,
        val: allowed ? 0 : -1,
        dat: receipt,
      },
    });
  }
  return cpy;
};

import type { MovementModel } from '../movement.model';
import type MovementBit from '../fce/movement.bit';
import type State from '../../99.core/state';
import type {
  AxialCoord,
  ActorSpatialPosition,
  MoveEvaluationQuery,
  MoveEvaluationReceipt,
} from '../fce/movement.interface';

export const assertAxialInvariant = (c: { q: number; r: number; s?: number }): AxialCoord => {
  if (!Number.isInteger(c.q) || !Number.isInteger(c.r)) {
    throw new Error(`[FLOAT_PROHIBITED] Axial coordinates must be integers. Received (${c.q}, ${c.r})`);
  }
  const s = c.s !== undefined ? c.s : -c.q - c.r;
  if (c.q + c.r + s !== 0) {
    throw new Error(`[AXIAL_INVARIANT_BREACH] q + r + s must equal 0. Received: ${c.q} + ${c.r} + ${s} = ${c.q + c.r + s}`);
  }
  return { q: c.q, r: c.r, s };
};

export const computeAxialDistance = (a: AxialCoord, b: AxialCoord): number => {
  return Math.max(Math.abs(a.q - b.q), Math.abs(a.r - b.r), Math.abs(a.s - b.s));
};

export const initMovement = (cpy: MovementModel, bal: MovementBit, _ste: State): MovementModel => {
  cpy.positions = {};
  if (bal?.slv) bal.slv({ movBit: { idx: 'init-movement', val: 0 } });
  return cpy;
};

export const updateMovement = (cpy: MovementModel, bal: MovementBit, _ste: State): MovementModel => {
  if (bal?.slv) bal.slv({ movBit: { idx: 'update-movement' } });
  return cpy;
};

export const setActorPositionMovement = (
  cpy: MovementModel,
  bal: MovementBit<{ entityId: string; locationId: string; q: number; r: number; locomotionTier?: any }>,
  _ste: State
): MovementModel => {
  const payload = bal?.dat;
  if (!payload?.entityId || !payload?.locationId || payload.q === undefined || payload.r === undefined) {
    if (bal?.slv) bal.slv({ movBit: { idx: 'set-actor-position-err', src: 'MISSING_PAYLOAD', val: -1 } });
    return cpy;
  }

  const coord = assertAxialInvariant({ q: payload.q, r: payload.r });
  const existing = cpy.positions[payload.entityId];

  const updated: ActorSpatialPosition = {
    entityId: payload.entityId,
    locationId: payload.locationId,
    coord,
    locomotionTier: payload.locomotionTier || existing?.locomotionTier || 'UNRESTRICTED',
    updatedAtMs: bal?.val && bal.val > 0 ? bal.val : (existing?.updatedAtMs ?? 0) + 1,
  };

  cpy.positions[payload.entityId] = updated;

  if (bal?.slv) {
    bal.slv({
      movBit: {
        idx: 'set-actor-position-ok',
        src: payload.entityId,
        dat: updated,
      },
    });
  }
  return cpy;
};

export const evaluateMoveMovement = async (
  cpy: MovementModel,
  bal: MovementBit<MoveEvaluationQuery>,
  ste: State
): Promise<MovementModel> => {
  const q = bal?.dat;
  if (!q?.entityId || !q?.origin || !q?.destination) {
    if (bal?.slv) bal.slv({ movBit: { idx: 'evaluate-move-err', src: 'MISSING_QUERY', val: -1 } });
    return cpy;
  }

  const origin = assertAxialInvariant(q.origin);
  const destination = assertAxialInvariant(q.destination);
  const distance = computeAxialDistance(origin, destination);

  // 1. Locomotion Tier Gate: Immobilized characters cannot move
  if (q.locomotionTier === 'IMMOBILIZED') {
    const errorMsg = `>> [LOCOMOTION_IMMOBILIZED_BREACH] Entity '${q.entityId}' is IMMOBILIZED and cannot traverse hex cells`;
    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt('[Console action] Update Console', { idx: 'cns00', src: errorMsg }).catch(() => {});
    }
    const receipt: MoveEvaluationReceipt = {
      allowed: false,
      entityId: q.entityId,
      axialDistance: distance,
      traversalCostUnits: 0,
      error: errorMsg,
    };
    if (bal?.slv) bal.slv({ movBit: { idx: 'locomotion-immobilized-breach', val: -1, dat: receipt } });
    return cpy;
  }

  // 2. Portal Occlusion Gate: Cross-room movement requires OPEN portal
  if (q.destinationLocationId && q.destinationLocationId !== q.sourceLocationId) {
    if (q.portalState && q.portalState !== 'OPEN') {
      const errorMsg = `>> [PORTAL_OCCLUDED_BREACH] Portal '${q.portalId || 'unknown'}' between '${q.sourceLocationId}' and '${q.destinationLocationId}' is ${q.portalState}`;
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt('[Console action] Update Console', { idx: 'cns00', src: errorMsg }).catch(() => {});
      }
      const receipt: MoveEvaluationReceipt = {
        allowed: false,
        entityId: q.entityId,
        axialDistance: distance,
        traversalCostUnits: 0,
        error: errorMsg,
      };
      if (bal?.slv) bal.slv({ movBit: { idx: 'portal-occluded-breach', val: -1, dat: receipt } });
      return cpy;
    }
  }

  // 3. Compute Integer Traversal Cost: Limping doubles cost
  const multiplier = q.locomotionTier === 'LIMPING' ? 2 : 1;
  const traversalCostUnits = distance * multiplier * 100; // 100 CostUnits per standard hex step

  const receipt: MoveEvaluationReceipt = {
    allowed: true,
    entityId: q.entityId,
    axialDistance: distance,
    traversalCostUnits,
  };

  if (bal?.slv) {
    bal.slv({
      movBit: {
        idx: 'evaluate-move-ok',
        src: q.entityId,
        val: traversalCostUnits,
        dat: receipt,
      },
    });
  }
  return cpy;
};

export const initEntity = (cpy: EntityModel, bal: EntityBit, _ste: State): EntityModel => {
  cpy.entities = {};
  if (bal?.slv) bal.slv({ entBit: { idx: 'init-entity', val: 0 } });
  return cpy;
};

export const updateEntity = (cpy: EntityModel, bal: EntityBit, _ste: State): EntityModel => {
  if (bal?.slv) bal.slv({ entBit: { idx: 'update-entity' } });
  return cpy;
};

export const upsertEntity = (
  cpy: EntityModel,
  bal: EntityBit<{ composite: CompositeEntityState }>,
  _ste: State
): EntityModel => {
  const payload = bal?.dat?.composite;
  if (!payload?.entityId) {
    if (bal?.slv) bal.slv({ entBit: { idx: 'upsert-entity-err', src: 'MISSING_COMPOSITE', val: -1 } });
    return cpy;
  }

  cpy.entities[payload.entityId] = payload;

  if (bal?.slv) {
    bal.slv({
      entBit: {
        idx: 'upsert-entity-ok',
        src: payload.entityId,
        dat: payload,
      },
    });
  }
  return cpy;
};

export const getEntity = (
  cpy: EntityModel,
  bal: EntityBit<{ entityId: string }>,
  _ste: State
): EntityModel => {
  const entityId = bal?.dat?.entityId || bal?.src;
  const entity = entityId ? cpy.entities[entityId] : undefined;

  if (bal?.slv) {
    bal.slv({
      entBit: {
        idx: entity ? 'get-entity-ok' : 'get-entity-not-found',
        src: entityId,
        val: entity ? 0 : -1,
        dat: entity,
      },
    });
  }
  return cpy;
};

export const listEntitiesByLocation = (
  cpy: EntityModel,
  bal: EntityBit<{ locationId: string }>,
  _ste: State
): EntityModel => {
  const locationId = bal?.dat?.locationId || bal?.src || '';
  const matched = Object.values(cpy.entities).filter((e) => e.locationId === locationId);

  if (bal?.slv) {
    bal.slv({
      entBit: {
        idx: 'list-entities-by-location-ok',
        src: locationId,
        val: matched.length,
        lst: matched.map((m) => m.entityId),
        dat: matched,
      },
    });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './entity.action';
import { EntityModel } from './entity.model';
import * as Buzz from './entity.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: EntityModel = new EntityModel(),
  act: Act.Actions,
  state?: State
): EntityModel {
  switch (act.type) {
    case Act.INIT_ENTITY:
      return Buzz.initEntity(clone(model, true), act.bale, state!);

    case Act.UPDATE_ENTITY:
      return Buzz.updateEntity(clone(model, true), act.bale, state!);

    case Act.UPSERT_ENTITY:
      return Buzz.upsertEntity(clone(model, true), act.bale, state!);

    case Act.GET_ENTITY:
      return Buzz.getEntity(clone(model, true), act.bale, state!);

    case Act.LIST_ENTITIES_BY_LOCATION:
      return Buzz.listEntitiesByLocation(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}
