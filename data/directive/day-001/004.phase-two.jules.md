      seqBit: { idx: valid ? 'assert-expected-seq' : 'assert-expected-seq-fail', val: actualHead },
      report,
      err: report.error,
    });
  }
  return cpy;
};

export const verifySequenceContinuity = (
  cpy: SequenceModel,
  bal: SequenceBit,
  _ste: State,
): SequenceModel => {
  const sequences: number[] = Array.isArray(bal?.lst) ? bal.lst : (bal?.dat as any)?.sequences || [];

  if (sequences.length === 0) {
    const report: SequenceValidationReport = { valid: true, expectedSeq: 0, actualSeq: 0 };
    if (bal?.slv) bal.slv({ report });
    return cpy;
  }

  let valid = true;
  let error: string | undefined;

  for (let i = 1; i < sequences.length; i++) {
    const prev = sequences[i - 1];
    const curr = sequences[i];
    if (curr !== prev + 1) {
      valid = false;
      error = `[SEQUENCE_GAP_DETECTED] Non-monotonic sequence gap between #${prev} and #${curr}`;
      break;
    }
  }

  const report: SequenceValidationReport = {
    valid,
    expectedSeq: sequences[sequences.length - 1],
    actualSeq: sequences[0],
    error,
  };

  if (bal?.slv) {
    bal.slv({
      seqBit: { idx: valid ? 'verify-sequence-continuity' : 'verify-sequence-continuity-fail' },
      report,
      err: error,
    });
  }
  return cpy;
};

export const commitSequence = async (
  cpy: SequenceModel,
  bal: SequenceBit,
  ste: State,
): Promise<SequenceModel> => {
  const streamId = bal?.src || cpy.defaultStreamId;
  const newSeq = typeof bal?.val === 'number' ? bal.val : (bal?.dat as any)?.newSeq;
  const eventId = (bal?.dat as any)?.eventId || 'evt_unknown';
  const timestampMs = (bal?.dat as any)?.timestampMs || 0;

  const current = cpy.streams[streamId] || {
    streamId,
    headSeq: 0,
    lastEventId: '0'.repeat(64),
    updatedAtMs: 0,
  };

  if (newSeq !== current.headSeq + 1) {
    const err = `[SEQUENCE_VIOLATION] Cannot commit Seq #${newSeq}: current head is #${current.headSeq}`;
    await emitConsole(ste, `:: [SEQUENCE:COMMIT_FAIL] ${err}`);
    if (bal?.slv) bal.slv({ err });
    return cpy;
  }

  cpy.streams[streamId] = {
    streamId,
    headSeq: newSeq,
    lastEventId: eventId,
    updatedAtMs: timestampMs,
  };

  await emitConsole(ste, `>> [SEQUENCE:COMMIT] [OK] Stream '${streamId}' advanced to Seq #${newSeq} (${eventId})`);

  if (bal?.slv) {
    bal.slv({
      seqBit: { idx: 'commit-sequence', src: streamId, val: newSeq },
      stream: cpy.streams[streamId],
    });
  }
  return cpy;
};

export const resetSequence = (
  cpy: SequenceModel,
  bal: SequenceBit,
  _ste: State,
): SequenceModel => {
  const streamId = bal?.src || cpy.defaultStreamId;
  cpy.streams[streamId] = {
    streamId,
    headSeq: 0,
    lastEventId: '0'.repeat(64),
    updatedAtMs: 0,
  };

  if (bal?.slv) {
    bal.slv({ seqBit: { idx: 'reset-sequence', src: streamId, val: 0 } });
  }
  return cpy;
};
