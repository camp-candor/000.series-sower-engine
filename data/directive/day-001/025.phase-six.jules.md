      a,
      b,
      product,
      quotient,
      remainder,

/**
 * 20.arithmetic.unit - Pure Banker's Rounding Scalar Actions
 */

import {
  BASIS_POINT_DENOMINATOR,
  TIE_BREAK_REMAINDER,
  type BasisPoint,
  type ScalingDivisionResult,
  toBasisPoint
} from './fce/arithmetic.interface.js';

export function arbitrateRemainder(
  quotient: number,
  remainder: number,
  halfDivisor: number = TIE_BREAK_REMAINDER
): number {
  if (remainder < halfDivisor) {
    return quotient;
  }
  if (remainder > halfDivisor) {
    return quotient + 1;
  }
  return quotient % 2 === 0 ? quotient : quotient + 1;
}

export function evaluateScalingDivision(a: number, b: number): ScalingDivisionResult {
  if (!Number.isInteger(a) || !Number.isInteger(b)) {
    throw new TypeError(`[FLOAT_PROHIBITED] Operands must be integers. Received: a=${a}, b=${b}`);
  }
  if (a < 0 || b < 0) {
    throw new RangeError(`[NEGATIVE_PROHIBITED] Basis points cannot be negative. Received: a=${a}, b=${b}`);
  }
  if (a > Number.MAX_SAFE_INTEGER || b > Number.MAX_SAFE_INTEGER) {
    throw new RangeError('[OVERFLOW_HAZARD] Operand exceeds Number.MAX_SAFE_INTEGER');
  }

  const product = a * b;
  const quotient = Math.floor(product / BASIS_POINT_DENOMINATOR);
  const remainder = product % BASIS_POINT_DENOMINATOR;
  const tieDetected = remainder === TIE_BREAK_REMAINDER;

  const rounded = arbitrateRemainder(quotient, remainder, TIE_BREAK_REMAINDER);
  const roundedOutput = toBasisPoint(rounded);

  const receipt: ScalingDivisionResult = {
    quotient,
    remainder,
    tieDetected,
    roundedOutput
  };

  return Object.freeze(receipt);
}

export function scaleBasisPoints(a: number, b: number): BasisPoint {
  const receipt = evaluateScalingDivision(a, b);
  return receipt.roundedOutput;
}

/**
 * Evaluates chained scalar multiplications enforcing strict left-associativity:
 * Scale(Scale(Scale(val, m1), m2), m3)
 * Rounds and clamps at each intermediate step to prevent integer overflow beyond 10^8.
 */
export function scaleBasisPointsChained(
  baseValue: number,
  ...multipliers: readonly number[]
): BasisPoint {
  if (multipliers.length === 0) {
    throw new RangeError('[EMPTY_MULTIPLIERS] Chained scaling requires at least one multiplier');
  }

  return multipliers.reduce<BasisPoint>(
    (accumulated, multiplier) => scaleBasisPoints(accumulated, multiplier),
    toBasisPoint(baseValue)
  );
}

export function divRoundHalfEven(numerator: number, denominator: number): number {
  if (denominator === 0) {
    throw new RangeError('[DIV_ZERO] Division by zero in integer arithmetic');
  }
  if (!Number.isInteger(numerator) || !Number.isInteger(denominator)) {
    throw new TypeError(
      `[FLOAT_PROHIBITED] Operands must be integers. Received: ${numerator} / ${denominator}`
    );
  }
  if (
    Math.abs(numerator) > Number.MAX_SAFE_INTEGER ||
    Math.abs(denominator) > Number.MAX_SAFE_INTEGER
  ) {
    throw new RangeError('[OVERFLOW_HAZARD] Operands exceed Number.MAX_SAFE_INTEGER');
  }

  let n = numerator;
  let d = denominator;

  if (d < 0) {
    n = -n;
    d = -d;
  }

  const quotient = Math.floor(n / d);
  const remainder = ((n % d) + d) % d;

  if (remainder === 0) {
    return quotient;
  }

  const doubleR = remainder * 2;
  if (doubleR < d) {
    return quotient;
  }
  if (doubleR > d) {
    return quotient + 1;
  }

  return quotient % 2 === 0 ? quotient : quotient + 1;
}
