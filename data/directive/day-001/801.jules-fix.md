```sql
-- Stage A: Canonical Narrative Facts Ledger & Scene Dependency Invalidation

CREATE TABLE IF NOT EXISTS facts (
    id TEXT PRIMARY KEY,
    claim_text TEXT NOT NULL,
    scope TEXT NOT NULL CHECK(scope IN ('CHARACTER', 'LOCATION', 'LOGISTIC', 'LAW')),
    authority_level TEXT NOT NULL CHECK(authority_level IN ('AXIOM', 'CANON', 'EVENT', 'RUNTIME_STATE', 'PROJECTION', 'SPECULATION')),
    standing TEXT NOT NULL DEFAULT 'ACTIVE_ENDORSED' CHECK(standing IN ('ACTIVE_ENDORSED', 'FLAGGED_FOR_REVIEW', 'DEPRECATED_ARCHIVED')),
    author TEXT NOT NULL,
    timestamp_ms INTEGER NOT NULL,
    version INTEGER NOT NULL DEFAULT 1
);

CREATE INDEX IF NOT EXISTS idx_facts_scope ON facts(scope, standing);

CREATE TABLE IF NOT EXISTS scenes (
    id TEXT PRIMARY KEY,
    episode_num INTEGER NOT NULL,
    scene_num INTEGER NOT NULL,
    text TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'clean' CHECK(status IN ('clean', 'dirty', 'reviewed')),
    is_dirty INTEGER NOT NULL DEFAULT 0 CHECK(is_dirty IN (0, 1))
);

CREATE INDEX IF NOT EXISTS idx_scenes_dirty ON scenes(is_dirty, status);

CREATE TABLE IF NOT EXISTS scene_fact_dependencies (
    scene_id TEXT NOT NULL REFERENCES scenes(id) ON DELETE CASCADE,
    fact_id TEXT NOT NULL REFERENCES facts(id) ON DELETE CASCADE,
    entity_type TEXT NOT NULL CHECK(entity_type IN ('character', 'location', 'possession', 'grievance', 'axiom')),
    pinned_version INTEGER NOT NULL,
    PRIMARY KEY (scene_id, fact_id)
);

CREATE INDEX IF NOT EXISTS idx_deps_fact_pinned ON scene_fact_dependencies(fact_id, pinned_version);

-- Invalidation Trigger: Fires when fact version increments or when standing is retired
CREATE TRIGGER IF NOT EXISTS trg_fact_version_bump
AFTER UPDATE ON facts
FOR EACH ROW
WHEN (NEW.version > OLD.version) OR (OLD.standing != 'DEPRECATED_ARCHIVED' AND NEW.standing = 'DEPRECATED_ARCHIVED')
BEGIN
    UPDATE scenes
    SET is_dirty = 1,
        status = 'dirty'
    WHERE id IN (
        SELECT sfd.scene_id
        FROM scene_fact_dependencies sfd
        WHERE sfd.fact_id = NEW.id
          AND sfd.pinned_version < NEW.version
    );
END;

-- Passive Centrality Telemetry View: Blast-radius in-degree calculation
CREATE VIEW IF NOT EXISTS v_fact_centrality_telemetry AS
SELECT 
    f.id AS fact_id,
    f.scope,
    f.version AS current_version,
    f.standing,
    COUNT(sfd.scene_id) AS dependent_scenes_count
FROM facts f
LEFT JOIN scene_fact_dependencies sfd ON f.id = sfd.fact_id
GROUP BY f.id, f.scope, f.version, f.standing;
```

```typescript
import {
  FactRecord,
  SceneRecord,
  SceneFactDependencyRecord,
  FactCentralityTelemetry,
} from './types.js';

export class LedgerSchemaManager {
  constructor(private db: D1Database) {}

  async initializeCoreSchema(migrationSql0001: string, migrationSql0004: string): Promise<void> {
    try {
      await this.db.exec(migrationSql0001);
      await this.db.exec(migrationSql0004);
      console.log('>> [D1:MIGRATE] Phase 3 Relational Core & Clocks migrations executed [OK]');
    } catch (err: any) {
      console.error(`>> [D1:MIGRATE_ERROR] ${err.message} [FAIL]`);
      throw err;
    }
  }

  async insertFact(fact: Omit<FactRecord, 'version'> & { version?: number }): Promise<void> {
    const version = fact.version ?? 1;
    await this.db
      .prepare(
        `INSERT INTO facts (id, claim_text, scope, authority_level, standing, author, timestamp_ms, version)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?)`
      )
      .bind(
        fact.id,
        fact.claim_text,
        fact.scope,
        fact.authority_level,
        fact.standing,
        fact.author,
        fact.timestamp_ms,
        version
      )
      .run();
  }

  async updateFactVersion(factId: string, newVersion: number, newClaimText?: string): Promise<void> {
    const current = await this.db
      .prepare('SELECT version FROM facts WHERE id = ?')
      .bind(factId)
      .first<{ version: number }>();

    if (!current) {
      throw new Error(`[LEDGER_ERROR] Fact not found: ${factId}`);
    }

    if (newVersion <= current.version) {
      throw new Error(
        `[ILLEGAL_VERSION_MUTATION] NEW.version (${newVersion}) must be strictly greater than OLD.version (${current.version})`
      );
    }

    if (newClaimText) {
      await this.db
        .prepare('UPDATE facts SET version = ?, claim_text = ?, timestamp_ms = ? WHERE id = ?')
        .bind(newVersion, newClaimText, Date.now(), factId)
        .run();
    } else {
      await this.db
        .prepare('UPDATE facts SET version = ?, timestamp_ms = ? WHERE id = ?')
        .bind(newVersion, Date.now(), factId)
        .run();
    }
  }

  async setFactStanding(factId: string, standing: FactRecord['standing']): Promise<void> {
    await this.db
      .prepare('UPDATE facts SET standing = ?, timestamp_ms = ? WHERE id = ?')
      .bind(standing, Date.now(), factId)
      .run();
  }

  async registerScene(scene: SceneRecord): Promise<void> {
    await this.db
      .prepare(
        `INSERT INTO scenes (id, episode_num, scene_num, text, status, is_dirty)
         VALUES (?, ?, ?, ?, ?, ?)`
      )
      .bind(
        scene.id,
        scene.episode_num,
        scene.scene_num,
        scene.text,
        scene.status,
        scene.is_dirty
      )
      .run();
  }

  async declareDependency(dep: SceneFactDependencyRecord): Promise<void> {
    await this.db
      .prepare(
        `INSERT INTO scene_fact_dependencies (scene_id, fact_id, entity_type, pinned_version)
         VALUES (?, ?, ?, ?)`
      )
      .bind(dep.scene_id, dep.fact_id, dep.entity_type, dep.pinned_version)
      .run();
  }

  async getScene(sceneId: string): Promise<SceneRecord null |> {
    return this.db
      .prepare('SELECT id, episode_num, scene_num, text, status, is_dirty FROM scenes WHERE id = ?')
      .bind(sceneId)
      .first<SceneRecord>();
  }

  async getFactCentralityTelemetry(): Promise<FactCentralityTelemetry[]> {
    const res = await this.db.prepare('SELECT * FROM v_fact_centrality_telemetry').all<FactCentralityTelemetry>();
    return res.results || [];
  }
}
```
