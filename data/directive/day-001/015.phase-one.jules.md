import type { TraumaModel } from '../trauma.model';
import type TraumaBit from '../fce/trauma.bit';
import type State from '../../99.core/state';
import type {
  SomaticTraumaState,
  TraumaDeltaPayload,
  LocomotionTier,
  RefusalStrainReport,
} from '../fce/trauma.interface';

export const clampBps = (val: number): number => {
  if (!Number.isInteger(val)) {
    throw new Error(`[FLOAT_PROHIBITED] Basis points must be integer. Received: ${val}`);
  }
  return Math.max(0, Math.min(10000, val));
};

export const bankersRoundBps = (numerator: number, denominator: number): number => {
  if (denominator === 0) throw new Error('[DIV_ZERO] Division by zero in basis-point arithmetic');
  const quotient = numerator / denominator;
  const floor = Math.floor(quotient);
  const remainder = Math.abs(quotient - floor);

  if (Math.abs(remainder - 0.5) < 1e-9) {
    return floor % 2 === 0 ? floor : floor + 1;
  }
  return Math.round(quotient);
};

export const deriveLocomotionTier = (legsAvailable: number, painBps: number): LocomotionTier => {
  if (legsAvailable <= 0 || painBps >= 9000) return 'IMMOBILIZED';
  if (legsAvailable === 1 || painBps >= 5000) return 'LIMPING';
  return 'UNRESTRICTED';
};

export const createDefaultSomaticState = (entityId: string): SomaticTraumaState => ({
  entityId,
  armsAvailable: 2,
  legsAvailable: 2,
  concussionLevel: 0,
  fatigueBps: 0,
  painBps: 0,
  refusalStrainBps: 0,
  locomotionTier: 'UNRESTRICTED',
  updatedAtMs: 0,
});

export const initTrauma = (cpy: TraumaModel, bal: TraumaBit, _ste: State): TraumaModel => {
  cpy.entities = {};
  if (bal?.slv) bal.slv({ trmBit: { idx: 'init-trauma', val: 0 } });
  return cpy;
};

export const updateTrauma = (cpy: TraumaModel, bal: TraumaBit, _ste: State): TraumaModel => {
  if (bal?.slv) bal.slv({ trmBit: { idx: 'update-trauma' } });
  return cpy;
};

export const applyTrauma = (cpy: TraumaModel, bal: TraumaBit<TraumaDeltaPayload>, _ste: State): TraumaModel => {
  const payload = bal?.dat;
  if (!payload?.entityId) {
    if (bal?.slv) bal.slv({ trmBit: { idx: 'apply-trauma-err', src: 'MISSING_ENTITY_ID', val: -1 } });
    return cpy;
  }

  const existing = cpy.entities[payload.entityId] || createDefaultSomaticState(payload.entityId);

  const rawArms = existing.armsAvailable + (payload.deltaArms ?? 0);
  const rawLegs = existing.legsAvailable + (payload.deltaLegs ?? 0);
  const rawConcussion = existing.concussionLevel + (payload.deltaConcussion ?? 0);

  const armsAvailable = Math.max(0, Math.min(2, rawArms));
  const legsAvailable = Math.max(0, Math.min(2, rawLegs));
  const concussionLevel = Math.max(0, Math.min(3, rawConcussion));

  const fatigueBps = clampBps(existing.fatigueBps + (payload.deltaFatigueBps ?? 0));
  const painBps = clampBps(existing.painBps + (payload.deltaPainBps ?? 0));
  const refusalStrainBps = clampBps(existing.refusalStrainBps + (payload.deltaRefusalStrainBps ?? 0));

  const locomotionTier = deriveLocomotionTier(legsAvailable, painBps);

  const updated: SomaticTraumaState = {
    entityId: payload.entityId,
    armsAvailable,
    legsAvailable,
    concussionLevel,
    fatigueBps,
    painBps,
    refusalStrainBps,
    locomotionTier,
    updatedAtMs: (bal?.val && bal.val > 0) ? bal.val : existing.updatedAtMs + 1,
  };

  cpy.entities[payload.entityId] = updated;

  if (bal?.slv) {
    bal.slv({
      trmBit: {
        idx: 'apply-trauma',
        src: payload.entityId,
        val: painBps,
        dat: updated,
      },
    });
  }
  return cpy;
};

export const recoverTrauma = (cpy: TraumaModel, bal: TraumaBit<TraumaDeltaPayload>, _ste: State): TraumaModel => {
  const payload = bal?.dat;
  if (!payload?.entityId || !cpy.entities[payload.entityId]) {
    if (bal?.slv) bal.slv({ trmBit: { idx: 'recover-trauma-err', src: 'ENTITY_NOT_FOUND', val: -1 } });
    return cpy;
  }

  const existing = cpy.entities[payload.entityId];
  const fatigueBps = clampBps(existing.fatigueBps - (payload.deltaFatigueBps ?? 0));
  const painBps = clampBps(existing.painBps - (payload.deltaPainBps ?? 0));
  const refusalStrainBps = clampBps(existing.refusalStrainBps - (payload.deltaRefusalStrainBps ?? 0));
  const locomotionTier = deriveLocomotionTier(existing.legsAvailable, painBps);

  const updated: SomaticTraumaState = {
    ...existing,
    fatigueBps,
    painBps,
    refusalStrainBps,
    locomotionTier,
    updatedAtMs: existing.updatedAtMs + 1,
  };

  cpy.entities[payload.entityId] = updated;

  if (bal?.slv) {
    bal.slv({
      trmBit: {
        idx: 'recover-trauma',
        src: payload.entityId,
        val: fatigueBps,
        dat: updated,
      },
    });
  }
  return cpy;
};

export const evaluateRefusalStrainTrauma = (
  cpy: TraumaModel,
  bal: TraumaBit<{ entityId: string }>,
  _ste: State
): TraumaModel => {
  const entityId = bal?.dat?.entityId || bal?.src || '';
  const current = cpy.entities[entityId];

  if (!current) {
    if (bal?.slv) bal.slv({ trmBit: { idx: 'evaluate-refusal-strain-err', src: 'ENTITY_NOT_FOUND', val: -1 } });
    return cpy;
  }

  const report: RefusalStrainReport = {
    entityId,
    refusalStrainBps: current.refusalStrainBps,
    isStrained: current.refusalStrainBps >= 8000,
    thresholdBps: 8000,
  };

  if (bal?.slv) {
    bal.slv({
      trmBit: {
        idx: 'evaluate-refusal-strain-trauma',
        src: entityId,
        val: current.refusalStrainBps,
        dat: report,
      },
    });
  }
  return cpy;
};

export const VERB_FAMILIES: Record<VerbFamilyKey, readonly string[]> = {
  LOCOMOTION_RAPID: ['SPRINT', 'VAULT', 'LEAP', 'LUNGE', 'RUN', 'DODGE', 'SCRAMBLE', 'BOUND', 'RAPID_STAIRS'],
  BIMANUAL_GRIP: ['CLIMB', 'BIMANUAL_LIFT', 'GRAPPLE_TWO_HANDED', 'TWO_HANDED_AIM', 'HOIST', 'EMBRACE', 'WIELD_TWO_HANDED'],

export const CONDITION_VERB_ROLLUP: Record<string, readonly string[]> = {
  SEVERED_LEFT_ARM: [...VERB_FAMILIES.BIMANUAL_GRIP],
  SEVERED_RIGHT_ARM: [...VERB_FAMILIES.BIMANUAL_GRIP],
  PARALYZED_LEFT_ARM: [...VERB_FAMILIES.BIMANUAL_GRIP],
  PARALYZED_RIGHT_ARM: [...VERB_FAMILIES.BIMANUAL_GRIP],

  FRACTURED_FEMUR: [...VERB_FAMILIES.LOCOMOTION_RAPID],
  NEUROPATHY_SHUFFLE: [...VERB_FAMILIES.LOCOMOTION_RAPID],
  STIFF_LEFT_KNEE: ['SPRINT', 'VAULT', 'LEAP', 'LUNGE'],
  CRUSHED_LARYNX: [...VERB_FAMILIES.VOCAL_DELIVERY],
  CONCUSSION_SEVERE: ['SPRINT', 'VAULT', 'AIM_RANGED', 'PICK_LOCK'],
};

export const compileSomaticMask = (
  entityId: string,
  wounds: WoundRecord[],
  armsAvailable: number,
  legsAvailable: number,
  locomotionTier: LocomotionTier
): SomaticMask => {
  const bannedSet = new Set<string>();
  const woundTags = new Set<string>();

  for (const w of wounds) {
    woundTags.add(w.conditionTag);
    const rollup = CONDITION_VERB_ROLLUP[w.conditionTag.toUpperCase()];
    if (rollup) {
      rollup.forEach((verb) => bannedSet.add(verb.toUpperCase()));
    }
  }

  if (armsAvailable < 2) {
    VERB_FAMILIES.BIMANUAL_GRIP.forEach((v) => bannedSet.add(v));
  }
  if (armsAvailable === 0) {
    VERB_FAMILIES.FINE_MOTOR.forEach((v) => bannedSet.add(v));
    bannedSet.add('GRASP');
    bannedSet.add('HOLD');
  }

  if (legsAvailable < 2 || locomotionTier === 'LIMPING') {
    bannedSet.add('SPRINT');
    bannedSet.add('VAULT');
  }
  if (locomotionTier === 'IMMOBILIZED') {
    VERB_FAMILIES.LOCOMOTION_RAPID.forEach((v) => bannedSet.add(v));
    bannedSet.add('WALK');
    bannedSet.add('STAND');
  }

  return {
    entityId,
    armsAvailable,
    legsAvailable,
    locomotionTier,
    openWounds: Array.from(woundTags).sort(),
    bannedKineticVerbs: Array.from(bannedSet).sort(),
  };
};

export const initWounds = (cpy: WoundsModel, bal: WoundsBit, _ste: State): WoundsModel => {
  cpy.woundsByEntity = {};
  cpy.masksByEntity = {};
  if (bal?.slv) bal.slv({ wndBit: { idx: 'init-wounds', val: 0 } });
  return cpy;
};

export const updateWounds = (cpy: WoundsModel, bal: WoundsBit, _ste: State): WoundsModel => {
  if (bal?.slv) bal.slv({ wndBit: { idx: 'update-wounds' } });
  return cpy;
};

export const addWound = (
  cpy: WoundsModel,
  bal: WoundsBit<{ entityId: string; wound: WoundRecord }>,
  _ste: State
): WoundsModel => {
  const payload = bal?.dat;
  if (!payload?.entityId || !payload?.wound) {
    if (bal?.slv) bal.slv({ wndBit: { idx: 'add-wound-err', src: 'MISSING_PAYLOAD', val: -1 } });
    return cpy;
  }

  if (!cpy.woundsByEntity[payload.entityId]) {
    cpy.woundsByEntity[payload.entityId] = [];
  }

  cpy.woundsByEntity[payload.entityId].push(payload.wound);

  if (bal?.slv) {
    bal.slv({
      wndBit: {
        idx: 'add-wound',
        src: payload.entityId,
        val: cpy.woundsByEntity[payload.entityId].length,
        dat: payload.wound,
      },
    });
  }
  return cpy;
};

export const computeSomaticMaskWounds = (
  cpy: WoundsModel,
  bal: WoundsBit<{
    entityId: string;
    armsAvailable: number;
    legsAvailable: number;
    locomotionTier: LocomotionTier;
  }>,
  _ste: State
): WoundsModel => {
  const payload = bal?.dat;
  if (!payload?.entityId) {
    if (bal?.slv) bal.slv({ wndBit: { idx: 'compute-somatic-mask-err', src: 'MISSING_ENTITY_ID', val: -1 } });
    return cpy;
  }

  const wounds = cpy.woundsByEntity[payload.entityId] || [];
  const mask = compileSomaticMask(
    payload.entityId,
    wounds,
    payload.armsAvailable ?? 2,
    payload.legsAvailable ?? 2,
    payload.locomotionTier ?? 'UNRESTRICTED'
  );

  cpy.masksByEntity[payload.entityId] = mask;

  if (bal?.slv) {
    bal.slv({
      wndBit: {
        idx: 'compute-somatic-mask-wounds',
        src: payload.entityId,
        val: mask.bannedKineticVerbs.length,
        dat: mask,
      },
    });
  }
  return cpy;
};

export const assertKineticVerbsWounds = async (
  cpy: WoundsModel,
  bal: WoundsBit<KineticAuditQuery>,
  ste: State
): Promise<WoundsModel> => {
  const payload = bal?.dat;
  if (!payload?.entityId || !payload?.candidateVerbs) {
    if (bal?.slv) bal.slv({ wndBit: { idx: 'assert-kinetic-verbs-err', src: 'MISSING_QUERY', val: -1 } });
    return cpy;
  }

  const mask = cpy.masksByEntity[payload.entityId] || {
    entityId: payload.entityId,
    armsAvailable: 2,
    legsAvailable: 2,
    locomotionTier: 'UNRESTRICTED',
    openWounds: [],
    bannedKineticVerbs: [],
  };

  const bannedSet = new Set(mask.bannedKineticVerbs);
  const breachedVerbs: string[] = [];

  // O(1) set-intersection check per candidate verb
  for (const verb of payload.candidateVerbs) {
    const normalized = verb.trim().toUpperCase();
    if (bannedSet.has(normalized)) {
      breachedVerbs.push(normalized);
    }
  }

  const allowed = breachedVerbs.length === 0;
  const receipt: KineticAuditReceipt = {
    allowed,
    entityId: payload.entityId,
    breachedVerbs: breachedVerbs.sort(),
    bannedKineticVerbs: mask.bannedKineticVerbs,
    timestampIso: new Date().toISOString(),
  };

  // Pure 7-bit ASCII logging to cns00 if violations occur
  if (!allowed && ste && typeof ste.hunt === 'function') {
    const errorMsg = `>> [SOMATIC_CAPABILITY_BREACH] Actor ${payload.entityId} attempted banned verb(s): [${breachedVerbs.join(', ')}]`;
    await ste.hunt('[Console action] Update Console', {
      idx: 'cns00',
      src: errorMsg,
    }).catch(() => {});
  }

  if (bal?.slv) {
    bal.slv({
      wndBit: {
        idx: allowed ? 'assert-kinetic-verbs-ok' : 'assert-kinetic-verbs-breached',
        src: payload.entityId,
        val: breachedVerbs.length,
        dat: receipt,
      },
    });
  }

  return cpy;
};
