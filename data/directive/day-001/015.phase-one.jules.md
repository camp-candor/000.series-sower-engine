import { SomaticState, SomaticBudgetReceipt } from './types';

export const BASE_MOVEMENT_UNIT_COST = 100; // Integer cost baseline

/**
 * Derives the banned kinetic verbs and calculates the strict integer movement
 * budget cost scalar from an actor's somatic state vector.
 */
export function deriveSomaticBudget(somatic: SomaticState): SomaticBudgetReceipt {
  const banned = new Set<string>();

  // 1. Arm Capability Analysis
  if (somatic.availableArms < 2) {
    banned.add('WIELD_TWO_HANDED');
    banned.add('CLIMB_ROPE');
    banned.add('DUAL_WIELD');
    banned.add('VAULT_OBSTACLE');
  }

  if (somatic.availableArms < 1) {
    banned.add('WIELD_ONE_HANDED');
    banned.add('PULL_LEVER');
    banned.add('GRASP_OBJECT');
    banned.add('PUNCH');
  }

  // 2. Locomotion & Leg Capability Analysis
  if (somatic.availableLegs < 2 || somatic.locomotionTier === 'LIMP') {
    banned.add('SPRINT');
    banned.add('LEAP');
  }

  if (somatic.locomotionTier === 'NEUROPATHY_SHUFFLE') {
    banned.add('SPRINT');
    banned.add('LEAP');
    banned.add('FAST_RETREAT');
  }

  if (somatic.availableLegs < 1 || somatic.locomotionTier === 'IMMOBILIZED') {
    banned.add('WALK');
    banned.add('SPRINT');
    banned.add('LEAP');
    banned.add('TACTICAL_REPOSITION');
  }

  // 3. Trauma Overlays
  for (const condition of somatic.activeTrauma) {
    if (condition === 'sprained_wrist_right' || condition === 'dislocated_thumb') {
      banned.add('WIELD_TWO_HANDED');
      banned.add('HEAVY_DRAW');
    }
    if (condition === 'brachial_plexus_severed' || condition === 'paralyzed_left_arm') {
      banned.add('WIELD_TWO_HANDED');
      banned.add('CLIMB_ROPE');
    }
    if (condition === 'concussed_severe') {
      banned.add('AIMED_RANGED_ATTACK');
      banned.add('RAPID_PARRY');
    }
  }

  // 4. Exhaustion & Pain Penalties (Integer Arithmetic in Bps)
  if (somatic.fatigueBps >= 7500 || somatic.painBps >= 8000) {
    banned.add('SPRINT');
    banned.add('HEAVY_ATTACK');
  }

  // Calculate integer movement scalar based on locomotion and fatigue
  // Formula: base * (10000 + fatigueBps) / 10000 * tier_multiplier
  let tierMultiplierBps = 10000;
  if (somatic.locomotionTier === 'LIMP') {
    tierMultiplierBps = 15000; // 1.5x cost
  } else if (somatic.locomotionTier === 'NEUROPATHY_SHUFFLE') {
    tierMultiplierBps = 25000; // 2.5x cost
  } else if (somatic.locomotionTier === 'IMMOBILIZED') {
    tierMultiplierBps = 100000; // prohibitively expensive
  }

  // Integer scale calculation with round-half-even behavior
  const fatigueScaled = Math.floor((BASE_MOVEMENT_UNIT_COST * (10000 + somatic.fatigueBps)) / 10000);
  const movementBudgetCostUnits = Math.floor((fatigueScaled * tierMultiplierBps) / 10000);

  const bannedArray = Array.from(banned).sort();

  return {
    entityId: somatic.entityId,
    movementBudgetCostUnits,
    bannedKineticVerbs: bannedArray,
    canActuateTwoHandedVerbs: somatic.availableArms >= 2 && !banned.has('WIELD_TWO_HANDED'),
    canSprint: !banned.has('SPRINT'),
  };
}

/**
 * Validates the Hand Capacity Invariant: sum(item.occupiesHands) <= character.availableArms
 */
export function validateHandCapacity(
  availableArms: number,
  heldItems: readonly { occupiesHands: number }[]
): { valid: boolean; totalOccupiedHands: number } {
  const totalOccupiedHands = heldItems.reduce((acc, item) => acc + item.occupiesHands, 0);
  return {
    valid: totalOccupiedHands <= availableArms,
    totalOccupiedHands,
  };
}

export class FoundationalLedgerPort {
  constructor(private readonly db: DatabaseClient) {}

  public async appendEvent(event: Omit<StoredLedgerEvent, 'sequenceId'>): Promise<void> {
    const stmt = this.db.prepare(`
      INSERT INTO event_log (
        event_id, stream_id, tick, event_type, actor_id, target_id,
        seed, payload_json, prev_event_hash, event_hash, created_at_ms
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).bind(
      event.eventId,
      event.streamId,
      event.tick,
      event.eventType,
      event.actorId,
      event.targetId,
      event.seed,
      event.payloadJson,
      event.prevEventHash,
      event.eventHash,
      event.createdAtMs
    );

    await stmt.run();
  }

  public async updateSomaticState(somatic: SomaticState): Promise<void> {
    const budget = deriveSomaticBudget(somatic);
    const stmt = this.db.prepare(`
      INSERT INTO vector_somatic (
        entity_id, available_arms, available_legs, locomotion_tier,
        fatigue_bps, pain_bps, refusal_strain_bps, active_trauma_json,
        banned_kinetic_verbs_json, updated_at_ms
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(entity_id) DO UPDATE SET
        available_arms = excluded.available_arms,
        available_legs = excluded.available_legs,
        locomotion_tier = excluded.locomotion_tier,
        fatigue_bps = excluded.fatigue_bps,
        pain_bps = excluded.pain_bps,
        refusal_strain_bps = excluded.refusal_strain_bps,
        active_trauma_json = excluded.active_trauma_json,
        banned_kinetic_verbs_json = excluded.banned_kinetic_verbs_json,
        updated_at_ms = excluded.updated_at_ms
    `).bind(
      somatic.entityId,
      somatic.availableArms,
      somatic.availableLegs,
      somatic.locomotionTier,
      somatic.fatigueBps,
      somatic.painBps,
      somatic.refusalStrainBps,
      JSON.stringify(somatic.activeTrauma),
      JSON.stringify(budget.bannedKineticVerbs),
      somatic.updatedAtMs
    );

    await stmt.run();
  }

  public async equipItem(
    entityId: string,
    item: LogisticalItem
  ): Promise<{ success: boolean; error?: string }> {
    // 1. Fetch current somatic limb budget
    const somaticStmt = this.db.prepare(`
      SELECT available_arms FROM vector_somatic WHERE entity_id = ?
    `).bind(entityId);
    const somaticRes = await somaticStmt.all<{ available_arms: number }>();

    if (somaticRes.results.length === 0) {
      return { success: false, error: 'ERR_ENTITY_SOMATIC_NOT_FOUND' };
    }

    const availableArms = somaticRes.results[0].available_arms;

    // 2. Fetch currently held items
    const heldStmt = this.db.prepare(`
      SELECT occupies_hands FROM vector_logistical
      WHERE owner_entity_id = ? AND slot = 'HELD' AND item_id != ?
    `).bind(entityId, item.itemId);
    const heldRes = await heldStmt.all<{ occupies_hands: number }>();

    const candidateHeld = [...heldRes.results, { occupies_hands: item.occupiesHands }];
    const capacityCheck = validateHandCapacity(availableArms, candidateHeld);

    if (!capacityCheck.valid) {
      return {
        success: false,
        error: `ERR_HAND_CAPACITY_EXCEEDED: required ${capacityCheck.totalOccupiedHands}, available ${availableArms}`,
      };
    }

    // 3. Persist item placement
    const equipStmt = this.db.prepare(`
      INSERT INTO vector_logistical (
        item_id, owner_entity_id, location_id, slot, occupies_hands,
        mass_grams, durability_bps, is_contested, updated_at_ms
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
      ON CONFLICT(item_id) DO UPDATE SET
        owner_entity_id = excluded.owner_entity_id,
        location_id = excluded.location_id,
        slot = excluded.slot,
        occupies_hands = excluded.occupies_hands,
        mass_grams = excluded.mass_grams,
        durability_bps = excluded.durability_bps,
        is_contested = excluded.is_contested,
        updated_at_ms = excluded.updated_at_ms
    `).bind(
      item.itemId,
      entityId,
      item.locationId,
      'HELD',
      item.occupiesHands,
      item.massGrams,
      item.durabilityBps,
      item.isContested ? 1 : 0,
      item.updatedAtMs
    );

    await equipStmt.run();
    return { success: true };
  }
}

    const twoHandedItem = [{ occupiesHands: 2 }];

    expect(check.valid).toBe(false);
  });
