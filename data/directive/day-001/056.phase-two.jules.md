```typescript
import type Ledger from './fce/ledger.interface';
import type { LedgerWorkingState } from '../16.state.unit/fce/state.interface';
import type { EventEnvelope, PreparedBatchStatement } from './fce/ledger.interface';
import { reduceLedgerState } from './buz/ledger.buzz';
import { canonicalizeStateJson } from '../16.state.unit/buz/state.buzz';
import { createHash } from 'node:crypto';

export class StagingAccumulator {
  private baseState: LedgerWorkingState;
  private volatileWorkingState: LedgerWorkingState;
  private pendingEvents: EventEnvelope[] = [];
  private lastHash: string;

  constructor(initialState: LedgerWorkingState) {
    this.baseState = initialState;
    this.volatileWorkingState = initialState;
    this.lastHash = initialState.stateHash;
  }

  public get workingState(): LedgerWorkingState {
    return this.volatileWorkingState;
  }

  public get stagedCount(): number {
    return this.pendingEvents.length;
  }

  public get stagedEnvelopes(): readonly EventEnvelope[] {
    return this.pendingEvents;
  }

  public stageEvent<T>(
    type: string,
    payload: T,
    meta: {
      eventId: string;
      streamId?: string;
      tick?: number;
      actorId?: string | null;
      targetId?: string | null;
      seed?: string;
      timestamp_ms?: number;
    }
  ): EventEnvelope<T> {
    const nextSeq = this.volatileWorkingState.lastSequenceNumber + 1;
    const streamId = meta.streamId || 'main';
    const tick = meta.tick ?? this.volatileWorkingState.tick;
    const seed = meta.seed || 'seed_stage';
    const timestampMs = meta.timestamp_ms ?? 0;

    const unsigned = {
      sequenceNumber: nextSeq,
      eventId: meta.eventId,
      streamId,
      tick,
      type,
      version: 1,
      actorId: meta.actorId ?? null,
      targetId: meta.targetId ?? null,
      seed,
      payload,
      prevEventHash: this.lastHash,
      timestamp_ms: timestampMs,
    };

    const envelopeHash = createHash('sha256')
      .update(canonicalizeStateJson(unsigned), 'utf8')
      .digest('hex');

    const sealedEnvelope: EventEnvelope<T> = {
      ...unsigned,
      stateHash: envelopeHash,
    };

    // Applies transition in volatile memory immediately; throws on invariant breach
    this.volatileWorkingState = reduceLedgerState(this.volatileWorkingState, sealedEnvelope as any, seed);
    this.lastHash = this.volatileWorkingState.stateHash;
    this.pendingEvents.push(sealedEnvelope as EventEnvelope);
    return sealedEnvelope;
  }

  public rollback(): void {
    this.volatileWorkingState = this.baseState;
    this.pendingEvents = [];
    this.lastHash = this.baseState.stateHash;
  }

  public commit(): {
    readonly committedEvents: readonly EventEnvelope[];
    readonly newState: LedgerWorkingState;
  } {
    const events = [...this.pendingEvents];
    this.baseState = this.volatileWorkingState;
    this.pendingEvents = [];
    return {
      committedEvents: events,
      newState: this.baseState,
    };
  }

  public compileBatchStatements(events: readonly EventEnvelope[]): PreparedBatchStatement[] {
    const stmts: PreparedBatchStatement[] = [];

    // 1. Append staged envelopes to event_log
    for (const ev of events) {
      stmts.push({
        sql: `INSERT INTO event_log (
          event_id, stream_id, tick, type, version,
          actor_id, target_id, seed, payload_json, prev_hash,
          state_hash, timestamp_ms
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
        params: [
          ev.eventId,
          ev.streamId,
          ev.tick,
          ev.type,
          ev.version,
          ev.actorId,
          ev.targetId,
          ev.seed,
          JSON.stringify(ev.payload),
          ev.prevEventHash,
          ev.stateHash,
          ev.timestamp_ms,
        ],
      });
    }

    // 2. Materialize modified entities_live
    for (const ent of Object.values(this.baseState.entities)) {
      stmts.push({
        sql: `INSERT INTO entities_live (
          entity_id, location_id, q, r, arms_available, legs_available,
          locomotion_tier, concussion_level, fatigue_bps, pain_bps, refusal_strain_bps,
          open_wounds_json, banned_kinetic_verbs_json, held_items_json, worn_gear_json,
          slung_items_json, cached_items_json, updated_at_ms
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT(entity_id) DO UPDATE SET
          location_id = excluded.location_id,
          q = excluded.q,
          r = excluded.r,
          arms_available = excluded.arms_available,
          legs_available = excluded.legs_available,
          locomotion_tier = excluded.locomotion_tier,
          concussion_level = excluded.concussion_level,
          fatigue_bps = excluded.fatigue_bps,
          pain_bps = excluded.pain_bps,
          refusal_strain_bps = excluded.refusal_strain_bps,
          open_wounds_json = excluded.open_wounds_json,
          banned_kinetic_verbs_json = excluded.banned_kinetic_verbs_json,
          held_items_json = excluded.held_items_json,
          worn_gear_json = excluded.worn_gear_json,
          slung_items_json = excluded.slung_items_json,
          cached_items_json = excluded.cached_items_json,
          updated_at_ms = excluded.updated_at_ms`,
        params: [
          ent.entityId,
          ent.locationId,
          ent.q,
          ent.r,
          ent.armsAvailable,
          ent.legsAvailable,
          ent.locomotionTier,

          ent.fatigueBps,
          ent.painBps,
          ent.refusalStrainBps,
          JSON.stringify(ent.openWounds),
          JSON.stringify(ent.bannedKineticVerbs),
          JSON.stringify(ent.heldItems),
          JSON.stringify(ent.wornGear),
          JSON.stringify(ent.slungItems),
          JSON.stringify(ent.cachedItems),
          ent.updatedAtMs,
        ],
      });
    }

    // 3. Materialize strings_ledger
    for (const str of Object.values(this.baseState.strings)) {
      stmts.push({
        sql: `INSERT INTO strings_ledger (
          relation_id, holder_id, debtor_id, weight_bps, strings_count, updated_at_ms
        ) VALUES (?, ?, ?, ?, ?, ?)
        ON CONFLICT(relation_id) DO UPDATE SET
          weight_bps = excluded.weight_bps,
          strings_count = excluded.strings_count,
          updated_at_ms = excluded.updated_at_ms`,
        params: [
          str.relation_id,
          str.holder_id,
          str.debtor_id,
          str.weight_bps,
          str.strings_count,
          str.updated_at_ms,
        ],
      });
    }

          grievance_id, instigator_id, victim_id, escalation_depth,

          grv.grievance_id,
          grv.instigator_id,
          grv.victim_id,
          grv.escalation_depth,
          grv.cooldown_cycles,
          grv.intensity_bps,
          grv.status,
          grv.updated_at_ms,
        ],
      });
    }

    // 5. Materialize paranoia_clocks
    for (const clk of Object.values(this.baseState.clocks)) {
      stmts.push({
        sql: `INSERT INTO paranoia_clocks (
          clock_id, entity_id, clock_name, value_bps, threshold_bps,

          clk.clock_name,
          clk.value_bps,
          clk.threshold_bps,
          clk.decay_rate_bps,
          clk.is_triggered,
          clk.updated_at_ms,
        ],
      });
    }

    return stmts;
  }
}
```

```typescript
import type { LedgerModel } from '../ledger.model';
import type LedgerBit from '../fce/ledger.bit';
import type State from '../../99.core/state';
import type { LedgerWorkingState } from '../../16.state.unit/fce/state.interface';
import type {
  EventEnvelope,
  StageEventPayload,
  ReduceEventPayload,
  ReplayLedgerPayload,
  CommitStagedReceipt,
} from '../fce/ledger.interface';
import { StagingAccumulator } from '../ledger.model';
import {
  cloneMutableStateMaps,
  computeWorkingStateHash,
  recursiveDeepFreeze,
} from '../../16.state.unit/buz/state.buzz';
import { validateHandCapacity } from '../../07.inventory.unit/buz/inventory.buzz';

export function clampBps(val: number): number {
  if (val <= 0) return 0;
  if (val >= 10000) return 10000;
  return Math.floor(val);
}

export function reduceLedgerState(
  state: LedgerWorkingState,
  event: EventEnvelope,
  seed: string
): LedgerWorkingState {
  // 1. Optimistic Concurrency Control (OCC) Gate
  const expectedSeq = state.lastSequenceNumber + 1;
  if (event.sequenceNumber !== expectedSeq) {
    throw new Error(
      `[OCC_SEQUENCE_CONFLICT] Stale sequence token: expected sequence #${expectedSeq}, got #${event.sequenceNumber}`
    );
  }

  // 2. Clone mutable domain dictionaries from frozen state
  const nextMaps = cloneMutableStateMaps(state);

  // 3. Dispatch pure domain state mutations
  switch (event.type) {
    case 'SOMATIC_TRAUMA_APPLIED': {
      const p = event.payload;
      const entity = nextMaps.entities[p.entity_id];
      if (entity) {
        if (p.fatigue_bps_delta) entity.fatigueBps = clampBps(entity.fatigueBps + p.fatigue_bps_delta);
        if (p.pain_bps_delta) entity.painBps = clampBps(entity.painBps + p.pain_bps_delta);
        if (p.refusal_strain_bps_delta) entity.refusalStrainBps = clampBps(entity.refusalStrainBps + p.refusal_strain_bps_delta);
        if (p.added_wounds && Array.isArray(p.added_wounds)) {
          entity.openWounds = Array.from(new Set([...entity.openWounds, ...p.added_wounds])).sort();
        }
        if (p.banned_verbs && Array.isArray(p.banned_verbs)) {
          entity.bannedKineticVerbs = Array.from(new Set([...entity.bannedKineticVerbs, ...p.banned_verbs])).sort();
        }
        entity.updatedAtMs = event.timestamp_ms;
      }
      break;
    }

    case 'INVENTORY_TRANSFERRED': {
      const p = event.payload;
      const entity = nextMaps.entities[p.entity_id];
      if (entity && p.held_items) {
        if (!validateHandCapacity(entity.armsAvailable, p.held_items.length)) {
          throw new Error(
            `[HAND_CAPACITY_BREACH] Entity ${p.entity_id} holds ${p.held_items.length} items with only ${entity.armsAvailable} arm(s)`
          );
        }
        entity.heldItems = [...p.held_items];
        entity.updatedAtMs = event.timestamp_ms;
      }
      break;
    }

    case 'SPATIAL_DISPLACEMENT_COMMITTED': {
      const p = event.payload;
      const entity = nextMaps.entities[p.entity_id];
      if (entity) {
        if (p.location_id) entity.locationId = p.location_id;
        if (p.q !== undefined) entity.q = p.q;
        if (p.r !== undefined) entity.r = p.r;
        entity.updatedAtMs = event.timestamp_ms;
      }
      break;
    }

    case 'SOCIAL_STRING_EXCHANGED': {
      const p = event.payload;
      if (p.holder_id && p.debtor_id) {
        if (p.holder_id.trim() === p.debtor_id.trim()) {
          throw new Error(
            `[ANTI_REFLEXIVITY_BREACH] Actor ${p.holder_id} cannot hold or spend leverage strings over self`
          );
        }
        const relId = `${p.holder_id.trim()}->${p.debtor_id.trim()}`;
        const existing = nextMaps.strings[relId] || {
          relation_id: relId,
          holder_id: p.holder_id.trim(),
          debtor_id: p.debtor_id.trim(),
          weight_bps: p.weight_bps ?? 1000,
          strings_count: 0,
          updated_at_ms: event.timestamp_ms,
        };

        const delta = p.strings_count_delta ?? p.strings_count ?? 1;
        const nextCount = Math.max(0, existing.strings_count + delta);
        nextMaps.strings[relId] = {
          ...existing,
          strings_count: nextCount,
          weight_bps: p.weight_bps ?? existing.weight_bps,
          updated_at_ms: event.timestamp_ms,
        };
      }
      break;
    }

    case 'GRIEVANCE_ESCALATED': {
      const p = event.payload;
      const grv = nextMaps.grievances[p.grievance_id];
      if (grv) {
        grv.escalation_depth += 1;
        if (p.intensity_delta_bps) {
          grv.intensity_bps = clampBps(grv.intensity_bps + p.intensity_delta_bps);
        }
        if (grv.intensity_bps >= 8000) grv.status = 'SIMMERING';
        grv.updated_at_ms = event.timestamp_ms;
      }
      break;
    }

    case 'PARANOIA_CLOCK_TICKED': {
      const p = event.payload;
      const clk = nextMaps.clocks[p.clock_id];
      if (clk) {
        const delta = p.delta_bps ?? 1000;
        clk.value_bps = clampBps(clk.value_bps + delta);
        clk.is_triggered = clk.value_bps >= clk.threshold_bps ? 1 : 0;
        clk.updated_at_ms = event.timestamp_ms;
      }
      break;
    }

    case 'SIMULATION_TICK_COMMITTED': {
      // Passive thermodynamic decay on discrete tick
      for (const clkId in nextMaps.clocks) {
        const clk = nextMaps.clocks[clkId];
        clk.value_bps = Math.max(0, clk.value_bps - clk.decay_rate_bps);
        clk.is_triggered = clk.value_bps >= clk.threshold_bps ? 1 : 0;
      }
      for (const grvId in nextMaps.grievances) {
        const grv = nextMaps.grievances[grvId];
        if (grv.cooldown_cycles > 0) grv.cooldown_cycles -= 1;
      }
      break;
    }

    default:
      break;
  }

  // 4. Compute next deterministic Merkle state hash
  const nextHash = computeWorkingStateHash(nextMaps);

  // 5. Seal output snapshot
  return recursiveDeepFreeze({
    reducerVersion: state.reducerVersion,
    seriesSlug: state.seriesSlug,
    tick: Math.max(state.tick, event.tick),
    lastSequenceNumber: event.sequenceNumber,
    lastEventHash: event.stateHash,
    entities: nextMaps.entities,
    strings: nextMaps.strings,
    grievances: nextMaps.grievances,
    clocks: nextMaps.clocks,
    beliefs: nextMaps.beliefs,
    unratifiedDetails: nextMaps.unratifiedDetails,
    stateHash: nextHash,
  }) as LedgerWorkingState;
}

export function foldLedgerEvents(
  initialState: LedgerWorkingState,
  events: readonly EventEnvelope[],
  seed: string
): LedgerWorkingState {
  let activeState = initialState;
  for (const event of events) {
    activeState = reduceLedgerState(activeState, event, seed);
  }
  return activeState;
}

export const stageEventLedger = async (
  cpy: LedgerModel,
  bal: LedgerBit<StageEventPayload>,
  ste: State
): Promise<LedgerModel> => {
  try {
    const payload = bal?.dat;
    if (!payload?.type) {
      throw new Error('[MALFORMED_INPUT] event type is required');
    }

    // Lazily initialize accumulator from working state if absent
    if (!cpy.activeAccumulator) {
      let workingState: any = null;
      if (ste?.hunt) {
        const res: any = await ste.hunt('[State action] Get State', {});
        workingState = res?.staBit?.dat;
      }
      if (!workingState) {
        throw new Error('[WORKING_STATE_NOT_FOUND] Cannot stage event without active workingState');
      }
      cpy.activeAccumulator = new StagingAccumulator(workingState);
    }

    const envelope = cpy.activeAccumulator.stageEvent(
      payload.type,
      payload.payload,
      payload.meta || { eventId: `evt_${Date.now()}` }
    );

    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [STAGE:EVENT] [OK] Staged '${payload.type}' (#${envelope.sequenceNumber}) in volatile RAM.`,
      }).catch(() => {});
    }

    if (bal?.slv) {
      bal.slv({
        ldgBit: {
          idx: 'stage-event-ledger-ok',
          val: cpy.activeAccumulator.stagedCount,
          dat: envelope as any,
        },
      });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [STAGE:FAIL] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ ldgBit: { idx: 'stage-event-ledger-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const rollbackStagedLedger = async (

  ste: State
): Promise<LedgerModel> => {
  if (cpy.activeAccumulator) {
    cpy.activeAccumulator.rollback();
  }

  if (ste?.hunt) {
    await ste.hunt('[Console action] Update Console', {
      idx: 'cns00',
      src: '>> [STAGE:ROLLBACK] [OK] Volatile staging buffer discarded. Zero SQLite rows touched.',
    }).catch(() => {});
  }

  if (bal?.slv) {
    bal.slv({ ldgBit: { idx: 'rollback-staged-ledger-ok', val: 0 } });
  }
  return cpy;
};

export const commitStagedLedger = async (
  cpy: LedgerModel,
  bal: LedgerBit,
  ste: State
): Promise<LedgerModel> => {
  try {
    if (!cpy.activeAccumulator || cpy.activeAccumulator.stagedCount === 0) {
      throw new Error('[NO_STAGED_EVENTS] Staging accumulator has zero pending events to commit');
    }

    const { committedEvents, newState } = cpy.activeAccumulator.commit();
    const batchStatements = cpy.activeAccumulator.compileBatchStatements(committedEvents);

    // Atomically execute multi-table batch in local SQLite
    if (ste?.hunt) {
      await ste.hunt('[Database action] Batch Database', {
        lst: batchStatements,
      }).catch(() => {});

      // Freeze and set new working state in 16.state.unit
      await ste.hunt('[State action] Freeze State', {
        dat: { target: newState },
      }).catch(() => {});

      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `>> [STAGE:COMMIT] [OK] Flashed ${committedEvents.length} event(s) atomically to SQLite. StateHash: ${newState.stateHash.slice(0, 16)}...`,
      }).catch(() => {});
    }

    const receipt: CommitStagedReceipt = {
      committedCount: committedEvents.length,
      newStateHash: newState.stateHash,
      lastSequenceNumber: newState.lastSequenceNumber,
      statementsCount: batchStatements.length,
    };

    if (bal?.slv) {
      bal.slv({
        ldgBit: {
          idx: 'commit-staged-ledger-ok',
          val: receipt.committedCount,
          dat: receipt as any,
        },
      });
    }
  } catch (err: any) {
    if (ste?.hunt) {
      await ste.hunt('[Console action] Update Console', {
        idx: 'cns00',
        src: `:: [STAGE:COMMIT:FAIL] ${err.message}`,
      }).catch(() => {});
    }
    if (bal?.slv) {
      bal.slv({ ldgBit: { idx: 'commit-staged-ledger-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const reduceLedger = (
  cpy: LedgerModel,
  bal: LedgerBit<ReduceEventPayload>,
  _ste: State
): LedgerModel => {
  try {
    const payload = bal?.dat;
    if (!payload?.event) {
      throw new Error('[MALFORMED_INPUT] event envelope required');
    }

    // Working state must be passed or accessed via model
    const currentState = cpy.activeAccumulator?.workingState;
    if (!currentState) {
      throw new Error('[WORKING_STATE_NOT_FOUND] No active state available for reduction');
    }

    const nextState = reduceLedgerState(currentState, payload.event, payload.seed || 'seed_reduce');

    if (bal?.slv) {
      bal.slv({
        ldgBit: {
          idx: 'reduce-ledger-ok',
          src: nextState.stateHash,
          val: nextState.lastSequenceNumber,
          dat: nextState as any,
        },
      });
    }
  } catch (err: any) {
    if (bal?.slv) {
      bal.slv({ ldgBit: { idx: 'reduce-ledger-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};

export const replayLedger = (
  cpy: LedgerModel,
  bal: LedgerBit<ReplayLedgerPayload>,
  _ste: State
): LedgerModel => {
  try {
    const payload = bal?.dat;
    if (!payload?.initialState || !payload?.events) {
      throw new Error('[MALFORMED_INPUT] initialState and events array required for replay');
    }

    const replayed = foldLedgerEvents(payload.initialState, payload.events, payload.seed || 'seed_replay');

    if (bal?.slv) {
      bal.slv({
        ldgBit: {
          idx: 'replay-ledger-ok',
          src: replayed.stateHash,
          val: replayed.lastSequenceNumber,
          dat: replayed as any,
        },
      });
    }
  } catch (err: any) {
    if (bal?.slv) {
      bal.slv({ ldgBit: { idx: 'replay-ledger-error', src: err.message, val: -1 } });
    }
  }

  return cpy;
};
```

```typescript
import clone from 'clone-deep';
import * as Act from './ledger.action';
import { LedgerModel } from './ledger.model';
import * as Buzz from './ledger.buzzer';
import type State from '../99.core/state';

export function reducer(
  model: LedgerModel = new LedgerModel(),
  act: Act.Actions,
  state?: State
): LedgerModel {
  switch (act.type) {
    case Act.INIT_LEDGER:
      return Buzz.initLedger(clone(model, true), act.bale, state!);

    case Act.UPDATE_LEDGER:
      return Buzz.updateLedger(clone(model, true), act.bale, state!);

    case Act.STAGE_EVENT_LEDGER:
      Buzz.stageEventLedger(clone(model, true), act.bale, state!);
      return model;

    case Act.COMMIT_STAGED_LEDGER:
      Buzz.commitStagedLedger(clone(model, true), act.bale, state!);
      return model;

    case Act.ROLLBACK_STAGED_LEDGER:
      Buzz.rollbackStagedLedger(clone(model, true), act.bale, state!);
      return model;

    case Act.REDUCE_LEDGER:
      return Buzz.reduceLedger(clone(model, true), act.bale, state!);

    case Act.REPLAY_LEDGER:
      return Buzz.replayLedger(clone(model, true), act.bale, state!);

    default:
      return model;
  }
}
```
