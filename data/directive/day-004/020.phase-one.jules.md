export const computeHorizonProgressBps = (tick: number, totalTicks: number): number => {
    if (!Number.isInteger(tick) || !Number.isInteger(totalTicks) || totalTicks <= 0 || tick < 0) {
        throw new RangeError('[INVALID_HORIZON_PROGRESS] tick and positive totalTicks must be integers')
    }
    const clampedTick = Math.min(tick, totalTicks)
    return Number((BigInt(clampedTick) * 10000n) / BigInt(totalTicks))
}

export const computeSolarDayProgressBps = (
    elapsedSeconds: number,
    dayDurationSeconds = 86400,
): number => {
    if (!Number.isInteger(elapsedSeconds) || elapsedSeconds < 0) {
        throw new RangeError('[INVALID_SOLAR_SECONDS] elapsedSeconds must be non-negative integer')
    }
    const wrapped = elapsedSeconds % dayDurationSeconds
    return Number((BigInt(wrapped) * 10000n) / BigInt(dayDurationSeconds))
}
