* `ROOT MENU` -> Resolves root slv to return to main library switchboard.




3. **Expressive Telemetry & Real-Time Feedback:**

const emitConsole = async (ste: State | undefined, src: string): Promise<void> => {

  if (lib && typeof lib.hunt === 'function') {
    try {

    } catch {
      // Fall through to standard out in headless environments
    }
  }
  if (ste && typeof ste.hunt === 'function') {

      // Degrade gracefully

const resolveEngineRoot = (): string => {
  let curr = process.cwd();
  while (curr && curr !== path.dirname(curr)) {
    if (
      fs.existsSync(path.join(curr, 'compiled')) ||
      fs.existsSync(path.join(curr, 'packages', '300.engine'))
    ) {
      return curr;
    }
    curr = path.dirname(curr);
  }
  return process.cwd();
};

const scanCompiledWorlds = (): string[] => {
  try {
    const engineRoot = resolveEngineRoot();
    const compiledDir = path.resolve(engineRoot, 'compiled');
    if (!fs.existsSync(compiledDir)) return [];

    return fs
      .readdirSync(compiledDir, { withFileTypes: true })
      .filter(
        (dirent) =>
          dirent.isDirectory() &&
          !dirent.name.startsWith('.') &&
          !dirent.name.startsWith('_') &&
          fs.existsSync(path.join(compiledDir, dirent.name, 'bible-state.json')),
      )
      .map((dirent) => dirent.name);
  } catch {
    return [];
  }
};

const deepFreeze = <T>(obj: T): T => {
  if (obj === null || typeof obj !== 'object') return obj;
  Object.freeze(obj);
  for (const key of Object.getOwnPropertyNames(obj)) {
    const val = (obj as any)[key];
    if (val !== null && typeof val === 'object' && !Object.isFrozen(val)) {
      deepFreeze(val);
    }
  }
  return obj;
};

export const updateMenu = async (
  cpy: MenuModel,
  bal?: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  const lib = (globalThis as any).LIBRARY || (global as any).LIBRARY;
  const isHeadless = !lib || typeof lib.hunt !== 'function';

  // Branch menu options based on engine flightDeckMode
  const isRuntime = cpy.flightDeckMode === 'RUNTIME';

  const bootList = [
    '[01] BOOT ENGINE & ENSURE SCHEMA',
    '[02] STATUS ENGINE & TABLE',
    '[03] AUDIT SIMULATION INVARIANTS',
    '[04] RESET ENGINE TO OFFLINE',
    '[ROOT MENU]',
  ];

  const runtimeList = [
    '[14] EXECUTE DETERMINISTIC',
    'ADVANCE TICK',
    'RUN SIM GAUNTLET',
    '[02] STATUS ENGINE & TABLE',
    '[03] AUDIT SIMULATION INVARIANTS',
    '[04] RESET ENGINE TO OFFLINE',
    '[ROOT MENU]',
  ];

  const lst = isRuntime ? runtimeList : bootList;

  if (isHeadless) {
    if (bal?.slv) {
      bal.slv({
        mnuBit: {
          idx: 'update-menu-headless',
          val: 1,
          src: isRuntime ? 'RUNTIME_MODE' : 'BOOT_MODE',
        },
      });
    }
    return cpy;
  }

  const bit = await lib.hunt(UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: Math.min(12, Math.max(6, lst.length + 2)),
  });

  const choiceBit = await lib.hunt(OPEN_CHOICE, {
    dat: { clr0: 'black', clr1: isRuntime ? 'cyan' : 'green' },
    src: 'vertical',
    lst,
    net: bit?.grdBit?.dat,
  });

  const picked = choiceBit?.chcBit?.src;
  cpy.activeSelection = picked || '';
  cpy.lastActionTimestamp = Date.now();

  switch (picked) {
    case '[01] BOOT ENGINE & ENSURE SCHEMA': {
      await emitConsole(ste, '>> [AIRLOCK::SCAN] Sweeping compiled/ universe manifests from host volume...');
      const worlds = scanCompiledWorlds();

      if (worlds.length === 0) {
        await emitConsole(ste, ':: [ERROR] No compiled worlds found in compiled/.');
        await emitConsole(ste, ':: Please compile a series in 000.goblin-lore before booting.');
        break;
      }

      let selectedSlug = worlds[0];

      if (worlds.length > 1) {
        const subList = [...worlds, 'CANCEL'];
        const subGrid = await lib.hunt(UPDATE_GRID, {
          x: 0,
          y: 4,
          xSpan: 4,
          ySpan: Math.min(12, Math.max(6, subList.length + 2)),
        });

        const choiceRes = await lib.hunt(OPEN_CHOICE, {
          dat: { clr0: 'black', clr1: 'yellow' },
          src: 'vertical',
          lst: subList,
          net: subGrid?.grdBit?.dat,
        });

        const chosenWorld = choiceRes?.chcBit?.src;
        if (!chosenWorld || chosenWorld === 'CANCEL') {
          await emitConsole(ste, '>> [AIRLOCK::ABORT] Ingestion cancelled by operator.');
          break;
        }
        selectedSlug = chosenWorld;
      }

      await emitConsole(ste, `>> [AIRLOCK::INGEST] Target selected: [${selectedSlug}]`);
      const engineRoot = resolveEngineRoot();
      const manifestPath = path.resolve(engineRoot, 'compiled', selectedSlug, 'bible-state.json');
      const rawManifest = fs.readFileSync(manifestPath, 'utf8');
      const parsedManifest = deepFreeze(JSON.parse(rawManifest));

      await emitConsole(ste, '>> [AIRLOCK::SEAL] In-memory defensive lock applied (Object.freeze)');
      await emitConsole(ste, '>> [AIRLOCK::PURITY] Zero Native I/O Fence verified: 0 disk handles forwarded');

      if (ste && typeof ste.hunt === 'function') {
        const bootRes: any = await ste.hunt(ActEng.BOOT_ENGINE, {
          dat: {
            bootParams: {
              seriesSlug: selectedSlug,
              bibleState: parsedManifest,
              seedGenesis: true,
            },
          },
        });

        // Switch to RUNTIME mode upon successful engine boot
        cpy.flightDeckMode = 'RUNTIME';
        cpy.activeSeriesSlug = selectedSlug;

        await emitConsole(ste, '------------------------------------------------------------');
        await emitConsole(ste, '>> [EPIC-08] AXIOMATIC GENESIS & COMPARTMENTALIZED LEDGERS [ACTIVE]');
        await emitConsole(ste, `>> [RUNTIME:READY] Active Universe: [${selectedSlug}]`);
        await emitConsole(ste, '>> [SWITCHBOARD] Navigation switched to RUNTIME DISPATCH in left deck.');
        await emitConsole(ste, '------------------------------------------------------------');
      }
      break;
    }

    case '[14] EXECUTE DETERMINISTIC': {
      await emitConsole(ste, '>> [EXECUTE] Dispatching Stage 3 Deterministic Encounter Lottery...');
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt(ActEng.STATUS_ENGINE, {});
      }
      await emitConsole(ste, '>> [EXECUTE:OK] Deterministic stage evaluation complete.');
      break;
    }

    case 'ADVANCE TICK': {
      await emitConsole(ste, '>> [TICK] Advancing logical clock (+1 tick) & executing decay sweep...');
      if (ste && typeof ste.hunt === 'function') {
        // Dispatches to 17.tick.unit and 12.paranoia.unit
        await ste.hunt('[Tick action] Advance Tick', { val: 1 });
      }
      await emitConsole(ste, '>> [TICK:OK] Clock advanced. Doom meters decayed.');
      break;
    }

    case 'RUN SIM GAUNTLET': {
      await emitConsole(ste, '>> [GAUNTLET] Launching continuous simulation gauntlet (100 ticks)...');
      if (ste && typeof ste.hunt === 'function') {
        for (let i = 0; i < 5; i++) {
          await ste.hunt('[Tick action] Advance Tick', { val: 20 });
        }
      }
      await emitConsole(ste, '>> [GAUNTLET:COMPLETE] 100 ticks evaluated in host RAM (0 MB VRAM).');
      break;
    }

    case '[02] STATUS ENGINE & TABLE': {
      await emitConsole(ste, '>> [STATUS] Querying 4-Vector table status and row inventories...');
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt(ActEng.STATUS_ENGINE, {});
      }
      break;
    }

    case '[03] AUDIT SIMULATION INVARIANTS': {
      await emitConsole(ste, '>> [AUDIT] Running non-destructive T_0 boundary probe...');
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt(ActEng.AUDIT_ENGINE, {});
      }
      break;
    }

    case '[04] RESET ENGINE TO OFFLINE': {
      await emitConsole(ste, '>> [RESET] Resetting simulation engine to OFFLINE status...');
      if (ste && typeof ste.hunt === 'function') {
        await ste.hunt(ActEng.RESET_ENGINE, {});
      }
      // Revert menu flightDeckMode back to BOOT
      cpy.flightDeckMode = 'BOOT';
      cpy.activeSeriesSlug = '';
      await emitConsole(ste, '>> [RESET:OK] Cockpit returned to BOOT & ENSURE SCHEMA mode.');
      break;
    }

    case '[ROOT MENU]': {
      if (rootSlv != null) {
        rootSlv({ mnuBit: { idx: 'root-menu' } });
      }
      return cpy;
    }
  }

  // Non-recursive re-entrant loop yield
  setTimeout(async () => {
    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt(ActMnu.UPDATE_MENU, {});
    }
  }, 333);

  return cpy;
};

export const switchFlightDeckMode = async (
  cpy: MenuModel,
  bal?: MenuBit,
  ste?: State,
): Promise<MenuModel> => {
  if (bal?.dat?.flightDeckMode) {
    cpy.flightDeckMode = bal.dat.flightDeckMode;
  }
  if (bal?.slv) {
    bal.slv({
      mnuBit: {
        idx: 'switch-flight-deck-mode',
        src: cpy.flightDeckMode,
      },
    });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './menu.action.js';
import { MenuModel } from './menu.model.js';
import * as Buzz from './menu.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: MenuModel = new MenuModel(),
  act: Act.Actions,
  state?: State,
) {
  switch (act.type) {
    case Act.INIT_MENU:
      return Buzz.initMenu(clone(model), act.bale, state);

    case Act.UPDATE_MENU:
      return Buzz.updateMenu(clone(model), act.bale, state);

    case Act.SWITCH_FLIGHT_DECK_MODE:
      return Buzz.switchFlightDeckMode(clone(model), act.bale, state);

    default:
      return model;
  }
}
