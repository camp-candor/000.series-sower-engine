# Task: Implement Phase 5 Discrete Realm Relational Schema Provisioning

## Objective

Implement **Phase 5: Discrete Realm Relational Schema Provisioning** in `packages/300.engine/02.database.unit` of `000.series-sower-engine`. This phase arms the active SQLite realm connection with self-healing DDL bootstrapping for the 7 Four-Vector tables and 3 Continuity tables, enforces the Gillian Arithmetic Doctrine (discrete integer basis points, zero floating-point arithmetic), and installs hard append-only and monotonic version triggers directly on the database handle.

## Technical Specifications & Requirements

1. **Quarantine Invariants (`check:native-io`):**
* Code must reside strictly within `packages/300.engine/02.database.unit/`.


* The directory `packages/300.engine/src/` is strictly prohibited.


* The runner harness `apps/995.library/995.library/**` remains immutable (0 lines modified).

2. **Self-Healing Four-Vector & Continuity Table Bootstrapping:**
* Execute idempotent `CREATE TABLE IF NOT EXISTS` DDL statements across all 10 tables on the mounted database handle:


1. `event_log`: Monotonic sequence counter (`sequence_number`), `event_id`, `event_type`, `tick`, `payload_json`, `state_hash`, `prev_hash`, `seed`, `timestamp_ms`.


2. `entities_live`: Coordinates `q`, `r`, `location_id`, `arms_available` ($0 \le x \le 2$), `legs_available` ($0 \le x \le 2$), `locomotion_tier`, `concussion_level` ($0 \le x \le 3$), `wounds_json`, `held_items_json`, `worn_gear_json`, `slung_items_json`.


3. `strings_ledger`: Interpersonal leverage debt tokens with anti-reflexive constraint (`CHECK(holder_id <> debtor_id)`) and integer basis points (`CHECK(weight_bps BETWEEN 0 AND 10000)`).


4. `grievance_status`: Narrative escalation stage, cooldown cycles, and intensity rating (`CHECK(intensity_bps BETWEEN 0 AND 10000)`).


5. `paranoia_clocks`: Ticking doom meters with basis points (`CHECK(value_bps BETWEEN 0 AND 10000)`, `CHECK(threshold_bps BETWEEN 0 AND 10000)`).


6. `actor_beliefs`: Epistemic horizon register (`CHECK(subjective_probability_bps BETWEEN 0 AND 10000)`, `source_type IN ('INFERENCE', 'TESTIMONY', 'DIRECT_OBSERVATION')`).


7. `unratified_details`: Quarantined emergent props and facts awaiting consensus (`CHECK(salience_bps BETWEEN 0 AND 10000)`, `status IN ('STAGED', 'DISCARDED', 'RATIFIED')`).


8. `facts`: Monotonic canonical claims ledger (`version INTEGER NOT NULL DEFAULT 1`, `scope IN ('CHARACTER', 'LOCATION', 'LOGISTIC', 'LAW')`, `standing IN ('ACTIVE_ENDORSED', 'FLAGGED_FOR_REVIEW', 'DEPRECATED_ARCHIVED')`).


9. `scenes`: Continuity scene tracker (`status IN ('clean', 'dirty', 'reviewed')`, `is_dirty INTEGER NOT NULL DEFAULT 0`).


10. `scene_fact_dependencies`: Junction table binding `scene_id` and `fact_id` with `pinned_version INTEGER NOT NULL`.




* Create passive graph centrality telemetry view: `v_fact_centrality_telemetry`.




3. **Append-Only & Monotonic Trigger Arming:**
* `trg_event_log_no_update`: `BEFORE UPDATE ON event_log` raises `ABORT` with message `[IMMUTABILITY_VIOLATION] Updates forbidden on event_log`.


* `trg_event_log_no_delete`: `BEFORE DELETE ON event_log` raises `ABORT` with message `[IMMUTABILITY_VIOLATION] Deletions forbidden on event_log`.


* `trg_fact_version_bump`: `AFTER UPDATE ON facts` when `NEW.version > OLD.version OR NEW.standing = 'DEPRECATED_ARCHIVED'`, cascades `is_dirty = 1` and `status = 'dirty'` to all matching scenes in `scene_fact_dependencies` where `pinned_version < NEW.version`.




4. **Pure 7-Bit ASCII Telemetry Streaming:**

-- ==============================================================================
-- IMMUTABILITY TRIGGERS ON event_log
-- ==============================================================================
CREATE TRIGGER IF NOT EXISTS trg_event_log_no_update
BEFORE UPDATE ON event_log
BEGIN
  SELECT RAISE(ABORT, '[IMMUTABILITY_VIOLATION] Updates forbidden on event_log');
END;

CREATE TRIGGER IF NOT EXISTS trg_event_log_no_delete
BEFORE DELETE ON event_log
BEGIN
  SELECT RAISE(ABORT, '[IMMUTABILITY_VIOLATION] Deletions forbidden on event_log');
END;

-- ==============================================================================
-- MONOTONIC VERSION CASCADE TRIGGER ON facts
-- ==============================================================================
CREATE TRIGGER IF NOT EXISTS trg_fact_version_bump
AFTER UPDATE ON facts
FOR EACH ROW
WHEN NEW.version > OLD.version OR NEW.standing = 'DEPRECATED_ARCHIVED'
BEGIN
  UPDATE scenes
  SET is_dirty = 1,
      status = 'dirty'
  WHERE id IN (
    SELECT scene_id
    FROM scene_fact_dependencies
    WHERE fact_id = NEW.id
      AND pinned_version < NEW.version
  );
END;
`;

import Database from 'better-sqlite3';
import { DatabaseModel } from '../database.model.js';
import type DatabaseBit from '../fce/database.bit.js';
import type { DatabaseConnectionMetadata, SchemaInventoryReport } from '../fce/database.interface.js';
import { FOUR_VECTOR_AND_CONTINUITY_DDL, SCHEMA_TRIGGERS_DDL } from './schemaDdl.buzz.js';
import type State from '../../99.core/state.js';

const UPDATE_CONSOLE = '[Console action] Update Console';

async function emitConsole(ste: State | undefined, src: string): Promise<void> {
  if (ste && typeof ste.hunt === 'function') {
    await ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src }).catch(() => {});
  }
}

export const closeDatabaseConnection = (cpy: DatabaseModel): void => {
  if (cpy.connection) {
    try {
      if (typeof cpy.connection.close === 'function') {
        cpy.connection.close();
      }
    } catch {
      // Ignore already closed handles
    }
    cpy.connection = null;
  }
  if (cpy.activeMetadata) {
    cpy.activeMetadata.isOpen = false;
  }
};

export const mountRealmDatabase = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  const seriesSlug = bal?.dat?.seriesSlug || 'default-realm';
  const targetPath = bal?.dat?.storageTarget || ':memory:';

  if (
    cpy.connection &&
    cpy.activeMetadata &&
    cpy.activeMetadata.seriesSlug === seriesSlug &&
    cpy.activeMetadata.storageTarget === targetPath &&
    cpy.activeMetadata.isOpen
  ) {
    await emitConsole(ste, `>> [DB:MOUNT] Retaining active connection for: [${seriesSlug}]`);
    if (bal?.slv) {
      bal.slv({
        dbsBit: {
          idx: 'mount-realm-database-pass',
          val: 1,
          src: seriesSlug,
          dat: { metadata: cpy.activeMetadata },
        },
      });
    }
    return cpy;
  }

  if (cpy.connection) {
    await emitConsole(
      ste,
      `>> [DB:SWITCH] Unmounting previous connection [${cpy.activeMetadata?.seriesSlug || 'unknown'}]...`,
    );
    closeDatabaseConnection(cpy);
  }

  await emitConsole(ste, `>> [DB:MOUNT] Opening discrete realm storage at [${targetPath}]...`);

  const db = new (Database as any)(targetPath);

  db.pragma('foreign_keys = ON');
  if (targetPath !== ':memory:') {
    db.pragma('journal_mode = WAL');
    db.pragma('synchronous = NORMAL');
  }

  const fkPragma = db.pragma('foreign_keys', { simple: true });
  const walPragma = targetPath === ':memory:' ? 'memory' : db.pragma('journal_mode', { simple: true });

  const metadata: DatabaseConnectionMetadata = {
    seriesSlug,
    storageTarget: targetPath,
    isOpen: true,
    walEnabled: String(walPragma).toLowerCase() === 'wal',
    foreignKeysEnabled: Boolean(fkPragma === 1 || fkPragma === true),
    mountedAtMs: Date.now(),
  };

  cpy.connection = db;
  cpy.activeMetadata = metadata;

  await emitConsole(
    ste,
    `>> [DB:PRAGMA] foreign_keys=${metadata.foreignKeysEnabled} journal_mode=${walPragma} [OK]`,
  );

  if (bal?.slv) {
    bal.slv({
      dbsBit: {
        idx: 'mount-realm-database-pass',
        val: 1,
        src: seriesSlug,
        dat: { metadata },
      },
    });
  }

  return cpy;
};

export const ensureSchema = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  if (!cpy.connection) {
    // If no active connection, mount transient in-memory handle by default
    await mountRealmDatabase(cpy, { dat: { seriesSlug: 'transient', storageTarget: ':memory:' } } as any, ste);
  }

  const db = cpy.connection;

  await emitConsole(ste, '>> [DB:SCHEMA] Bootstrapping Four-Vector and Continuity tables...');
  db.exec(FOUR_VECTOR_AND_CONTINUITY_DDL);

  await emitConsole(ste, '>> [DB:TRIGGERS] Arming append-only and monotonic invalidation triggers...');
  db.exec(SCHEMA_TRIGGERS_DDL);

  // Compile Schema Inventory
  const tables = db
    .prepare("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name ASC")
    .all()
    .map((r: any) => r.name);

  const views = db
    .prepare("SELECT name FROM sqlite_master WHERE type='view' ORDER BY name ASC")
    .all()
    .map((r: any) => r.name);

  const triggers = db
    .prepare("SELECT name FROM sqlite_master WHERE type='trigger' ORDER BY name ASC")
    .all()
    .map((r: any) => r.name);

  const isArmored =
    triggers.includes('trg_event_log_no_update') &&
    triggers.includes('trg_event_log_no_delete') &&
    triggers.includes('trg_fact_version_bump');

  const report: SchemaInventoryReport = {
    tables,
    views,
    triggers,
    isArmored,
  };

  cpy.schemaReport = report;

  await emitConsole(
    ste,
    `>> [DB:READY] Relational schema armed. ${tables.length} tables, ${views.length} view, ${triggers.length} triggers [OK]`,
  );

  if (bal?.slv) {
    bal.slv({
      dbsBit: {
        idx: 'ensure-schema-pass',
        val: 1,
        src: cpy.activeMetadata?.seriesSlug || 'unknown',
        dat: { report },
      },
    });
  }

  return cpy;
};

export const initDatabase = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  if (bal?.dat?.seriesSlug || bal?.dat?.storageTarget) {
    await mountRealmDatabase(cpy, bal, ste);
  }
  return cpy;
};

### 5. `packages/300.engine/02.database.unit/database.reduce.ts`

Wire reducer routing for `ENSURE_SCHEMA`:

```typescript
import clone from 'clone-deep';
import * as ActDbs from './database.action.js';
import * as Buzz from './database.buzzer.js';
import type DatabaseModel from './database.model.js';
import type DatabaseBit from './fce/database.bit.js';
import type State from '../99.core/state.js';

export const reduceDatabase = async (
  model: DatabaseModel,
  action: string,
  bit: DatabaseBit,
  ste: State,
): Promise<DatabaseModel> => {
  const cpy = clone(model);
  switch (action) {
    case ActDbs.INIT_DATABASE:
      return await Buzz.initDatabase(cpy, bit, ste);
    case ActDbs.MOUNT_REALM_DATABASE:
      return await Buzz.mountRealmDatabase(cpy, bit, ste);
    case ActDbs.ENSURE_SCHEMA:
      return await Buzz.ensureSchema(cpy, bit, ste);
    default:
      return cpy;
  }
};

export default reduceDatabase;

```
