```typescript
/**
 * 10.location.unit - Integer Axial Coordinate Packing Transformer
 * Exact 32-bit unsigned integer packing adhering to Gillian Arithmetic Doctrine.
 */
import type { AxialCoord, PackedAxial } from '../fce/spatial-overlay.interface.js';

const AXIAL_OFFSET = 32768; // 16-bit signed range [-32768, 32767]

export const packAxial = (q: number, r: number): PackedAxial => {
  if (!Number.isInteger(q) || !Number.isInteger(r)) {
    throw new Error(`[AXIAL_PACK_ERROR] Coordinates must be discrete integers: q=${q}, r=${r}`);
  }

  const qShifted = (q + AXIAL_OFFSET) & 0xffff;
  const rShifted = (r + AXIAL_OFFSET) & 0xffff;

  return ((qShifted << 16) | rShifted) >>> 0;
};

export const unpackAxial = (packed: PackedAxial): AxialCoord => {
  if (!Number.isInteger(packed) || packed < 0 || packed > 0xffffffff) {
    throw new Error(`[AXIAL_UNPACK_ERROR] Invalid unsigned 32-bit packed coordinate: ${packed}`);
  }

  const qShifted = (packed >>> 16) & 0xffff;
  const rShifted = packed & 0xffff;

  const q = qShifted - AXIAL_OFFSET;
  const r = rShifted - AXIAL_OFFSET;
  const s = -q - r; // Cubic planar zero-sum invariant: q + r + s === 0

  return Object.freeze({ q, r, s });
};
```

```typescript
import type { PortalRecord } from '../fce/location.interface.js';
import type { PackedAxial, SpatialOverlaySnapshot } from '../fce/spatial-overlay.interface.js';

const canonicalizeJson = (val: unknown): string => {
  if (val === null || typeof val !== 'object') {
    return JSON.stringify(val);
  }
  if (Array.isArray(val)) {
    return '[' + val.map((item) => canonicalizeJson(item)).join(',') + ']';
  }
  const keys = Object.keys(val as Record<string, unknown>).sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  return '{' + keys.map((k) => JSON.stringify(k) + ':' + canonicalizeJson((val as any)[k])).join(',') + '}';
};

export const buildSpatialOverlaySnapshot = (
  portals: ReadonlyMap<string, PortalRecord>,
  dynamicObstacles: readonly PackedAxial[] = [],
  costOverrides?: Readonly<Record<number, number>>,
  acousticOverridesCdB?: Readonly<Record<string, number>>,
): SpatialOverlaySnapshot => {
  const closedPortals: string[] = [];

  for (const [portalId, portal] of portals.entries()) {
    if (portal.status === 'LOCKED' || portal.status === 'COLLAPSED') {
      closedPortals.push(portalId);
    }
  }

  closedPortals.sort((a, b) => (a < b ? -1 : a > b ? 1 : 0));
  const sortedObstacles = [...dynamicObstacles].sort((a, b) => a - b);

  const preImage = {
    acousticDampingOverridesCdB: acousticOverridesCdB || {},
    closedPortals,
    impassableCells: sortedObstacles,
    traversalCostOverrides: costOverrides || {},
  };

  const canonical = canonicalizeJson(preImage);
  const overlayHash = crypto.createHash('sha256').update(canonical).digest('hex');

  return Object.freeze({
    overlayHash,
    closedPortals: Object.freeze(closedPortals),
    impassableCells: Object.freeze(sortedObstacles),
    traversalCostOverrides: costOverrides ? Object.freeze({ ...costOverrides }) : undefined,
    acousticDampingOverridesCdB: acousticOverridesCdB ? Object.freeze({ ...acousticOverridesCdB }) : undefined,
  });
};
```

```typescript
/**
 * 10.location.unit - Location Reducer Router
 */
import clone from 'clone-deep';
import * as Act from './location.action.js';
import { LocationModel } from './location.model.js';
import * as Buzz from './location.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: LocationModel = new LocationModel(),
  act: Act.Actions,
  state?: State,
): LocationModel | Promise<LocationModel> {
  switch (act.type) {
    case Act.INIT_LOCATION:
      return Buzz.initLocation(clone(model), act.bale, state);

    case Act.UPDATE_LOCATION:
      return Buzz.updateLocation(clone(model), act.bale, state);

    case Act.REGISTER_LOCATION:
      return Buzz.registerLocation(clone(model), act.bale as any, state);

    case Act.SET_PORTAL_LOCK:
      return Buzz.setPortalLockAction(clone(model), act.bale as any, state);

    case Act.VALIDATE_ADJACENCY:
      return Buzz.validateAdjacencyAction(clone(model), act.bale as any, state);

    case Act.CHECK_PORTAL_CROSSING:
      return Buzz.checkPortalCrossingAction(clone(model), act.bale as any, state);

    case Act.TRAVERSE_LOCATION:
      return Buzz.traverseLocationAction(clone(model), act.bale as any, state);

    case Act.AUDIT_TOPOLOGY:
      return Buzz.auditTopologyAction(clone(model), act.bale as any, state);

    case Act.BIND_SPATIAL_PORT:
      return Buzz.bindSpatialPortAction(clone(model), act.bale as any, state);

    case Act.EVALUATE_SPATIAL_PATH:
      return Buzz.evaluateSpatialPathAction(clone(model), act.bale as any, state);

    case Act.EVALUATE_SPATIAL_PERCEPTION:
      return Buzz.evaluateSpatialPerceptionAction(clone(model), act.bale as any, state);

    case Act.SANITIZE_PATH_QUERY: {
      const result = Buzz.sanitizePathQuery((act.bale?.dat as any)?.rawQuery);
      if (act.bale?.slv) {
        act.bale.slv({
          locBit: {
            idx: result.ok ? 'sanitize-path-success' : 'sanitize-path-fail',
            val: result.ok ? 1 : 0,
            dat: { error: result.error, pathReceipt: result as any },
          },
        });
      }
      return model;
    }

    case Act.SANITIZE_PERCEPTION_QUERY: {
      const result = Buzz.sanitizePerceptionQuery((act.bale?.dat as any)?.rawQuery);
      if (act.bale?.slv) {
        act.bale.slv({
          locBit: {
            idx: result.ok ? 'sanitize-perception-success' : 'sanitize-perception-fail',
            val: result.ok ? 1 : 0,
            dat: { error: result.error, perceptCandidate: result as any },
          },
        });
      }
      return model;
    }

    case Act.BUILD_SPATIAL_OVERLAY: {
      const snapshot = Buzz.buildSpatialOverlaySnapshot(
        model.portals,
        (act.bale?.dat as any)?.dynamicObstacles || [],
        (act.bale?.dat as any)?.costOverrides,
        (act.bale?.dat as any)?.acousticOverrides,
      );
      if (act.bale?.slv) {
        act.bale.slv({
          locBit: {
            idx: 'build-spatial-overlay-success',
            src: snapshot.overlayHash,
            dat: { overlaySnapshot: snapshot },
          },
        });
      }
      return model;
    }

    default:
      return model;
  }
}
```

```typescript
    const coords = [
      { q: 0, r: 0 },
      { q: 12, r: -4 },
      { q: -55, r: 23 },
      { q: 32000, r: -32000 },
    ];

    for (const c of coords) {

      expect(Number.isInteger(packed)).toBe(true);
      expect(packed).toBeGreaterThanOrEqual(0);

      const unpacked = unpackAxial(packed);
      expect(unpacked.q).toBe(c.q);
      expect(unpacked.r).toBe(c.r);
      expect(unpacked.q + unpacked.r + unpacked.s).toBe(0); // Cubic zero-sum invariant
    }
  });
```
