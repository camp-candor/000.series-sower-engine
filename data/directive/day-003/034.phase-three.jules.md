/**
 * 03.entity.unit - Somatic State Query Extractor
 * Extracts and asserts Gillian integer basis-point bounds from raw entity state.
 */
import type { SomaticState, LocomotionTier } from '../fce/somatic-budget.interface.js';

const VALID_TIERS: readonly LocomotionTier[] = Object.freeze([
  'UNRESTRICTED',
  'LIMP',
  'NEUROPATHY_SHUFFLE',
  'IMMOBILIZED',
]);

const assertInteger = (val: unknown, fieldName: string): number => {
  if (typeof val !== 'number' || !Number.isInteger(val)) {
    throw new Error(`[SOMATIC_INTEGER_BREACH] ${fieldName} must be a discrete integer. Received: ${val}`);
  }
  return val;
};

const assertBasisPoint = (val: unknown, fieldName: string): number => {
  const intVal = assertInteger(val, fieldName);
  if (intVal < 0 || intVal > 10000) {
    throw new Error(`[GILLIAN_BPS_BREACH] ${fieldName} must be within [0, 10000] bps. Received: ${intVal}`);
  }
  return intVal;
};

export const extractSomaticState = (raw: any): SomaticState => {
  if (!raw || typeof raw !== 'object') {
    throw new Error('[SOMATIC_QUERY_ERROR] Entity record must be an object');
  }

  const entityId = typeof raw.entityId === 'string' ? raw.entityId : 'char_unknown';
  const availableArms = Math.min(2, Math.max(0, assertInteger(raw.availableArms ?? 2, 'availableArms')));
  const availableLegs = Math.min(2, Math.max(0, assertInteger(raw.availableLegs ?? 2, 'availableLegs')));

  let locomotionTier: LocomotionTier = 'UNRESTRICTED';
  if (typeof raw.locomotionTier === 'string' && VALID_TIERS.includes(raw.locomotionTier as LocomotionTier)) {
    locomotionTier = raw.locomotionTier as LocomotionTier;
  } else if (availableLegs < 1) {
    locomotionTier = 'IMMOBILIZED';
  } else if (availableLegs < 2) {
    locomotionTier = 'LIMP';
  }

  const fatigueBps = assertBasisPoint(raw.fatigueBps ?? 0, 'fatigueBps');
  const painBps = assertBasisPoint(raw.painBps ?? 0, 'painBps');
  const refusalStrainBps = assertBasisPoint(raw.refusalStrainBps ?? 0, 'refusalStrainBps');

  const activeTrauma: string[] = [];
  if (Array.isArray(raw.activeTrauma)) {
    for (const t of raw.activeTrauma) {
      if (typeof t === 'string' && t.trim().length > 0) {
        activeTrauma.push(t.trim().toLowerCase());
      }
    }
  }

  return Object.freeze({
    entityId,
    availableArms,
    availableLegs,
    locomotionTier,
    fatigueBps,
    painBps,
    refusalStrainBps,
    activeTrauma: Object.freeze(activeTrauma.sort()),
  });
};

/**
 * 03.entity.unit - Biological Condition Budget Calculator
 * Derives scalar movement costs, banned kinetic verb masks, and hand capacities.
 */
import type {
  SomaticState,
  SomaticBudgetReceipt,
  HeldItemDescriptor,
  HandLoadEvaluation,
} from '../fce/somatic-budget.interface.js';

export const BASE_MOVEMENT_UNIT_COST = 100; // Integer centisecond base cost

export const deriveSomaticBudget = (somatic: SomaticState): SomaticBudgetReceipt => {
  const banned = new Set<string>();

  // 1. Arm Capability Analysis
  if (somatic.availableArms < 2) {
    banned.add('WIELD_TWO_HANDED');
    banned.add('CLIMB_ROPE');
    banned.add('DUAL_WIELD');
    banned.add('VAULT_OBSTACLE');
  }

  if (somatic.availableArms < 1) {
    banned.add('WIELD_ONE_HANDED');
    banned.add('PULL_LEVER');
    banned.add('GRASP_OBJECT');
    banned.add('PUNCH');
  }

  // 2. Locomotion & Leg Capability Analysis
  if (somatic.availableLegs < 2 || somatic.locomotionTier === 'LIMP') {
    banned.add('SPRINT');
    banned.add('LEAP');
  }

  if (somatic.locomotionTier === 'NEUROPATHY_SHUFFLE') {
    banned.add('SPRINT');
    banned.add('LEAP');
    banned.add('FAST_RETREAT');
  }

  if (somatic.availableLegs < 1 || somatic.locomotionTier === 'IMMOBILIZED') {
    banned.add('WALK');
    banned.add('SPRINT');
    banned.add('LEAP');
    banned.add('TACTICAL_REPOSITION');
  }

  // 3. Trauma Condition Overlays
  for (const condition of somatic.activeTrauma) {
    if (condition === 'sprained_wrist_right' || condition === 'dislocated_thumb') {
      banned.add('WIELD_TWO_HANDED');
      banned.add('HEAVY_DRAW');
    }
    if (condition === 'brachial_plexus_severed' || condition === 'paralyzed_left_arm') {
      banned.add('WIELD_TWO_HANDED');
      banned.add('CLIMB_ROPE');
    }
    if (condition === 'concussed_severe') {
      banned.add('AIMED_RANGED_ATTACK');
      banned.add('RAPID_PARRY');
    }
  }

  // 4. Exhaustion & Pain Penalties (Integer Bps Bounds)
  if (somatic.fatigueBps >= 7500 || somatic.painBps >= 8000) {
    banned.add('SPRINT');
    banned.add('HEAVY_ATTACK');
  }

  // 5. Integer Scalar Movement Budget Calculation
  let tierMultiplierBps = 10000; // 1.0x baseline
  if (somatic.locomotionTier === 'LIMP') {
    tierMultiplierBps = 15000; // 1.5x cost
  } else if (somatic.locomotionTier === 'NEUROPATHY_SHUFFLE') {
    tierMultiplierBps = 25000; // 2.5x cost
  } else if (somatic.locomotionTier === 'IMMOBILIZED') {
    tierMultiplierBps = 100000; // 10.0x cost
  }

  // Strict integer scaling using integer division
  const fatigueScaled = Math.floor((BASE_MOVEMENT_UNIT_COST * (10000 + somatic.fatigueBps)) / 10000);
  const movementBudgetCostUnits = Math.floor((fatigueScaled * tierMultiplierBps) / 10000);

  const bannedKineticVerbs = Object.freeze(Array.from(banned).sort());

  return Object.freeze({
    entityId: somatic.entityId,
    movementBudgetCostUnits,
    bannedKineticVerbs,
    canActuateTwoHandedVerbs: somatic.availableArms >= 2 && !banned.has('WIELD_TWO_HANDED'),
    canSprint: !banned.has('SPRINT'),
  });
};

export const validateHandCapacity = (
  availableArms: number,
  heldItems: readonly HeldItemDescriptor[],
): HandLoadEvaluation => {
  let totalOccupiedHands = 0;
  for (const item of heldItems) {
    totalOccupiedHands += item.occupiesHands;
  }

  const valid = totalOccupiedHands <= availableArms;
  return Object.freeze({
    valid,
    totalOccupiedHands,
    availableArms,
    error: valid

/**
 * 10.location.unit - Traversal Budget Guard Buzzer
 * Enforces fail-closed halting when spatial traversal exceeds somatic energy budget.
 */

export interface BudgetGuardResult {
  readonly permitted: boolean;
  readonly traversalCostUnits: number;
  readonly allowedBudgetUnits: number;
  readonly marginUnits: number;
  readonly error?: string;
}

export const verifyTraversalBudget = (
  traversalCostUnits: number,
  allowedBudgetUnits: number,
): BudgetGuardResult => {
  if (!Number.isInteger(traversalCostUnits) || !Number.isInteger(allowedBudgetUnits)) {
    throw new Error('[BUDGET_GUARD_ERROR] Budget and traversal costs must be discrete integers');
  }

  const marginUnits = allowedBudgetUnits - traversalCostUnits;
  const permitted = marginUnits >= 0;

  return Object.freeze({
    permitted,
    traversalCostUnits,
    allowedBudgetUnits,
    marginUnits,
    error: permitted
      ? undefined
      : `BUDGET_EXCEEDED: Traversal cost of ${traversalCostUnits} exceeds somatic energy budget of ${allowedBudgetUnits}`,
  });
};

/**
 * 03.entity.unit - Entity Reducer Router
 */
import clone from 'clone-deep';
import * as Act from './entity.action.js';
import { EntityModel } from './entity.model.js';
import * as Buzz from './entity.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: EntityModel = new EntityModel(),
  act: Act.Actions,
  state?: State,
): EntityModel | Promise<EntityModel> {
  switch (act.type) {
    case Act.INIT_ENTITY:
      return Buzz.initEntity(clone(model), act.bale, state);

    case Act.UPDATE_ENTITY:
      return Buzz.updateEntity(clone(model), act.bale, state);

    case Act.DERIVE_SOMATIC_BUDGET: {
      try {
        const somatic = Buzz.extractSomaticState(act.bale?.dat);
        const receipt = Buzz.deriveSomaticBudget(somatic);
        if (act.bale?.slv) {
          act.bale.slv({
            entBit: {
              idx: 'derive-somatic-budget-success',
              val: receipt.movementBudgetCostUnits,
              dat: receipt as any,
            },
          });
        }
      } catch (err: any) {
        if (act.bale?.slv) {
          act.bale.slv({
            entBit: {
              idx: 'derive-somatic-budget-fail',
              val: -1,
              dat: { error: err.message } as any,
            },
          });
        }
      }
      return model;
    }

    case Act.VALIDATE_HAND_CAPACITY: {
      const arms = (act.bale?.dat as any)?.availableArms ?? 2;
      const items = (act.bale?.dat as any)?.heldItems ?? [];
      const evaluation = Buzz.validateHandCapacity(arms, items);
      if (act.bale?.slv) {
        act.bale.slv({
          entBit: {
            idx: evaluation.valid ? 'validate-hand-capacity-success' : 'validate-hand-capacity-fail',
            val: evaluation.valid ? 1 : 0,
            dat: evaluation as any,
          },
        });
      }
      return model;
    }

    default:
      return model;
  }
}
