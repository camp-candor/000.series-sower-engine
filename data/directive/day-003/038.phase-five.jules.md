  await emitConsole(ste, '>> [SCHEMA:ENSURE] [OK] event_log equipped with spatial proof quadruple.');
  await emitConsole(ste, '>> [TRIGGERS:ARM] [OK] Immutability abort sentinels armed on event_log.');

  if (bal?.slv) {
    bal.slv({ datBit: { idx: 'ensure-schema-success', val: 1 } });
  }
  return cpy;
};

export const migrateSpatialProofSchemaAction = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  if (!cpy.dbHandle) {
    throw new Error('[DATABASE_ERROR] Cannot migrate schema without active connection');
  }

  const applied = migrateSpatialProofSchema(cpy.dbHandle);
  cpy.hasSpatialSchema = applied;

  await emitConsole(ste, '>> [SCHEMA:ALTER] [OK] Spatial proof columns verified in event_log.');

  if (bal?.slv) {
    bal.slv({
      datBit: {
        idx: 'migrate-spatial-proof-success',
        val: applied ? 1 : 0,
        dat: { migrationApplied: applied },
      },
    });
  }
  return cpy;
};

export const appendEventLogWithProof = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  if (!cpy.dbHandle) {
    throw new Error('[DATABASE_ERROR] Active database handle required to append event');
  }

  const payload: AppendSpatialEventPayload | undefined = bal?.dat?.appendEvent;
  if (!payload) {
    if (bal?.slv) {
      bal.slv({ datBit: { idx: 'append-event-fail', dat: { error: 'PAYLOAD_MANDATORY' } } });
    }
    return cpy;
  }

  const topologyHash = payload.spatialProof?.topologyHash || '';
  const overlayHash = payload.spatialProof?.overlayHash || '';
  const queryHash = payload.spatialProof?.queryHash || '';
  const receiptHash = payload.spatialProof?.receiptHash || '';
  const timestampMs = payload.timestampMs ?? Date.now();

  const stmt = cpy.dbHandle.prepare(`
    INSERT INTO event_log (
      event_id, tick, event_type, actor_id, payload_json,
      prev_event_hash, event_hash, topology_hash, overlay_hash,
      query_hash, receipt_hash, timestamp_ms
    ) VALUES (
      ?, ?, ?, ?, ?,
      ?, ?, ?, ?,
      ?, ?, ?
    )
  `);

  const info = stmt.run(
    payload.eventId,
    payload.tick,
    payload.eventType,
    payload.actorId || null,
    payload.payloadJson,
    payload.prevEventHash,
    payload.eventHash,
    topologyHash,
    overlayHash,
    queryHash,
    receiptHash,
    timestampMs,
  );

  const seq = Number(info.lastInsertRowid);
  cpy.totalEventsCommitted = seq;

  const committedRecord: EventLogRecord = Object.freeze({
    sequence_number: seq,
    event_id: payload.eventId,
    tick: payload.tick,
    event_type: payload.eventType,
    actor_id: payload.actorId,
    payload_json: payload.payloadJson,
    prev_event_hash: payload.prevEventHash,
    event_hash: payload.eventHash,
    topology_hash: topologyHash,
    overlay_hash: overlayHash,
    query_hash: queryHash,
    receipt_hash: receiptHash,
    timestamp_ms: timestampMs,
  });

  if (bal?.slv) {
    bal.slv({
      datBit: {
        idx: 'append-event-success',
        val: seq,
        dat: { committedRecord },
      },
    });
  }
  return cpy;
};

import clone from 'clone-deep';
import * as Act from './database.action.js';
import { DatabaseModel } from './database.model.js';

export function reducer(
  model: DatabaseModel = new DatabaseModel(),
  act: Act.Actions,
  state?: State,
): DatabaseModel | Promise<DatabaseModel> {
  switch (act.type) {
    case Act.INIT_DATABASE:
      return Buzz.initDatabase(clone(model), act.bale, state);

    case Act.ENSURE_SCHEMA:

    case Act.APPEND_EVENT_LOG_WITH_PROOF:
      return Buzz.appendEventLogWithProof(clone(model), act.bale, state);

    default:
      return model;
  }
}
