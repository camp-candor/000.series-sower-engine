/**
 * 98.menu.unit - Menu Reducer Router
 */
import clone from 'clone-deep';
import * as Act from './menu.action.js';
import { MenuModel } from './menu.model.js';
import * as Buzz from './menu.buzzer.js';
import type State from '../99.core/state.js';

export function reducer(
  model: MenuModel = new MenuModel(),
  act: Act.Actions,
  state?: State,
): MenuModel | Promise<MenuModel> {
  switch (act.type) {
    case Act.INIT_MENU:
      return Buzz.initMenu(clone(model), act.bale, state);

    case Act.UPDATE_MENU:
      return Buzz.updateMenu(clone(model), act.bale, state);

    case Act.OPEN_SPATIAL_PROBE_MENU:
      return Buzz.openSpatialProbeMenuBuzz(clone(model), act.bale, state);

    case Act.DISPATCH_SPATIAL_PROBE:
      return Buzz.dispatchSpatialProbeBuzz(clone(model), act.bale, state);

    case Act.TEST_SPATIAL_WORK_CEILING:
      return Buzz.testSpatialWorkCeilingBuzz(clone(model), act.bale, state);

    default:
      return model;
  }
}
