1. **Quarantine Invariant (`check:native-io`):**
* `node:fs` and `node:path` must **only** be imported inside `packages/300.engine/98.menu.unit/`.


* The path `packages/300.engine/src/` is strictly prohibited. All code must reside directly inside `98.menu.unit` or its subdirectories (`fce/`, `buz/`).

2. **Artifact Discovery Contract:**
* Traverse from `process.cwd()` upward to locate the repository root containing `./compiled/`.


* Enumerate first-level subdirectories in `./compiled/`.


* Filter out hidden directories (`.git`, `.cache`), `.tmp` directories, and directories without a readable `bible-state.json`.




3. **Manifest Pre-Flight Validation ($T_0 = \$0.00$):**
* Implement `validateUniverseManifestShape(candidate: unknown): boolean`.


* Verify top-level structure:
* `seriesSlug`: non-empty string.


* `schemaVersion`: string.


* `merkleRootHex` (or `stateHash`): 64-character lowercase hex string (`/^[a-f0-9]{64}$/`).


* The four entity collections exist as objects or arrays: `characters`, `locations`, `possessions`, `grievances`.




* Filter out invalid candidates before presenting them to the operator.

### 4. `packages/300.engine/98.menu.unit/buz/00.menu.buzz.ts`

Implement `resolveEngineRoot`, `validateUniverseManifestShape`, and `scanCompiledUniverses`:

```typescript
import fs from 'node:fs';
import path from 'node:path';
import type { MenuModel } from '../menu.model.js';
import type MenuBit from '../fce/menu.bit.js';
import type { CompiledUniverseDescriptor } from '../fce/menu.interface.js';
import type State from '../../99.core/state.js';

export function resolveEngineRoot(startDir: string = process.cwd()): string {
  let curr = path.resolve(startDir);
  while (curr && curr !== path.dirname(curr)) {
    if (
      fs.existsSync(path.join(curr, 'compiled')) ||
      fs.existsSync(path.join(curr, 'packages', '300.engine'))
    ) {
      return curr;
    }
    curr = path.dirname(curr);
  }
  return path.resolve(startDir);
}

export function validateUniverseManifestShape(candidate: any): boolean {
  if (!candidate || typeof candidate !== 'object') {
    return false;
  }

  const slug = candidate.seriesSlug || candidate.slug;
  if (typeof slug !== 'string' || slug.trim().length === 0) {
    return false;
  }

  const hash = candidate.merkleRootHex || candidate.stateHash || candidate._meta?.corpusHash;
  if (typeof hash !== 'string' || !/^[a-f0-9]{64}$/i.test(hash)) {
    return false;
  }

  // Ensure collections exist either at root or nested under entities
  const entities = candidate.entities || candidate;
  const hasCharacters = Boolean(entities.characters && typeof entities.characters === 'object');
  const hasLocations = Boolean(entities.locations && typeof entities.locations === 'object');
  const hasPossessions = Boolean(
    (entities.possessions || entities.items) &&
    typeof (entities.possessions || entities.items) === 'object'
  );
  const hasGrievances = Boolean(entities.grievances && typeof entities.grievances === 'object');

  return hasCharacters && hasLocations && hasPossessions && hasGrievances;
}

export function scanCompiledUniverses(baseDir?: string): CompiledUniverseDescriptor[] {
  const root = baseDir ? path.resolve(baseDir) : resolveEngineRoot();
  const compiledDir = path.join(root, 'compiled');

  if (!fs.existsSync(compiledDir)) {
    return [];
  }

  const entries = fs.readdirSync(compiledDir, { withFileTypes: true });
  const results: CompiledUniverseDescriptor[] = [];

  for (const entry of entries) {
    if (!entry.isDirectory() || entry.name.startsWith('.') || entry.name.startsWith('_')) {
      continue;
    }

    const manifestPath = path.join(compiledDir, entry.name, 'bible-state.json');
    if (!fs.existsSync(manifestPath)) {
      continue;
    }

    try {
      const stats = fs.statSync(manifestPath);
      const rawContent = fs.readFileSync(manifestPath, 'utf8');
      const parsed = JSON.parse(rawContent);

      if (!validateUniverseManifestShape(parsed)) {
        continue;
      }

      const hash = parsed.merkleRootHex || parsed.stateHash || parsed._meta?.corpusHash || '';

      results.push({
        seriesSlug: entry.name,
        manifestPath,
        byteSize: stats.size,
        merkleRootHex: hash.toLowerCase(),
        hasBibleState: true,
      });
    } catch {
      // Malformed JSON fails closed
      continue;
    }
  }

  return results.sort((a, b) => a.seriesSlug.localeCompare(b.seriesSlug));
}

export const initMenu = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  if (bal && bal.slv != null) {
    bal.slv({ mnuBit: { idx: 'init-menu' } });
  }
  return cpy;
};

export const updateMenu = async (cpy: MenuModel, bal: MenuBit, ste: State) => {
  const library = (globalThis as any).LIBRARY;

  // Headless execution fallback
  if (!library || typeof library.hunt !== 'function') {
    const discovered = scanCompiledUniverses();
    cpy.discoveredUniverses = discovered;
    if (bal && bal.slv != null) {
      bal.slv({
        mnuBit: {
          idx: 'update-menu-headless',
          dat: { discovered },
        },
      });
    }
    return cpy;
  }

  // Terminal UI logic continues into Phase 2...
  if (bal && bal.slv != null) {
    bal.slv({ mnuBit: { idx: 'update-menu-complete' } });
  }
  return cpy;
};
```

### 5. `packages/300.engine/98.menu.unit/menu.test.ts`

Author unit verification suite to validate discovery and shape gating offline:

```typescript
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import fs from 'node:fs';
import path from 'node:path';
import os from 'node:os';
import {
  scanCompiledUniverses,
  validateUniverseManifestShape,
  resolveEngineRoot,
} from './buz/00.menu.buzz.js';

describe('98.menu.unit :: Local Compiled Universe Discovery (Phase 1)', () => {
  let tempDir: string;

  beforeEach(() => {
    tempDir = fs.mkdtempSync(path.join(os.tmpdir(), 'sower-test-engine-'));
  });

  afterEach(() => {
    fs.rmSync(tempDir, { recursive: true, force: true });
  });

  it('validates correct manifest shape against contract', () => {
    const validManifest = {
      seriesSlug: 'wiregrass-basin',
      schemaVersion: '1.0.0',
      merkleRootHex: 'a'.repeat(64),
      entities: {
        characters: { char_bog: {} },
        locations: { loc_pipe: {} },
        possessions: { item_scraper: {} },
        grievances: { grv_impasse: {} },
      },
    };
    expect(validateUniverseManifestShape(validManifest)).toBe(true);
  });

  it('rejects manifests missing required entity collections or valid hashes', () => {
    expect(validateUniverseManifestShape(null)).toBe(false);
    expect(validateUniverseManifestShape({})).toBe(false);
    expect(
      validateUniverseManifestShape({
        seriesSlug: 'bad-hash',
        merkleRootHex: 'not-a-64-char-hex',
        entities: { characters: {}, locations: {}, possessions: {}, grievances: {} },
      })
    ).toBe(false);
    expect(
      validateUniverseManifestShape({
        seriesSlug: 'missing-grievances',
        merkleRootHex: 'b'.repeat(64),
        entities: { characters: {}, locations: {}, possessions: {} },
      })
    ).toBe(false);
  });

  it('scans compiled/ directory and discovers valid universes while ignoring invalid ones', () => {
    const compiledPath = path.join(tempDir, 'compiled');
    fs.mkdirSync(compiledPath, { recursive: true });

    // 1. Valid universe
    const validSlugDir = path.join(compiledPath, 'wiregrass-basin');
    fs.mkdirSync(validSlugDir, { recursive: true });
    fs.writeFileSync(
      path.join(validSlugDir, 'bible-state.json'),
      JSON.stringify({
        seriesSlug: 'wiregrass-basin',
        schemaVersion: '1.0.0',
        merkleRootHex: 'c'.repeat(64),
        entities: {
          characters: { char_bog: {} },
          locations: { loc_pipe: {} },
          possessions: { item_scraper: {} },
          grievances: { grv_impasse: {} },
        },
      }),
      'utf8'
    );

    // 2. Directory without bible-state.json
    fs.mkdirSync(path.join(compiledPath, 'empty-realm'), { recursive: true });

    // 3. Directory with corrupt JSON
    const corruptSlugDir = path.join(compiledPath, 'corrupted-realm');
    fs.mkdirSync(corruptSlugDir, { recursive: true });
    fs.writeFileSync(path.join(corruptSlugDir, 'bible-state.json'), '{ invalid-json', 'utf8');

    const discovered = scanCompiledUniverses(tempDir);
    expect(discovered).toHaveLength(1);
    expect(discovered[0].seriesSlug).toBe('wiregrass-basin');
    expect(discovered[0].merkleRootHex).toBe('c'.repeat(64));
    expect(discovered[0].hasBibleState).toBe(true);
    expect(discovered[0].byteSize).toBeGreaterThan(0);
  });
});
```
