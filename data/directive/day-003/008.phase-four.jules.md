# Task: Implement Phase 4 Dynamic Partitioned Storage Path Allocation

## Objective

Implement **Phase 4: Dynamic Partitioned Storage Path Allocation** within `packages/300.engine` of `000.series-sower-engine`. This subsystem enables the **Discrete Realm Vault** architecture: dynamically resolving isolated, slug-partitioned SQLite storage paths (`data/sqlite/<seriesSlug>.local.db` or `:memory:`) inside `15.series.unit`, and wiring `02.database.unit` to cleanly unmount existing database handles, mount dedicated single-universe storage, and enforce relational pragmas (`foreign_keys = ON`, `journal_mode = WAL`, `synchronous = NORMAL`).

---

## Architectural Invariants & Monorepo Boundaries

1. **Absolute Ban on `packages/300.engine/src/`:** All code must reside strictly within numbered unit folders (e.g., `packages/300.engine/15.series.unit/`, `packages/300.engine/02.database.unit/`). Under no circumstances may any file be placed inside a `src/` directory within `packages/300.engine/`.


2. **Immutable Runner Boundary:** The runner harness `apps/995.library/995.library/**` is strictly immutable (0 lines modified).


3. **Discrete Realm Isolation:** Each world slug (`seriesSlug`) maps exclusively to its own database target. Monotonic sequence counters, foreign keys, and triggers must remain hermetically isolated per realm.


4. **Clean Connection Disconnect & Pragma Hardening:** Mounting a new realm database must cleanly close any existing connection handle, checkpoint open WAL logs, and arm SQLite connection pragmas before schema verification begins.


5. **Pure 7-Bit ASCII Telemetry:** All logs, telemetry, error strings, and comments must strictly use 7-bit clean ASCII characters (e.g., `>>`, `::`, `[OK]`, `[FAIL]`, `[VAULT]`, `[MOUNT]`, `[PRAGMA]`).


6. **NodeNext Explicit `.js` Imports:** All relative module import paths across TypeScript source files must explicitly terminate with `.js` extensions.

### 5. `packages/300.engine/15.series.unit/buz/series.buzz.ts`

Implement deterministic series storage resolution logic:

```typescript
import { SeriesModel } from '../menu.model.js';
import type SeriesBit from '../fce/series.bit.js';
import type { SeriesStorageAllocation } from '../fce/series.interface.js';
import type State from '../../99.core/state.js';

const UPDATE_CONSOLE = '[Console action] Update Console';

async function emitConsole(ste: State | undefined, src: string): Promise<void> {
  if (ste && typeof ste.hunt === 'function') {
    await ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src }).catch(() => {});
  }
}

export function computeStoragePathForSlug(slug: string, useMemory = false): string {
  if (useMemory || slug === ':memory:') {
    return ':memory:';
  }
  const sanitized = slug.trim().toLowerCase().replace(/[^a-z0-9_-]/g, '-');
  return `data/sqlite/${sanitized}.local.db`;
}

export const initSeries = async (
  cpy: SeriesModel,
  bal?: SeriesBit,
  ste?: State,
): Promise<SeriesModel> => {
  cpy.activeSeriesSlug = '';
  cpy.activeStorageTarget = '';
  cpy.allocations = {};

  if (bal?.slv) {
    bal.slv({
      srsBit: {
        idx: 'init-series-pass',
        val: 1,
        src: 'READY',
      },
    });
  }
  return cpy;
};

export const updateSeries = async (
  cpy: SeriesModel,
  bal?: SeriesBit,
  ste?: State,
): Promise<SeriesModel> => {
  if (bal?.slv) {
    bal.slv({
      srsBit: {
        idx: 'update-series-pass',
        val: 1,
        src: cpy.activeSeriesSlug,
      },
    });
  }
  return cpy;
};

export const resolveSeriesStorage = async (
  cpy: SeriesModel,
  bal?: SeriesBit,
  ste?: State,
): Promise<SeriesModel> => {
  const seriesSlug = bal?.dat?.seriesSlug || bal?.src || 'default-realm';
  const useMemory = Boolean(bal?.dat?.useMemory || process.env.VITEST === 'true');
  const explicitPath = bal?.dat?.targetPath;

  const storageTarget = explicitPath || computeStoragePathForSlug(seriesSlug, useMemory);
  const nowMs = bal?.dat?.timestampMs || Date.now();

  const allocation: SeriesStorageAllocation = {
    seriesSlug,
    storageTarget,
    isMemoryOnly: storageTarget === ':memory:',
    resolvedAtMs: nowMs,
  };

  cpy.activeSeriesSlug = seriesSlug;
  cpy.activeStorageTarget = storageTarget;
  cpy.allocations[seriesSlug] = allocation;

  await emitConsole(
    ste,
    `>> [VAULT:RESOLVE] Slug: [${seriesSlug}] Target: [${storageTarget}] (Memory: ${allocation.isMemoryOnly})`,
  );

  if (bal?.slv) {
    bal.slv({
      srsBit: {
        idx: 'resolve-series-storage-pass',
        val: 1,
        src: seriesSlug,
        dat: { allocation },
      },
    });
  }

  return cpy;
};

```

### 7. `packages/300.engine/15.series.unit/series.reduce.ts`

Wire reducer for storage actions:

```typescript
import clone from 'clone-deep';
import * as ActSrs from './series.action.js';
import * as Buzz from './series.buzzer.js';
import type SeriesModel from './series.model.js';
import type SeriesBit from './fce/series.bit.js';
import type State from '../99.core/state.js';

export const reduceSeries = async (
  model: SeriesModel,
  action: string,
  bit: SeriesBit,
  ste: State,
): Promise<SeriesModel> => {
  const cpy = clone(model);
  switch (action) {
    case ActSrs.INIT_SERIES:
      return await Buzz.initSeries(cpy, bit, ste);
    case ActSrs.UPDATE_SERIES:
      return await Buzz.updateSeries(cpy, bit, ste);
    case ActSrs.RESOLVE_SERIES_STORAGE:
      return await Buzz.resolveSeriesStorage(cpy, bit, ste);
    default:
      return cpy;
  }
};

export default reduceSeries;

```

### 10. `packages/300.engine/02.database.unit/buz/database.buzz.ts`

Implement connection disconnect switchboard and pragma arming:

```typescript
import Database from 'better-sqlite3';
import { DatabaseModel } from '../database.model.js';
import type DatabaseBit from '../fce/database.bit.js';
import type { DatabaseConnectionMetadata } from '../fce/database.interface.js';
import type State from '../../99.core/state.js';

const UPDATE_CONSOLE = '[Console action] Update Console';

async function emitConsole(ste: State | undefined, src: string): Promise<void> {
  if (ste && typeof ste.hunt === 'function') {
    await ste.hunt(UPDATE_CONSOLE, { idx: 'cns00', src }).catch(() => {});
  }
}

export const closeDatabaseConnection = (cpy: DatabaseModel): void => {
  if (cpy.connection) {
    try {
      if (typeof cpy.connection.close === 'function') {
        cpy.connection.close();
      }
    } catch {
      // Ignore already closed handles
    }
    cpy.connection = null;
  }
  if (cpy.activeMetadata) {
    cpy.activeMetadata.isOpen = false;
  }
};

export const mountRealmDatabase = async (
  cpy: DatabaseModel,
  bal?: DatabaseBit,
  ste?: State,
): Promise<DatabaseModel> => {
  const seriesSlug = bal?.dat?.seriesSlug || 'default-realm';
  const targetPath = bal?.dat?.storageTarget || ':memory:';

  // If already connected to the identical target, skip reconnecting
  if (
    cpy.connection &&
    cpy.activeMetadata &&
    cpy.activeMetadata.seriesSlug === seriesSlug &&
    cpy.activeMetadata.storageTarget === targetPath &&
    cpy.activeMetadata.isOpen
  ) {
    await emitConsole(ste, `>> [DB:MOUNT] Retaining active connection for: [${seriesSlug}]`);
    if (bal?.slv) {
      bal.slv({
        dbsBit: {
          idx: 'mount-realm-database-pass',
          val: 1,
          src: seriesSlug,
          dat: { metadata: cpy.activeMetadata },
        },
      });
    }
    return cpy;
  }

  // Cleanly close prior connection
  if (cpy.connection) {
    await emitConsole(
      ste,
      `>> [DB:SWITCH] Unmounting previous connection [${cpy.activeMetadata?.seriesSlug || 'unknown'}]...`,
    );
    closeDatabaseConnection(cpy);
  }

  await emitConsole(ste, `>> [DB:MOUNT] Opening discrete realm storage at [${targetPath}]...`);

  // Open SQLite database instance
  const db = new (Database as any)(targetPath);

  // Arm SQLite Performance & Integrity Pragmas
  db.pragma('foreign_keys = ON');
  if (targetPath !== ':memory:') {
    db.pragma('journal_mode = WAL');
    db.pragma('synchronous = NORMAL');
  }

  const fkPragma = db.pragma('foreign_keys', { simple: true });
  const walPragma = targetPath === ':memory:' ? 'memory' : db.pragma('journal_mode', { simple: true });

  const metadata: DatabaseConnectionMetadata = {
    seriesSlug,
    storageTarget: targetPath,
    isOpen: true,
    walEnabled: String(walPragma).toLowerCase() === 'wal',
    foreignKeysEnabled: Boolean(fkPragma === 1 || fkPragma === true),
    mountedAtMs: Date.now(),
  };

  cpy.connection = db;
  cpy.activeMetadata = metadata;

  await emitConsole(
    ste,
    `>> [DB:PRAGMA] foreign_keys=${metadata.foreignKeysEnabled} journal_mode=${walPragma} [OK]`,
  );

  if (bal?.slv) {
    bal.slv({
      dbsBit: {
        idx: 'mount-realm-database-pass',
        val: 1,
        src: seriesSlug,
        dat: { metadata },
      },
    });
  }

  return cpy;
};
```
