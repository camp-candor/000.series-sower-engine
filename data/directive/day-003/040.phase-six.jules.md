```typescript
    throw new Error('[PERCEPT_INGEST_ERROR] PerceptCandidate must be an object');
  }

  const channel: SensoryChannel = candidate.channel === 'ACOUSTIC' ? 'ACOUSTIC' : 'OPTICAL';

  if (!Number.isInteger(candidate.sourceCell) || !Number.isInteger(candidate.observerCell)) {
    throw new Error('[PERCEPT_INGEST_ERROR] sourceCell and observerCell must be packed integers');
  }

  const signalCdB = Number.isInteger(candidate.signalCdB) ? candidate.signalCdB : 0;
  const ambientNoiseCdB = Number.isInteger(candidate.ambientNoiseCdB) ? candidate.ambientNoiseCdB : 0;
  const snrMarginCdB = Number.isInteger(candidate.snrMarginCdB) ? candidate.snrMarginCdB : signalCdB - ambientNoiseCdB;

  const bearingDegrees = candidate.bearingDegrees !== null && Number.isInteger(candidate.bearingDegrees)
    ? Math.max(0, Math.min(359, candidate.bearingDegrees))
    : null;

  return Object.freeze({
    schemaVersion: candidate.schemaVersion ?? 1,
    kernelVersion: candidate.kernelVersion ?? '1.0.0',
    topologyHash: candidate.topologyHash ?? '',
    overlayHash: candidate.overlayHash ?? '',
    queryHash: candidate.queryHash ?? '',
    receiptHash: candidate.receiptHash ?? '',
    channel,
    sourceCell: candidate.sourceCell,
    observerCell: candidate.observerCell,
    signalCdB,
    ambientNoiseCdB,
    snrMarginCdB,
    bearingDegrees,
    isOccluded: Boolean(candidate.isOccluded),
  });
};
```

```typescript
/**
 * 11.belief.unit - Somatic Sensory Impairment Filter Gate
 * Evaluates biological receptor trauma, orientation cones, and acoustic thresholds.
 */
import type {
  PerceptCandidate,
  SomaticSensoryProfile,
  SensoryFilterEvaluation,
} from '../fce/percept.interface.js';

export const CONCUSSION_ACOUSTIC_ELEVATION_CDB = 1500; // +15 dB required for concussed ears

export const evaluateSensoryFilter = (
  percept: PerceptCandidate,
  profile: SomaticSensoryProfile,
): SensoryFilterEvaluation => {
  // 1. Hard Physical Occlusion Check
  if (percept.isOccluded) {
    return Object.freeze({
      received: false,
      effectiveQualityBps: 0,
      rejectionReason: 'GEOMETRIC_OCCLUSION',
    });
  }

  // 2. Optical Sensory Channel Evaluation
  if (percept.channel === 'OPTICAL') {
    if (profile.visualAcuity === 'BLIND') {
      return Object.freeze({
        received: false,
        effectiveQualityBps: 0,
        rejectionReason: 'SOMATIC_OPTICAL_BLINDNESS',
      });
    }

    if (profile.visualAcuity === 'MONOCULAR' && percept.bearingDegrees !== null) {
      // Map 0..359 degrees to 0..5 hex sector (60 deg per sector)
      const targetSector = Math.floor(percept.bearingDegrees / 60) % 6;
      const facingSector = Math.max(0, Math.min(5, profile.facingSector));

      // Calculate angular sector delta on modulo-6 ring: min((B - F) % 6, (F - B) % 6)
      const sectorDelta = Math.min(
        (targetSector - facingSector + 6) % 6,
        (facingSector - targetSector + 6) % 6,
      );

      // Monocular vision clamps field of view to 120-degree cone (sector delta <= 1)
      if (sectorDelta > 1) {
        return Object.freeze({
          received: false,
          effectiveQualityBps: 0,
          rejectionReason: 'MONOCULAR_BLIND_SECTOR',
        });
      }
    }

    // Optical quality scales from SNR margin into basis points [1000..10000]
    const effectiveQualityBps = Math.min(10000, Math.max(1000, percept.snrMarginCdB * 2));
    return Object.freeze({
      received: true,
      effectiveQualityBps,
    });
  }

  // 3. Acoustic Sensory Channel Evaluation
  if (percept.channel === 'ACOUSTIC') {
    let effectiveHearingThreshold = profile.hearingThresholdCdB;
    if (profile.hasConcussiveDeafness) {
      effectiveHearingThreshold += CONCUSSION_ACOUSTIC_ELEVATION_CDB;
    }

    if (percept.signalCdB < effectiveHearingThreshold) {
      return Object.freeze({
        received: false,
        effectiveQualityBps: 0,
        rejectionReason: 'BELOW_HEARING_THRESHOLD',
      });
    }

    if (percept.snrMarginCdB <= 0) {
      return Object.freeze({
        received: false,
        effectiveQualityBps: 0,
        rejectionReason: 'SIGNAL_DAMPED_INTO_AMBIENCE',
      });
    }

    // Acoustic quality derived from SNR margin
    const effectiveQualityBps = Math.min(10000, Math.max(1000, percept.snrMarginCdB * 2));
    return Object.freeze({
      received: true,
      effectiveQualityBps,
    });
  }

  return Object.freeze({
    received: false,
    effectiveQualityBps: 0,
    rejectionReason: 'GEOMETRIC_OCCLUSION',
  });
};
```

```typescript
/**
 * 11.belief.unit - Actor Belief Ledger Record Persistence & Combination
 * Synthesizes surviving sensory evidence into immutable BeliefRecords.
 */
import crypto from 'node:crypto';
import type { BeliefRecord } from '../fce/belief.interface.js';
import type { PerceptCandidate, SensoryFilterEvaluation } from '../fce/percept.interface.js';

export const deriveBeliefFromPercept = (
  actorId: string,
  eventId: string,
  propositionSlug: string,
  subjectId: string,
  predicate: string,
  objectId: string,
  percept: PerceptCandidate,
  filter: SensoryFilterEvaluation,
): BeliefRecord | null => {
  // Ignorance by Omission: if filter blocked signal, ZERO belief is created
  if (!filter.received) {
    return null;
  }

  const beliefId = 'bel_' + crypto.randomBytes(8).toString('hex');
  const subjectiveConfidenceBp = Math.min(10000, Math.max(1000, filter.effectiveQualityBps));

  return Object.freeze({
    belief_id: beliefId,
    holder_id: actorId,
    subject_id: subjectId,
    predicate,
    object_id: objectId,
    proposition_slug: propositionSlug,
    subjective_confidence_bp: subjectiveConfidenceBp,
    source_type: 'DIRECT_OBSERVATION',
    source_event_id: eventId,
    parent_belief_id: null,
    hop_count: 0,
    resists_correction: subjectiveConfidenceBp >= 9000,
    receipt_hash: percept.receiptHash,
  });
};

/**
 * Correlation-Aware Noisy-OR Combination Rule in Discrete Integer Basis Points
 * p_combined = 10000 - floor(((10000 - p1) * (10000 - p2)) / 10000)
 */
export const combineBeliefConfidencesNoisyOr = (
  confidenceA: number,
  confidenceB: number,
): number => {
  if (!Number.isInteger(confidenceA) || !Number.isInteger(confidenceB)) {
    throw new Error('[GILLIAN_INTEGER_BREACH] Confidence scores must be discrete integer basis points');
  }

  const p1 = Math.min(10000, Math.max(0, confidenceA));
  const p2 = Math.min(10000, Math.max(0, confidenceB));

  const remainingDoubtA = 10000 - p1;
  const remainingDoubtB = 10000 - p2;
  const combinedDoubt = Math.floor((remainingDoubtA * remainingDoubtB) / 10000);

  return Math.min(10000, Math.max(0, 10000 - combinedDoubt));
};
```
