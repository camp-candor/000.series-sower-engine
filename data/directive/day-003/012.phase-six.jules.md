# Task: Implement Phase 6 Genesis State Minting in `13.primordial.unit`

## Objective

Implement **Phase 6: Genesis State Minting** within `packages/300.engine/13.primordial.unit` of `000.series-sower-engine`. When the engine boot sequence reaches Phase 2, `13.primordial.unit` must:

1. Conduct an idempotent inspection of `event_log` in `02.database.unit`.


2. If clean, unpack the deep-frozen, in-memory `bibleState` provided by `00.engine.unit` into `entities_live`, `strings_ledger`, `grievance_status`, and `paranoia_clocks`.


3. Construct Sequence #1 (Genesis Tick $S_0$), link its previous hash to `GENESIS_HASH_64` (64 zeroes), calculate the chained event hash using canonical RFC 8785 JSON serialization and SHA-256, and mint it into `event_log`.


4. Dispatch `COMMIT_SEQUENCE` to `14.sequence.unit` to advance the monotonic sequence counter strictly to 1 under Optimistic Concurrency Control.


5. Stream pure 7-bit ASCII receipts to terminal cockpit console pane `cns00`.



---

## Architectural Invariants & Monorepo Boundaries

1. **Absolute Ban on `packages/300.engine/src/`:** All code, interfaces, buzzers, and tests must reside strictly within `packages/300.engine/13.primordial.unit/`.


2. **Zero Native I/O in Simulation Kernel (`check:native-io`):** `13.primordial.unit` must NEVER import `node:fs`, `node:child_process`, or external networking. It ingests its universe seed strictly from the in-memory `bal.dat.bibleState` delivered over `ste.hunt`.


3. **Idempotent Boot Safety:** If `event_log` already holds Sequence #1 (`evt_genesis_000` or count > 0), the seeder halts non-destructively, preserving existing records and returning `seeded: true` without overwriting data.


4. **The Gillian Arithmetic Doctrine ($0\dots 10{,}000\text{ bps}$):** All floating-point numbers are prohibited; all tension metrics, debts, and probabilities are stored as discrete integer basis points.


5. **No Wall-Clock Access:** `Date.now()`, `new Date()`, and `performance.now()` are forbidden. Integer timestamps must be supplied through injected `timestampMs` payloads.


6. **Immutable Runner Harness:** Files in `apps/995.library/995.library/**` must remain untouched (0 lines modified).


7. **Pure 7-Bit ASCII Compliance:** Telemetry receipts, comments, and strings must avoid multi-byte UTF-8 emojis.


8. **Explicit NodeNext `.js` Imports:** All relative imports must end with `.js` extensions.

export function computeManifestHRoot(bibleState: Readonly<Record<string, any>>): string {
  if (bibleState.merkleRootHex && /^[a-f0-9]{64}$/i.test(bibleState.merkleRootHex)) {
    return bibleState.merkleRootHex.toLowerCase();
  }
  if (bibleState.stateHash && /^[a-f0-9]{64}$/i.test(bibleState.stateHash)) {
    return bibleState.stateHash.toLowerCase();
  }
  const canonical = canonicalizeJson(bibleState);
  return sha256Hex(canonical);
}

export const verifyGenesisPrimordialAction = async (
  cpy: PrimordialModel,
  bal: PrimordialBit,
  ste?: State,
): Promise<PrimordialModel> => {
  let isSeeded = false;
  let entitiesCount = 0;
  let recordedHash = cpy.hRoot;

  if (ste && typeof ste.hunt === 'function') {
    const eventRes: any = await ste.hunt(ActDbs.QUERY_DATABASE, {
      dat: { query: "SELECT * FROM event_log WHERE sequence_number = 1 OR event_id = 'evt_genesis_000'" },
    });
    const entityRes: any = await ste.hunt(ActDbs.QUERY_DATABASE, {
      dat: { query: 'SELECT COUNT(*) as count FROM entities_live' },
    });

    const hasGenesisEvent = Boolean(eventRes?.databaseBit?.dat?.row || eventRes?.databaseBit?.dat?.record);
    entitiesCount = Number(entityRes?.databaseBit?.dat?.row?.count || 0);
    isSeeded = Boolean(hasGenesisEvent && entitiesCount > 0);

    if (hasGenesisEvent) {
      const rec = eventRes.databaseBit.dat.row || eventRes.databaseBit.dat.record;
      recordedHash = rec.state_hash || recordedHash;
    }
  } else {
    isSeeded = cpy.genesisSeeded;
  }

  cpy.genesisSeeded = isSeeded;

  await emitConsole(
    ste,
    `>> [PRIMORDIAL:VERIFY] [OK] Genesis state verified: ${isSeeded ? 'SEEDED' : 'EMPTY'} (${entitiesCount} entities online). Zero rows overwritten.`,
  );

  if (bal.slv) {
    bal.slv({
      prmBit: {
        idx: isSeeded ? 'verify-genesis-pass' : 'verify-genesis-empty',
        val: isSeeded ? 1 : 0,
        src: recordedHash,
        dat: { isSeeded, entitiesCount, hRoot: recordedHash },
      },
      hRoot: recordedHash,
    });
  }

  return cpy;
};

export const seedPrimordialAction = async (
  cpy: PrimordialModel,
  bal: PrimordialBit,
  ste?: State,
): Promise<PrimordialModel> => {
  const bibleState = bal.dat?.bibleState;
  const seriesSlug = bal.dat?.seriesSlug || cpy.activeSeriesSlug || 'default-realm';
  const seededAtMs = bal.dat?.seededAtMs || 1700000000000;

  if (!bibleState) {
    throw new Error(`[PRIMORDIAL:FATAL] Missing bibleState payload for realm: ${seriesSlug}`);
  }

  const hRoot = computeManifestHRoot(bibleState);
  cpy.hRoot = hRoot;
  cpy.activeSeriesSlug = seriesSlug;

  // 1. Idempotent Ledger Sequence Inspection
  if (ste && typeof ste.hunt === 'function') {
    const checkRes: any = await ste.hunt(ActDbs.QUERY_DATABASE, {
      dat: { query: "SELECT sequence_number, event_id FROM event_log WHERE sequence_number = 1 OR event_id = 'evt_genesis_000'" },
    });

    if (checkRes?.databaseBit?.dat?.row || checkRes?.databaseBit?.dat?.record) {
      cpy.genesisSeeded = true;
      await emitConsole(
        ste,
        `>> [PRIMORDIAL:VERIFY] [OK] Genesis already seeded. H_root: ${hRoot.slice(0, 16)}... Zero rows overwritten.`,
      );
      if (bal.slv) {
        bal.slv({
          prmBit: {
            idx: 'seed-primordial-idempotent',
            val: 1,
            src: hRoot,
            dat: { hRoot, seeded: true },
          },
          hRoot,
        });
      }
      return cpy;
    }
  }

  try {
    const operations: any[] = [];
    const entities = bibleState.entities || bibleState;
    const characters = entities.characters || {};
    const locations = entities.locations || {};
    const grievances = entities.grievances || {};
    const paranoia = entities.paranoia_clocks || entities.clocks || {};

    let characterCount = 0;
    let locationCount = 0;
    let grievanceCount = 0;
    let clockCount = 0;

    // 2. Unpack Locations into entities_live
    for (const [locId, locData] of Object.entries<any>(locations)) {
      locationCount++;
      operations.push({
        table: 'entities_live',
        action: 'INSERT',
        record: {
          entity_id: locId,
          location_id: locId,
          q: locData.q ?? 0,
          r: locData.r ?? 0,
          arms_available: 0,
          legs_available: 0,
          locomotion_tier: 'IMMOBILIZED',
          concussion_level: 0,
          wounds_json: '[]',
          held_items_json: '[]',
          worn_gear_json: '[]',
          slung_items_json: '[]',
          updated_at_ms: seededAtMs,
        },
      });
    }

    // 3. Unpack Characters into entities_live
    for (const [charId, charData] of Object.entries<any>(characters)) {
      characterCount++;
      operations.push({
        table: 'entities_live',
        action: 'INSERT',
        record: {
          entity_id: charId,
          location_id: charData.locationId || charData.location_id || 'loc_origin_hub',
          q: charData.q ?? charData.coordQ ?? 0,
          r: charData.r ?? charData.coordR ?? 0,
          arms_available: charData.armsAvailable ?? charData.arms_available ?? 2,
          legs_available: charData.legsAvailable ?? charData.legs_available ?? 2,
          locomotion_tier: charData.locomotionTier || charData.locomotion_tier || 'UNRESTRICTED',
          concussion_level: charData.concussionLevel ?? charData.concussion_level ?? 0,
          wounds_json: canonicalizeJson(charData.openWounds || charData.wounds || []),
          held_items_json: canonicalizeJson(charData.heldItems || charData.held_items || ['item_founder_token']),
          worn_gear_json: canonicalizeJson(charData.wornGear || charData.worn_gear || []),
          slung_items_json: canonicalizeJson(charData.slungItems || charData.slung_items || []),
          updated_at_ms: seededAtMs,
        },
      });
    }

    // 4. Unpack Grievances into grievance_status
    for (const [grvId, grvData] of Object.entries<any>(grievances)) {
      grievanceCount++;
      operations.push({
        table: 'grievance_status',
        action: 'INSERT',
        record: {
          grievance_id: grvId,
          initiator_id: grvData.initiatorId || grvData.instigatorId || grvData.initiator_id || 'char_wart',
          target_id: grvData.targetId || grvData.victimId || grvData.target_id || 'char_bog',
          escalation_stage: grvData.escalationStage ?? grvData.escalationDepth ?? 0,
          cooldown_cycles: grvData.cooldownCycles ?? 0,
          intensity_bps: grvData.intensityBps ?? grvData.intensity_bps ?? 2500,
          status: grvData.status || 'DORMANT',
          updated_at_ms: seededAtMs,
        },
      });
    }

    // 5. Unpack Paranoia Clocks
    for (const [clkId, clkData] of Object.entries<any>(paranoia)) {
      clockCount++;
      operations.push({
        table: 'paranoia_clocks',
        action: 'INSERT',
        record: {
          clock_id: clkId,
          entity_id: clkData.entityId || clkData.entity_id || null,
          value_bps: clkData.valueBps ?? clkData.value_bps ?? 2400,
          threshold_bps: clkData.thresholdBps ?? clkData.threshold_bps ?? 7500,
          decay_rate_bps: clkData.decayRateBps ?? clkData.decay_rate_bps ?? 50,
          updated_at_ms: seededAtMs,
        },
      });
    }

    // 6. Mint Sequence #1 Genesis Event Envelope
    const genesisEventPayload = {
      genesisTick: 0,
      seriesSlug,
      hRoot,
      characterCount,
      locationCount,
      grievanceCount,
      clockCount,
    };

    const genesisEnvelope = {
      eventId: 'evt_genesis_000',
      sequenceId: 1,
      tick: 0,
      eventType: 'GENESIS_SEEDED',
      seed: 'genesis_seed_0',
      payload: genesisEventPayload,
      prevEventHash: GENESIS_HASH_64,
      stateHash: hRoot,
      timestampMs: seededAtMs,
    };

    const eventHash = computeChainedEventHash(GENESIS_HASH_64, genesisEnvelope);

    operations.push({
      table: 'event_log',
      action: 'INSERT',
      record: {
        sequence_number: 1,
        event_id: genesisEnvelope.eventId,
        event_type: genesisEnvelope.eventType,
        tick: 0,
        payload_json: canonicalizeJson(genesisEnvelope.payload),
        state_hash: hRoot,
        prev_hash: GENESIS_HASH_64,
        seed: genesisEnvelope.seed,
        timestamp_ms: seededAtMs,
      },
    });

    // 7. Execute Atomic Batch Insert in 02.database.unit
    if (ste && typeof ste.hunt === 'function') {
      const dbRes: any = await ste.hunt(ActDbs.EXEC_DATABASE, {
        dat: { operations },
      });

      if (dbRes?.databaseBit?.val === 0) {
        throw new Error(`[DB_INSERT_REJECTED] ${dbRes?.databaseBit?.dat?.error || 'Batch seeding rejected'}`);
      }

      // 8. Monotonic Sequence Counter Advancement in 14.sequence.unit
      await ste.hunt(ActSeq.COMMIT_SEQUENCE, {
        val: 1,
        dat: {
          sequenceNumber: 1,
          eventId: genesisEnvelope.eventId,
          eventHash,
          seriesSlug,
        },
      });
    }

    cpy.genesisSeeded = true;

    const result: GenesisSeedingResult = {
      seeded: true,
      seriesSlug,
      hRoot,
      eventHash,
      entitiesCount: characterCount + locationCount,
      grievancesCount,
      clocksCount: clockCount,
      seededAtMs,
    };

    await emitConsole(
      ste,
      `>> [PRIMORDIAL:SEEDED] [OK] Genesis (S_0) minted! H_root: ${hRoot.slice(0, 16)}... | EventHash: ${eventHash.slice(0, 16)}...`,
    );
    await emitConsole(
      ste,
      `>> [DATABASE:TABLES] Seeded ${result.entitiesCount} entities, ${result.grievancesCount} grievances, ${result.clocksCount} paranoia clocks.`,
    );
    await emitConsole(
      ste,
      '>> [SEQUENCE:COMMIT] Monotonic sequence advanced to #1 (Total: 1)',
    );

    if (bal.slv) {
      bal.slv({
        prmBit: {
          idx: 'seed-primordial-pass',
          val: 1,
          src: hRoot,
          dat: { result, hRoot },
        },
        result,
        hRoot,
      });
    }
  } catch (err: any) {
    const errorMsg = err.message || String(err);
    await emitConsole(ste, `:: [PRIMORDIAL:SEED_FAIL] ${errorMsg}`);
    if (bal.slv) {
      bal.slv({
        prmBit: {
          idx: 'seed-primordial-fail',
          val: 0,
          dat: { error: errorMsg },
        },
        error: errorMsg,
      });
    }
  }

  return cpy;
};

import clone from 'clone-deep';
import * as Act from './primordial.action.js';
import { PrimordialModel } from './primordial.model.js';
import * as Buzz from './primordial.buzzer.js';
import type PrimordialBit from './fce/primordial.bit.js';
import type State from '../99.core/state.js';

export const reducePrimordial = async (
  model: PrimordialModel = new PrimordialModel(),
  action: string,
  bit: PrimordialBit,
  ste: State,
): Promise<PrimordialModel> => {
  const cpy = clone(model);
  switch (action) {
    case Act.INIT_PRIMORDIAL:
      return await Buzz.initPrimordial(cpy, bit, ste);
    case Act.UPDATE_PRIMORDIAL:
      return await Buzz.updatePrimordial(cpy, bit, ste);
    case Act.COMPUTE_HROOT_PRIMORDIAL:
      return Buzz.computeHrootPrimordialAction(cpy, bit, ste);
    case Act.SEED_PRIMORDIAL:
      return await Buzz.seedPrimordialAction(cpy, bit, ste);
    case Act.VERIFY_GENESIS_PRIMORDIAL:
      return await Buzz.verifyGenesisPrimordialAction(cpy, bit, ste);
    default:
      return cpy;
  }
};

export default reducePrimordial;
